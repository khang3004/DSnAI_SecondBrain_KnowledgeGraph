\csbranch{2}{B}{2\;\; RIDGE ($\ell_2$) vs LASSO ($\ell_1$) $-$ Prior \& Geometry}{%
  \renewcommand{\arraystretch}{1.15}\scriptsize
  \begin{tabular}{@{}>{\raggedright\arraybackslash}p{1.25cm}>{\raggedright\arraybackslash}p{3.1cm}>{\raggedright\arraybackslash}p{3.1cm}@{}}
    & \textbf{\textcolor{csB}{Ridge ($\ell_2$)}} & \textbf{\textcolor{csB}{Lasso ($\ell_1$)}}\\ \hline
    Penalty & $\lambda\norm{\vecb w}_2^2$ (Gaussian prior) & $\lambda\norm{\vecb w}_1$ (Laplace prior)\\
    Solution & $(\mat\Phi\T\mat\Phi+\lambda\eye_D)\inv\mat\Phi\T\vecb y$ & $\mathcal S_\lambda(c_j)=\sign(c_j)(|c_j|-\lambda)_+$\\
    SVD filter & $\frac{\sigma_j^2}{\sigma_j^2+\lambda}$ (smooth damping) & Cross-polytope corners $\Rightarrow w_j=0$\\
    Bayes pred. & $\Normal(\vecb m_N\T\vecb\phi_*,\sigma^2+\vecb\phi_*\T\mat S_N\vecb\phi_*)$ & MAP only (non-Gaussian post.)\\
  \end{tabular}\par\vspace{2pt}
  \small\textbf{Elastic-Net:} $\lambda_1\norm{\vecb w}_1+\frac{\lambda_2}{2}\norm{\vecb w}_2^2$ groups correlated features.}

\csbranch{3}{C}{3\;\; KEY THEOREM $-$ Canonical Exp-Family GLM}{%
  \csfm{p(y_n\mid\theta_n,\phi)=h(y_n,\phi)\exp\!\left(\frac{y_n\theta_n-A(\theta_n)}{\phi}\right)}
  \begin{csbul}
    \item \textbf{Cumulants:} $\mu_n=\E[y_n]=A'(\theta_n)$,\; $\Var(y_n)=\phi A''(\theta_n)$
    \item \textbf{Canonical link} $g=(A')\inv\implies \theta_n=\eta_n=\vecb w\T\vecb\phi_n$
    \item \textbf{Universal Gradient \& Hessian} ($W_{nn}=A''(\eta_n)$):
      \csfm{\grad_{\vecb w}\mathrm{NLL}=\tfrac{1}{\phi}\mat\Phi\T(\vecb\mu-\vecb y),\quad \mat H(\vecb w)=\tfrac{1}{\phi}\mat\Phi\T\mat W\mat\Phi\succeq\zeros}
  \end{csbul}}

    \textbf{\textcolor{csD}{Family}} & \textbf{\textcolor{csD}{$A(\eta)$}} & \textbf{\textcolor{csD}{Link $g(\mu)$}} & \textbf{\textcolor{csD}{Mean $\mu=A'(\eta)$ / $W_{nn}$}}\\ \hline

    \item \textbf{Probit vs Logit:} Probit tails $\mathcal O(e^{-\eta^2/2})$ $\Rightarrow$ outlier-sensitive
  \end{csbul}}

\csnote{8}{A}{TRAPS $-$ before you trust the output}{%
  \begin{csbul}

    \item \textbf{Unscaled / normal eqs:} scale $\vecb\phi_j$ (not $w_0$); avoid $(\mat\Phi\T\mat\Phi)\inv$

\begin{cscard}{csA}{FORMULAS TO REMEMBER}
  \begin{csbul}
    \item Ridge ($\ell_2$) \& Effective DoF:\csfm{\hat{\vecb w}_{\mathrm{Ridge}}=(\mat\Phi\T\mat\Phi+\lambda\eye_D)\inv\mat\Phi\T\vecb y,\quad \mathrm{dof}(\lambda)=\sum_{j=1}^r\frac{\sigma_j^2}{\sigma_j^2+\lambda}}
    \item Bayesian Linear Posterior ($\lambda=\sigma^2/\tau^2$):\csfm{\mat S_N\inv=\mat S_0\inv+\frac{1}{\sigma^2}\mat\Phi\T\mat\Phi,\quad \vecb m_N=\mat S_N\!\left(\mat S_0\inv\vecb m_0+\frac{1}{\sigma^2}\mat\Phi\T\vecb y\right)}
    \item Lasso KKT \& Soft-Thresholding ($\mat\Phi\T\mat\Phi=\eye_D$):\csfm{\hat w_j=\mathcal S_\lambda(c_j)=\sign(c_j)\max(0,|c_j|-\lambda),\quad |\vecb\varphi_j\T\vecb r|\le\lambda}
    \item Canonical GLM Score \& Fisher Matrix:\csfm{\grad_{\vecb w}\mathrm{NLL}=\frac{1}{\phi}\mat\Phi\T(\vecb\mu-\vecb y),\quad \mathcal I(\vecb w)=\frac{1}{\phi}\mat\Phi\T\mat W\mat\Phi}
    \item Numerically Stable Log-Sum-Exp / Softplus:\csfm{\log\sum_k e^{\eta_k}=m+\log\sum_k e^{\eta_k-m},\;\; \softplus(\eta)=\max(\eta,0)+\log(1+e^{-|\eta|})}
  \end{csbul}
\end{cscard}

\begin{cscard}{csB}{IRLS \& COORDINATE DESCENT RECIPES}
  \textbf{IRLS (Canonical GLM Newton--Raphson):}
  \begin{enumerate}[leftmargin=1.2em, itemsep=1pt, topsep=1pt, parsep=0pt]
    \item Compute $\vecb\eta=\mat\Phi\vecb w^{(t)}$, $\vecb\mu=A'(\vecb\eta)$, $\mat W=\diag(A''(\vecb\eta))$
    \item Working response: $\vecb z=\vecb\eta+\mat W\inv(\vecb y-\vecb\mu)$
    \item Solve $(\mat\Phi\T\mat W\mat\Phi+\lambda\eye_D)\vecb w^{(t+1)}=\mat\Phi\T\mat W\vecb z$
  \end{enumerate}
  \vspace{2pt}
  \textbf{Lasso Coordinate Descent ($\norm{\vecb\varphi_j}_2^2=N$):}
  \begin{enumerate}[leftmargin=1.2em, itemsep=1pt, topsep=1pt, parsep=0pt]
    \item Partial residual: $\rho_j=\vecb\varphi_j\T(\vecb r+w_j\vecb\varphi_j)$
    \item Update $w_j^{\mathrm{new}}=\frac{1}{N}\mathcal S_\lambda(\rho_j)$ and refresh $\vecb r$
  \end{enumerate}
\end{cscard}

\begin{cscard}{csE}{SHRINKAGE \& TAIL CHEAT-LINE}
  \begin{csbul}
    \item \textbf{Ridge attenuation:} singular mode $j$ scaled by $\sigma_j^2/(\sigma_j^2+\lambda)$
    \item \textbf{Lasso deadzone:} feature $j$ zeroed iff $|\vecb\varphi_j\T(\vecb y-\mat\Phi\hat{\vecb w})|\le\lambda$
    \item \textbf{Softmax identifiability:} $\ell_2$ penalty enforces $\sum_{c=1}^C\hat w_{cj}=0$
  \end{csbul}
\end{cscard}

\begin{cscard}{csG}{PITFALL CHECKLIST}
  \begin{csbul}
    \item \textbf{Complete separation in Logistic Regression:} when classes are linearly separable, $\mathrm{NLL}\to 0$ as $\norm{\vecb w}_2\to\infty$ and IRLS weights $W_{nn}=\mu_n(1-\mu_n)\to 0$; always include $\lambda>0$
    \item \textbf{Normal equations square conditioning:} $\kappa(\mat\Phi\T\mat\Phi)=\kappa(\mat\Phi)^2$; solve OLS/Ridge via thin QR or SVD
    \item \textbf{Scale sensitivity of $\ell_1/\ell_2$:} standardize columns of $\mat\Phi$ to zero mean and unit variance before fitting, and never penalize the bias $w_0$
    \item \textbf{Float64 exp overflow:} never evaluate $1/(1+e^{-\eta})$ or $\log(1+e^\eta)$ raw; use branch-split sigmoid and Log-Sum-Exp
    \item \textbf{OLS on count data:} predicts negative counts $\hat y<0$ and miscalibrates variance $\Var(y)=\mu$; use Poisson GLM
  \end{csbul}
\end{cscard}

\begin{cscard}{csD}{10-SECOND VISUAL RECAP}
  \sffamily\footnotesize
  \textbf{Every canonical GLM} maps $\vecb x\mapsto\eta=\vecb w\T\vecb\phi(\vecb x)\mapsto\mu=A'(\eta)$.
  Because $g=(A')\inv$, the gradient is universally $\frac{1}{\phi}\mat\Phi\T(\vecb\mu-\vecb y)$ and the Hessian $\frac{1}{\phi}\mat\Phi\T\mat W\mat\Phi$ is always convex ($\succeq\zeros$).
  \textbf{Ridge} rotates to the SVD basis and damps noisy axes; \textbf{Lasso} clips small correlations to exact zero at the $\ell_1$ diamond corners.
\end{cscard}
