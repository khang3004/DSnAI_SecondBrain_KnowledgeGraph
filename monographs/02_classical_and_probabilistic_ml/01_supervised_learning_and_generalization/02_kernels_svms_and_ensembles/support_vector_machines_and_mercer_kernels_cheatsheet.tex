\csbranch{1}{A}{1\;\; WHAT $-$ Hard/Soft Margin \& Hinge Loss}{%
  \csfm{\min_{\vecb w,b,\vecb\xi}\;\frac{1}{2}\norm{\vecb w}_2^2+C\sum_{i=1}^N\xi_i\quad\text{s.t.}\;\;y_i(\inner{\vecb w}{\vecb x_i}+b)\ge 1-\xi_i,\;\xi_i\ge 0}
  \begin{csbul}
    \item \textbf{Signed dist.:} $r_i=\frac{\inner{\vecb w}{\vecb x_i}+b}{\norm{\vecb w}_2}$;\; slab width $= \frac{2}{\norm{\vecb w}_2}$
    \item \textbf{Hinge ERM} ($\lambda=\frac{1}{2C}$): $\min_{\vecb w,b}\frac{1}{2C}\norm{\vecb w}_2^2+\sum_i\max(0,1-y_if(\vecb x_i))$
    \item \textbf{$\nu$-SVM:} $\nu\in(0,1]$ bounds margin errors \& SV fraction
  \end{csbul}}

\csbranch{2}{B}{2\;\; WOLFE DUAL QP \& KKT 3-REGIME PARTITION}{%
  \csfm{\max_{\vecb\alpha\in[0,C]^N}\sum_{i=1}^N\alpha_i-\frac{1}{2}\sum_{i,j=1}^N\alpha_i\alpha_jy_iy_jk(\vecb x_i,\vecb x_j)\;\;\text{s.t.}\;\sum_{i=1}^N\alpha_iy_i=0}
  \renewcommand{\arraystretch}{1.12}\scriptsize
  \begin{tabular}{@{}>{\raggedright\arraybackslash}p{1.65cm}>{\raggedright\arraybackslash}p{2.2cm}>{\raggedright\arraybackslash}p{3.55cm}@{}}
    \textbf{\textcolor{csB}{Multiplier}} & \textbf{\textcolor{csB}{Margin / Slack}} & \textbf{\textcolor{csB}{Geometric Role}}\\ \hline
    $\alpha_i=0$ & $y_if(\vecb x_i)\ge 1,\,\xi_i^*=0$ & Non-SV (outside slab, ignored)\\
    $0<\alpha_i<C$ & $y_if(\vecb x_i)=1,\,\xi_i^*=0$ & \textbf{Free SV} ($\mathcal M$): on $\pm 1$, fixes $b^*$\\
    $\alpha_i=C$ & $y_if(\vecb x_i)=1-\xi_i^*$ & \textbf{Bounded SV}: inside slab ($\xi_i^*>1$ err)\\
  \end{tabular}}

\csbranch{3}{C}{3\;\; KEY THEOREMS $-$ Mercer \& Representer}{%
  \begin{minipage}[t]{0.60\linewidth}
    \begin{csbul}
      \item \textbf{Mercer PSD:} $k(\vecb x,\vecb x')=\inner{\phi(\vecb x)}{\phi(\vecb x')}_{\Hilbert_k}$ iff Gram $\mat K\succeq\zeros$ ($\vecb c\T\mat K\vecb c\ge 0$)
      \item \textbf{Spectral:} $k(\vecb x,\vecb x')=\sum_{m=1}^\infty\lambda_m\psi_m(\vecb x)\psi_m(\vecb x')$
      \item \textbf{Representer Theorem:} $\min_{f\in\Hilbert_k}\sum_i\ell(y_i,f(\vecb x_i))+\Omega(\norm{f}_{\Hilbert_k})$ has exact finite-span solution:
    \end{csbul}
    \csfm{f^*(\vecb x)=\sum_{i\in\mathcal S}\alpha_iy_i\,k(\vecb x_i,\vecb x)+b^*}
  \end{minipage}\hfill

\csbranch{4}{D}{4\;\; CONNECTIONS $-$ Hulls, KRR \& Surrogate Losses}{%
  \begin{csbul}
    \item \textbf{Dual Convex Hulls:} hard-margin SVM finds closest points between class convex hulls; $C<\infty$ shrinks them to \emph{reduced convex hulls}
    \item \textbf{SVM vs KRR:} Kernel Ridge $(\mat K+\lambda\eye_N)\inv\vecb y$ is closed-form but \emph{dense} ($\alpha_i\ne 0$); SVM hinge flat region ($t\ge 1$) gives \emph{sparse} support vectors
    \item \textbf{Standard Mercer Kernels:} Linear $\inner{\vecb x}{\vecb x'}$; Poly $(\inner{\vecb x}{\vecb x'}+c_0)^d$; Gaussian RBF $\exp(-\gamma\norm{\vecb x-\vecb x'}_2^2)$ with $\gamma=(2\ell^2)^{-1}$ ($\dim\Hilbert_k=\infty$)
    \item \textbf{Template Matching:} RBF SVM places local bumps of width $\ell$ at SVs
  \end{csbul}}

\begin{cscard}{csA}{FORMULAS TO REMEMBER}
  \begin{csbul}
    \item Primal Weight \& Support Expansion:\csfm{\vecb w^*=\sum_{i\in\mathcal S}\alpha_iy_i\phi(\vecb x_i),\quad f(\vecb x)=\sum_{i\in\mathcal S}\alpha_iy_ik(\vecb x_i,\vecb x)+b^*}
    \item KKT Complementary Slackness:\csfm{\alpha_i\big[y_if(\vecb x_i)-1+\xi_i^*\big]=0,\qquad (C-\alpha_i)\xi_i^*=0}
    \item Numerically Stable Bias Recovery ($\mathcal M=\{m:0<\alpha_m<C\}$):\csfm{b^*=\frac{1}{|\mathcal M|}\sum_{m\in\mathcal M}\!\left(y_m-\sum_{i\in\mathcal S}\alpha_iy_ik(\vecb x_i,\vecb x_m)\right)}
    \item Gaussian RBF \& Polynomial Kernels:\csfm{k_{\mathrm{RBF}}(\vecb x,\vecb x')=\exp\!\big(-\gamma\norm{\vecb x-\vecb x'}_2^2\big),\;\; k_{\mathrm{poly}}=(\inner{\vecb x}{\vecb x'}+c_0)^d}
  \end{csbul}
\end{cscard}

\begin{cscard}{csB}{PLATT SMO 2-VARIABLE UPDATE RECIPE}
  \begin{enumerate}[leftmargin=1.2em, itemsep=1pt, topsep=1pt, parsep=0pt]
    \item Pick KKT-violating $\alpha_i$, then $\alpha_j$ maximizing $|E_i-E_j|$ ($E_k=f(\vecb x_k)-y_k$)
    \item Box bounds preserving $\alpha_iy_i+\alpha_jy_j=\mathrm{const}$:
      $y_i\ne y_j\Rightarrow L=\max(0,\alpha_j-\alpha_i),\,H=\min(C,C+\alpha_j-\alpha_i)$;
      $y_i=y_j\Rightarrow L=\max(0,\alpha_i+\alpha_j-C),\,H=\min(C,\alpha_i+\alpha_j)$
    \item Curvature $\eta=k_{ii}+k_{jj}-2k_{ij}>0$: clip $\alpha_j^{\mathrm{new}}=\mathrm{clip}(\alpha_j+\frac{y_j(E_i-E_j)}{\eta},L,H)$
    \item Back-substitute $\alpha_i^{\mathrm{new}}=\alpha_i+y_iy_j(\alpha_j-\alpha_j^{\mathrm{new}})$ and update $b, E_k$
  \end{enumerate}
\end{cscard}

\begin{cscard}{csE}{HYPERPARAMETER $(C,\gamma)$ CHEAT-LINE}
  \begin{csbul}
    \item \textbf{Large $C$ ($\lambda=\frac{1}{2C}\downarrow$):} narrow margin, penalizes slack heavily $\to$ higher variance
    \item \textbf{Small $C$ ($\lambda\uparrow$):} wide margin, allows slack $\xi_i>0$ $\to$ smoother boundary
    \item \textbf{Large $\gamma=(2\ell^2)^{-1}$:} spiky local RBF bumps, $N_{\mathrm{sv}}/N\to 1$ (memorization)
    \item \textbf{Small $\gamma$:} flat wide RBF kernels, approaches linear classifier
  \end{csbul}
\end{cscard}

\begin{cscard}{csG}{PITFALL CHECKLIST}
  \begin{csbul}
    \item \textbf{Quadratic $\mathcal O(N^2)$ Gram matrix bottleneck:} materializing $\mat K\in\R^{N\times N}$ at $N=10^5$ takes $80\text{ GB}$ RAM; use cached SMO or primal linear/RFF approximations
    \item \textbf{Skipping feature standardization:} a single high-variance coordinate dominates $\norm{\vecb x-\vecb x'}_2^2$, collapsing RBF similarities to $0$
    \item \textbf{Recovering $b^*$ from a single support vector:} numerical roundoff in $\alpha_i$ corrupts single-point $b^*$; always average over all free support vectors $0<\alpha_m<C$
    \item \textbf{Fitting Platt scaling on the training set:} SVMs push training points to $|f(\vecb x_i)|\ge 1$, biasing in-sample sigmoid fits; calibrate strictly out-of-fold
    \item \textbf{Using non-Mercer similarity matrices:} if $\lambda_{\min}(\mat K)<0$, the dual QP is non-concave and unbounded
  \end{csbul}
\end{cscard}

\begin{cscard}{csD}{10-SECOND VISUAL RECAP}
  \sffamily\footnotesize
  \textbf{SVM} finds the thickest separating slab of width $2/\norm{\vecb w}_2$.
  Passing to the \textbf{Wolfe dual} replaces $\vecb w$ with $\sum_{i\in\mathcal S}\alpha_iy_i\vecb x_i$, where \textbf{KKT complementary slackness} zeroes out every safe point ($y_if(\vecb x_i)>1\Rightarrow\alpha_i=0$).
  Replacing $\inner{\vecb x_i}{\vecb x_j}$ with a PSD \textbf{Mercer kernel} $k(\vecb x_i,\vecb x_j)$ computes this maximum-margin hyperplane in an infinite-dimensional RKHS $\Hilbert_k$.
\end{cscard}
