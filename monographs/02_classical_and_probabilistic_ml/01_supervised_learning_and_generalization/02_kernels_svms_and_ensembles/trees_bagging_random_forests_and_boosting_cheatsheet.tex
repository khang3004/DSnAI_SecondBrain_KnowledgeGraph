\csbranch{1}{A}{1\;\; WHAT $-$ CART Axis-Aligned Partitioning}{%
  \begin{minipage}[t]{0.60\linewidth}
    \csfm{F_m(\vecb x)=\sum_{j=1}^{J_m}w_{jm}\,\mathbb I(\vecb x\in R_{jm})}
    \begin{csbul}
      \item Greedy: $\argmin_{d,t}\big[\frac{|\mathcal D_L|}{|\mathcal D_i|}c(\mathcal D_L)+\frac{|\mathcal D_R|}{|\mathcal D_i|}c(\mathcal D_R)\big]$
      \item \textbf{Reg.:} $c(\mathcal D_i)=\frac{1}{|\mathcal D_i|}\sum(y_n-\bar y_i)^2$, leaf $\bar y_i$
      \item \textbf{Classif.:} Gini $1-\sum\hat\pi_{ic}^2$; Entr.\ $-\sum\hat\pi_{ic}\log\hat\pi_{ic}$
      \item \textbf{Pruning:} $R_\alpha(T)=R(T)+\alpha|T|$
    \end{csbul}
  \end{minipage}\hfill

\csbranch{2}{B}{2\;\; BAGGING vs RANDOM FORESTS $-$ Variance Floor}{%
  \csfm{\Var\!\left(\frac{1}{B}\sum_{b=1}^B F_b(\vecb x)\right)=\rho\sigma^2+\frac{1-\rho}{B}\sigma^2\xrightarrow{B\to\infty}\rho\sigma^2}
  \begin{csbul}
    \item \textbf{Bootstrap OOB:} $\Prob(i\notin\mathcal D^{(b)})=(1-\frac{1}{N})^N\to e^{-1}\approx 36.8\%$ free validation
    \item \textbf{Why Bagging plateaus:} dominant features split near root in every tree $\Rightarrow$ high correlation $\rho$
    \item \textbf{Random Forest decorrelation:} sample $m$ features per node ($m=\lfloor\sqrt D\rfloor$ classif., $\lfloor D/3\rfloor$ reg.) $\Rightarrow$ slashes $\rho\sigma^2$ floor!
  \end{csbul}}

\csbranch{3}{C}{3\;\; KEY THEOREM $-$ 2nd-Order GBDT (XGBoost/LightGBM)}{%
  \csfm{\tilde{\mathcal L}^{(m)}=\sum_{j=1}^J\!\left[G_{jm}w_j+\frac{1}{2}(H_{jm}+\lambda)w_j^2\right]+\gamma J}
  \begin{csbul}
    \item $G_{jm}=\sum_{i\in I_j}\partial_f\ell(y_i,f_{m-1})$,\; $H_{jm}=\sum_{i\in I_j}\partial_f^2\ell(y_i,f_{m-1})\ge 0$
    \item \textbf{Newton Leaf Weight:} $w_j^*=-\frac{G_{jm}}{H_{jm}+\lambda}$ (curvature preconditioned)
    \item \textbf{Exact Split Gain:}
      \csfm{\mathrm{Gain}=\tfrac{1}{2}\!\left[\frac{G_L^2}{H_L+\lambda}+\frac{G_R^2}{H_R+\lambda}-\frac{(G_L+G_R)^2}{H_L+H_R+\lambda}\right]-\gamma}
  \end{csbul}}

\csbranch{4}{D}{4\;\; CONNECTIONS $-$ AdaBoost \& Functional Gradient}{%
  \begin{csbul}
    \item \textbf{AdaBoost.M1} = forward stagewise additive modeling on exp.\ loss $e^{-\tilde yf}$:
      \csfm{\beta_m=\tfrac{1}{2}\log\frac{1-\mathrm{err}_m}{\mathrm{err}_m},\quad \omega_{i,m+1}\propto\omega_{i,m}e^{2\beta_m\mathbb I(\tilde y_i\ne F_m(\vecb x_i))}}
    \item \textbf{1st-Order GBM:} fits tree $F_m$ to pseudo-residuals $r_{im}=-\partial_f\ell(y_i,f_{m-1}(\vecb x_i))$, updates $f_m=f_{m-1}+\nu F_m$
    \item \textbf{Scale Invariance:} monotone feature transforms $x_d\mapsto\psi(x_d)$ never change split thresholds
  \end{csbul}}

\csnote{8}{A}{TRAPS $-$ before you trust the output}{%
  \begin{csbul}
    \item \textbf{Zero extrapolation:} leaf constants $w_{jm}$ clamp flat outside training $[x_{\min},x_{\max}]$
    \item \textbf{Unshrunk boosting ($\nu=1$):} $M\to\infty$ overfits noise (esp.\ AdaBoost on outliers)
    \item \textbf{High-card.\ / OOB:} high-$K$ splits overfit; OOB leaks groups
  \end{csbul}}

\begin{cscard}{csA}{FORMULAS TO REMEMBER}
  \begin{csbul}
    \item CART Impurity Criteria ($\hat\pi_{ic}$ class proportions):\csfm{G_i=1-\sum_{c=1}^C\hat\pi_{ic}^2,\qquad H_i=-\sum_{c=1}^C\hat\pi_{ic}\log\hat\pi_{ic}}
    \item Bagging / RF Variance Decomposition:\csfm{\Var(\bar F_B)=\rho\sigma^2+\frac{1-\rho}{B}\sigma^2,\quad \Prob(i\notin\mathcal D^{(b)})\to e^{-1}\approx 0.368}
    \item AdaBoost.M1 Weight \& Step ($\tilde y_i\in\{\pm 1\}$):\csfm{\beta_m=\frac{1}{2}\log\frac{1-\mathrm{err}_m}{\mathrm{err}_m},\quad \omega_{i,m+1}=\omega_{i,m}e^{2\beta_m\mathbb I(\tilde y_i\ne F_m(\vecb x_i))}e^{-\beta_m}}
    \item XGBoost Optimal Leaf \& Split Gain:\csfm{w_j^*=-\frac{G_{jm}}{H_{jm}+\lambda},\;\; \mathrm{Gain}=\frac{1}{2}\!\left[\frac{G_L^2}{H_L+\lambda}+\frac{G_R^2}{H_R+\lambda}-\frac{(G_L+G_R)^2}{H_L+H_R+\lambda}\right]-\gamma}
  \end{csbul}
\end{cscard}

\begin{cscard}{csB}{HISTOGRAM GBDT RECIPE (XGBOOST / LIGHTGBM)}
  \begin{enumerate}[leftmargin=1.2em, itemsep=1pt, topsep=1pt, parsep=0pt]
    \item Bin features into $K_{\mathrm{bin}}\approx 256$ quantiles; init $f_0=\argmin_c\sum_i\ell(y_i,c)$
    \item For $m=1\dots M$: compute $g_{im}=\partial_f\ell(y_i,f_{m-1})$, $h_{im}=\partial_f^2\ell(y_i,f_{m-1})$
    \item Accumulate bin histograms $(G_{d,k},H_{d,k})$ and scan prefix sums for max $\mathrm{Gain}>0$
    \item Set leaf weights $w_j^*=-G_{jm}/(H_{jm}+\lambda)$ and update $f_m=f_{m-1}+\nu F_m$
  \end{enumerate}
\end{cscard}

\begin{cscard}{csC}{DECISION GUIDE}
  \renewcommand{\arraystretch}{1.22}\small
  \begin{tabular}{@{}>{\raggedright\arraybackslash}p{3.2cm}>{\raggedright\arraybackslash}p{4.6cm}@{}}
    \textbf{If you have\ldots} & \textbf{use\ldots}\\ \hline
    structured tabular data, max accuracy & Hist-GBDT ($\nu\le 0.05$, early stopping)\\
    heavy label noise / outliers & Random Forest or Huber/Log-loss GBDT (avoid AdaBoost)\\
    trending time-series target & Detrend with linear model before fitting trees\\
    high-cardinality categoricals & Fisher-sorted histograms or out-of-fold target encoding\\
    need well-calibrated $p(y\mid\vecb x)$ & GBDT log-loss + isotonic / Platt calibration\\
  \end{tabular}
\end{cscard}

\begin{cscard}{csE}{ENSEMBLE SCALING CHEAT-LINE}
  \begin{csbul}
    \item \textbf{RF feature subsampling:} $m=\lfloor\sqrt D\rfloor$ (classif.) or $\lfloor D/3\rfloor$ (reg.)
    \item \textbf{RF tree depth:} grow deep unpruned trees ($\mathrm{Bias}\approx 0$, averaging kills $\Var$)
    \item \textbf{GBDT tree depth:} shallow $d_{\max}\in[3,8]$ ($2^d$-way feature interactions)
    \item \textbf{Curvature damping:} $\lambda$ in $H_{jm}+\lambda$ prevents step blow-up when $p_i(1-p_i)\to 0$
  \end{csbul}
\end{cscard}

\begin{cscard}{csG}{PITFALL CHECKLIST}
  \begin{csbul}
    \item \textbf{Staircase extrapolation failure:} trees predict piecewise constants $w_{jm}$ and cannot extrapolate trends outside the training bounding box
    \item \textbf{In-sample target-encoding leakage:} computing category mean targets on full training data leaks $y_i$ into splits; use strictly out-of-fold or ordered statistics
    \item \textbf{Unshrunk boosting ($\nu=1, M\to\infty$):} unlike RF, adding more boosted trees eventually memorizes noise; always pair $\nu\in[0.01,0.1]$ with validation early stopping
    \item \textbf{Trusting OOB error under group/temporal correlation:} bootstrap samples leak correlated rows across in-bag and OOB sets; use group/time-blocked CV
    \item \textbf{Impurity feature-importance bias:} raw Gini/gain importance inflates continuous and high-cardinality features; verify with permutation importance or SHAP
  \end{csbul}
\end{cscard}

\begin{cscard}{csD}{10-SECOND VISUAL RECAP}
  \sffamily\footnotesize
  \textbf{Single CART} partitions $\R^D$ into axis-aligned boxes (zero scaling needed, but high variance).
  \textbf{Bagging} averages $B$ bootstrap trees down to the correlation floor $\rho\sigma^2$, and \textbf{Random Forests} lower $\rho$ by subsampling $m$ features per split.
  \textbf{GBDT} starts with shallow high-bias trees and sequentially takes Newton steps $-G_{jm}/(H_{jm}+\lambda)$ in function space.
\end{cscard}
