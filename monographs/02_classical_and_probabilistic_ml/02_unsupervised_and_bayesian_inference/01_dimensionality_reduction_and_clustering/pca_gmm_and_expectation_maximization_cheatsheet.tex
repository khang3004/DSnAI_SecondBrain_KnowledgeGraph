\csbranch{1}{A}{1\;\; PCA $-$ three equivalent views}{%
  \begin{minipage}[t]{0.60\linewidth}
    \csfm{\mat S = \frac{1}{N}\tilde{\mat X}\T\tilde{\mat X} = \mat U\mat\Lambda\mat U\T}
    \begin{csbul}
      \item \textbf{Max variance:} $\max_{\norm{\vecb b}=1}\vecb b\T\mat S\vecb b = \lambda_1$
      \item \textbf{Min recon.\ error:} $J_M = \sum_{i=M+1}^D \lambda_i$
      \item \textbf{SVD link:} $\tilde{\mat X}=\mat U_X\mat\Sigma_X\mat V_X\T$, $\lambda_k=\sigma_k^2/N$
      \item \textbf{Dual Gram ($N\!\ll\!D$):} $\frac{1}{N}\tilde{\mat X}\tilde{\mat X}\T$ in $\mathcal O(N^3)$
    \end{csbul}
  \end{minipage}\hfill
  \begin{minipage}[t]{0.37\linewidth}\centering\vspace{4pt}

\csbranch{2}{B}{2\;\; PPCA \& FACTOR ANALYSIS $-$ generative subspace}{%
  \csfm{\vecb z_n\sim\Normal(\zeros,\eye_M),\qquad \vecb x_n\mid\vecb z_n\sim\Normal(\mat W\vecb z_n+\vecb\mu,\,\mat\Psi)}
  \begin{csbul}
    \item \textbf{Marginal:} $\vecb x_n\sim\Normal(\vecb\mu,\mat C)$ with $\mat C=\mat W\mat W\T+\mat\Psi$
    \item \textbf{Woodbury:} $\mat C\inv=\sigma^{-2}\eye_D-\sigma^{-2}\mat W\mat M\inv\mat W\T$, $\mat M=\mat W\T\mat W+\sigma^2\eye_M$
    \item \textbf{PPCA ($\mat\Psi=\sigma^2\eye_D$) MLE:} $\mat W_{\mathrm{ML}}=\mat U_M(\mat\Lambda_M-\sigma^2\eye_M)^{1/2}\mat R$, $\sigma_{\mathrm{ML}}^2=\frac{1}{D-M}\sum_{i>M}\lambda_i$; limit $\sigma^2\!\to\!0$ recovers standard PCA
    \item \textbf{FA ($\mat\Psi=\diag(\vecb\psi)$):} scale-invariant; fitted via EM in $\mathcal O(TNDM)$
  \end{csbul}}

\csbranch{3}{C}{3\;\; GENERAL EM $-$ ELBO \& monotone ascent}{%
  \begin{minipage}[t]{0.59\linewidth}
    \csfm{\log p(\mat X\mid\vecb\theta) = \mathcal{L}(q,\vecb\theta) + \KL{q(\mat Z)}{p(\mat Z\mid\mat X,\vecb\theta)}}
    \begin{csbul}
      \item \textbf{E-step:} $q^{(t+1)}\gets p(\mat Z\mid\mat X,\vecb\theta^{(t)})$ ($\mathrm{KL}=0$, tight contact at $\vecb\theta^{(t)}$)
      \item \textbf{M-step:} $\vecb\theta^{(t+1)}\gets\argmax_{\vecb\theta} Q(\vecb\theta;\vecb\theta^{(t)})$
      \item \textbf{Monotone:} $\log p(\mat X\mid\vecb\theta^{(t+1)})\ge\log p(\mat X\mid\vecb\theta^{(t)})$
    \end{csbul}
  \end{minipage}\hfill
  \begin{minipage}[t]{0.38\linewidth}\centering\vspace{2pt}
    \begin{tikzpicture}[scale=0.45, every node/.style={font=\tiny}]

\csbranch{4}{D}{4\;\; GMM vs.\ K-MEANS $-$ soft to hard limit}{%
  \csfm{r_{nk} = \frac{\pi_k\Normal(\vecb x_n\mid\vecb\mu_k,\mat\Sigma_k)}{\sum_{j=1}^K \pi_j\Normal(\vecb x_n\mid\vecb\mu_j,\mat\Sigma_j)},\qquad N_k = \sum_{n=1}^N r_{nk}}
  \begin{csbul}
    \item \textbf{GMM M-step:} $\vecb\mu_k\gets\frac{1}{N_k}\sum_n r_{nk}\vecb x_n$, $\pi_k\gets N_k/N$, and $\mat\Sigma_k\gets\frac{1}{N_k}\sum_n r_{nk}(\vecb x_n-\vecb\mu_k)(\vecb x_n-\vecb\mu_k)\T + \epsilon\eye_D$
    \item \textbf{$K$-Means limit:} set $\mat\Sigma_k=\epsilon\eye_D$, $\pi_k=1/K$; as $\epsilon\to 0^+$, $r_{nk}\to\mathbb{I}[k=\argmin_j\norm{\vecb x_n-\vecb\mu_j}_2^2]$ (hard Lloyd's algorithm)
  \end{csbul}}

\csnote{8}{A}{TRAPS $-$ before you trust the output}{%
  \begin{csbul}
    \item \textbf{GMM singularity:} $\vecb\mu_k=\vecb x_n, \sigma_k\to 0 \Rightarrow \mathcal L\to+\infty$; add $+\epsilon\eye_D$ jitter
    \item \textbf{Underflow in $r_{nk}$:} always use log-sum-exp in E-step
    \item \textbf{Identifiability:} GMM has $K!$ label switches; PPCA/FA has $\mat W\mat R$ rotation
    \item \textbf{PCA scale trap:} standardize features when units differ; use Gram if $N<D$
  \end{csbul}}

\begin{cscard}{csA}{FORMULAS TO REMEMBER}
  \begin{csbul}
    \item PCA projection \& residual:\csfm{\hat{\vecb x}_n = \bar{\vecb x} + \mat U_M\mat U_M\T(\vecb x_n-\bar{\vecb x}),\quad J_M = \sum_{i=M+1}^D \lambda_i}
    \item PPCA posterior ($\mat M = \mat W\T\mat W + \sigma^2\eye_M$):\csfm{p(\vecb z_n\mid\vecb x_n) = \Normal\!\bigl(\mat M\inv\mat W\T(\vecb x_n-\vecb\mu),\,\sigma^2\mat M\inv\bigr)}
    \item ELBO decomposition:\csfm{\log p(\mat X\mid\vecb\theta) = \E_q[\log p(\mat X,\mat Z\mid\vecb\theta)] + \Entropy(q) + \KL{q}{p(\mat Z\mid\mat X,\vecb\theta)}}
    \item Kernel PCA centering ($\mat H = \eye_N - \frac{1}{N}\ones\ones\T$):\csfm{\tilde{\mat K} = \mat H\mat K\mat H,\quad \tilde{\mat K}\vecb\alpha_m = N\lambda_m\vecb\alpha_m}
  \end{csbul}
\end{cscard}

\begin{cscard}{csB}{LOG-SUM-EXP STABILIZED GMM-EM}
  \begin{enumerate}[leftmargin=1.2em, itemsep=1pt, topsep=1pt, parsep=0pt]
    \item \textbf{Seed:} $\{\vecb\mu_k^{(0)}\}$ via $k$-means++ ($p\propto \min_{j<t}\norm{\vecb x_n-\vecb\mu_j}_2^2$)
    \item \textbf{E-step:} $\beta_{nk}=\log\pi_k+\log\Normal(\vecb x_n\mid\vecb\mu_k,\mat\Sigma_k)$, $m_n=\max_j\beta_{nj}$, $\log r_{nk}=\beta_{nk}-m_n-\log\sum_j e^{\beta_{nj}-m_n}$
    \item \textbf{M-step:} update $N_k$, $\pi_k=N_k/N$, $\vecb\mu_k$, and $\mat\Sigma_k + \epsilon\eye_D$
    \item \textbf{Halt:} stop when $\Delta\sum_n\log Z_n < \tau$
  \end{enumerate}
\end{cscard}

\begin{cscard}{csE}{SCALING \& INVARIANCE CHEAT-LINE}
  \begin{csbul}
    \item \textbf{Woodbury speedup:} $\mat C\inv\in\R^{D\times D}$ in $\mathcal O(M^3)$ via $\mat M\in\R^{M\times M}$
    \item \textbf{Variance retained:} pick $M$ where $\sum_{m=1}^M\lambda_m/\Tr(\mat S)\ge 0.90$
    \item \textbf{Scale invariance:} FA is invariant to feature rescaling; PCA/PPCA are \emph{not}
  \end{csbul}
\end{cscard}

\begin{cscard}{csG}{PITFALL CHECKLIST}
  \begin{csbul}
    \item \textbf{Covariance collapse ($\det\mat\Sigma_k\to 0$):} a single point $\vecb x_n=\vecb\mu_k$ sends likelihood to $\infty$; always add $\epsilon\eye_D$ or Wishart prior
    \item \textbf{Raw exp underflow:} $\Normal(\vecb x_n\mid\vecb\mu_k,\mat\Sigma_k)\approx e^{-D/2}$ underflows in high $D$; compute responsibilities in log-space
    \item \textbf{Unstandardized PCA:} large-unit features dominate $\vecb u_1$; z-score first if units are incommensurate
    \item \textbf{Supervised blindness:} PCA maximizes total variance, not class separation (use Fisher LDA when labels exist)
    \item \textbf{Rotational gauge:} PPCA/FA loadings $\mat W$ are only identifiable up to $\mat W\mat R$ ($\mat R\in\mathrm{O}(M)$)
    \item \textbf{Local optima:} run multiple $k$-means++ restarts for $K$-Means and GMM
  \end{csbul}
\end{cscard}

\begin{cscard}{csD}{10-SECOND VISUAL RECAP}
  \sffamily\footnotesize
  \textbf{PCA} projects orthogonally onto the top-$M$ eigenvectors of $\mat S$ ($\sigma^2\to 0$ limit of \textbf{PPCA}).
  \textbf{GMM} replaces hard \textbf{$K$-Means} Voronoi cells with soft posterior responsibilities $r_{nk}$.
  \textbf{EM} climbs $\log p(\mat X\mid\vecb\theta)$ monotonically without step sizes by alternating exact posterior inference ($q=p(\mat Z\mid\mat X,\vecb\theta)$) and complete-data MLE ($\argmax_{\vecb\theta} Q$).
\end{cscard}
