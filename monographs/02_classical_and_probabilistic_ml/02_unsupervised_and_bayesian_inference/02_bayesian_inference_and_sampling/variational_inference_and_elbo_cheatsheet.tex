\csbranch{1}{A}{1\;\; THE ELBO IDENTITY $-$ three faces}{%
  \begin{minipage}[t]{0.62\linewidth}
    \csfm{\log p_{\vecb\theta}(\vecb x) = \mathrm{ELBO}(q;\vecb\theta) + \KL{q(\vecb z)}{p_{\vecb\theta}(\vecb z\mid\vecb x)}}
    \begin{csbul}
      \item \textbf{Ratio:} $\E_q\!\bigl[\log\frac{p_{\vecb\theta}(\vecb x,\vecb z)}{q(\vecb z)}\bigr]$
      \item \textbf{Free Energy:} $\E_q[\log p_{\vecb\theta}(\vecb x,\vecb z)] + \Entropy(q) = -\mathcal F$
      \item \textbf{Recon $-$ KL:} $\E_q[\log p_{\vecb\theta}(\vecb x\mid\vecb z)] - \KL{q(\vecb z)}{p_{\vecb\theta}(\vecb z)}$
      \item \textbf{IWAE ($S$ draws):} $\mathcal L_1\le\mathcal L_S\uparrow\log p_{\vecb\theta}(\vecb x)$
    \end{csbul}
  \end{minipage}\hfill

\csbranch{2}{B}{2\;\; MEAN-FIELD CAVI \& NATURAL GRADIENTS}{%
  \csfm{q(\vecb z)=\prod_{j=1}^J q_j(\vecb z_j)\;\Longrightarrow\; \log q_j^*(\vecb z_j) = \E_{-j}[\log p_{\vecb\theta}(\vecb x,\vecb z)] + \mathrm{const}}
  \begin{csbul}
    \item \textbf{Markov blanket locality:} $\log q_j^*(\vecb z_j) = \E_{\vecb z_{\mathrm{mb}(j)}}[\log p_{\vecb\theta}(\vecb z_j,\vecb z_{\mathrm{mb}(j)},\vecb x)] + c$
    \item \textbf{Coordinate concavity:} each CAVI step monotonically increases ELBO
    \item \textbf{Natural gradient (SVI):} for conjugate exp-family $\vecb\lambda$, $\tilde{\grad}_{\vecb\lambda}\mathrm{ELBO} = \mat F(\vecb\lambda)\inv\grad_{\vecb\lambda}\mathrm{ELBO} = \E_q[\vecb\eta(\vecb x,\vecb z)] - \vecb\lambda$; update $\vecb\lambda_{t+1}=(1-\rho_t)\vecb\lambda_t+\rho_t\hat{\vecb\eta}_t$
  \end{csbul}}

\csbranch{4}{D}{4\;\; REVERSE vs.\ FORWARD KL $-$ EP \& Laplace}{%
  \begin{csbul}
    \item \textbf{VI ($\min_q\KL{q}{p}$):} \emph{zero-avoiding / mode-seeking} ($q\to 0$ where $p\to 0$); underestimates marginal variance on correlated posteriors: $\sigma_j^2=(\mat\Sigma\inv)_{jj}\inv\le\Sigma_{jj}$
    \item \textbf{EP ($\min_{\tilde q}\KL{q_i^{\mathrm{tilted}}}{\tilde q}$):} \emph{zero-forcing / mode-covering}; matches moments of tilted $q_i^{\mathrm{tilted}}\propto f_i\,q_{\setminus i}$ in cavity $q_{\setminus i}\propto q/\tilde f_i$
    \item \textbf{Laplace:} local 2nd-order Gaussian $\Normal(\hat{\vecb\theta},\mat H\inv)$ at MAP $\hat{\vecb\theta}$, $\mat H=-\hess\log p(\hat{\vecb\theta},\mathcal D)$
  \end{csbul}}

\csbranch{5}{E}{5\;\; AMORTIZED VI (VAE) $-$ speed vs.\ gap}{%
  \begin{minipage}[t]{0.59\linewidth}
    \begin{csbul}
      \item \textbf{Amortization:} shared encoder $q_{\vecb\phi}(\vecb z_n\mid\vecb x_n)$ gives $\mathcal O(1)$ test inference vs.\ $\mathcal O(N)$ local $\vecb\psi_n$
      \item \textbf{Amortization gap:} $\mathrm{ELBO}(\vecb\psi_n^*) - \mathrm{ELBO}(q_{\vecb\phi^*})\ge 0$
      \item \textbf{Closed-form Gaussian KL:} $\frac{1}{2}\sum_{d=1}^D(\sigma_d^2+\mu_d^2-1-\log\sigma_d^2)$
    \end{csbul}
  \end{minipage}\hfill

\csnote{8}{A}{TRAPS $-$ before you trust the output}{%

    \item \textbf{Unbaselined BBVI explodes:} always subtract optimal control variate $c_i^*$ + RB

\begin{cscard}{csA}{FORMULAS TO REMEMBER}
  \begin{csbul}
    \item Three ELBO forms ($\tilde{\mathcal L}=\log p_{\vecb\theta}(\vecb x,\vecb z)-\log q(\vecb z)$):\csfm{\mathrm{ELBO} = \E_q[\tilde{\mathcal L}] = \E_q[\log p_{\vecb\theta}(\vecb x,\vecb z)]+\Entropy(q) = \E_q[\log p_{\vecb\theta}(\vecb x\mid\vecb z)]-\KL{q}{p}}
    \item Optimal BBVI control variate ($g_i=\partial_{\phi_i}\log q_{\vecb\phi}(\vecb z)$):\csfm{c_i^* = \frac{\Cov(g_i\tilde{\mathcal L},\,g_i)}{\Var[g_i]}}
    \item Change-of-variables density for $\vecb z = g_{\vecb\phi}(\vecb\epsilon,\vecb x)$:\csfm{\log q_{\vecb\phi}(\vecb z) = \log p(\vecb\epsilon) - \log\abs*{\det\pd{\vecb z}{\vecb\epsilon}}}
    \item EP Cavity--Tilted--Project update:\csfm{q_{\setminus i}\propto \frac{q}{\tilde f_i},\quad q_i^{\mathrm{tilted}}=\frac{f_i q_{\setminus i}}{Z_i},\quad \tilde f_i^{\mathrm{new}}\propto\Bigl(\frac{\mathrm{proj}_{\mathcal Q}(q_i^{\mathrm{tilted}})}{q_{\setminus i}}\Bigr)^\delta\tilde f_i^{1-\delta}}
  \end{csbul}
\end{cscard}

\begin{cscard}{csB}{ALGORITHMIC RECIPES (CAVI vs.\ RVI)}
  \textbf{Mean-Field CAVI:}
  \begin{enumerate}[leftmargin=1.2em, itemsep=0.5pt, topsep=0.5pt, parsep=0pt]
    \item Initialize factors $q_1(\vecb z_1),\dots,q_J(\vecb z_J)$
    \item For $j=1,\dots,J$: $q_j(\vecb z_j)\propto\exp\!\bigl(\E_{\vecb z_{\mathrm{mb}(j)}}[\log p_{\vecb\theta}(\vecb z_j,\vecb z_{\mathrm{mb}(j)},\vecb x)]\bigr)$
    \item Repeat until $\Delta\mathrm{ELBO}<\varepsilon$
  \end{enumerate}
  \textbf{Reparameterized SVI (VAE):}
  \begin{enumerate}[leftmargin=1.2em, itemsep=0.5pt, topsep=0.5pt, parsep=0pt]
    \item Draw minibatch $\vecb x_n$ and $\vecb\epsilon_n\sim\Normal(\zeros,\eye)$; encode $(\vecb\mu_n,\vecb\sigma_n)=\mathrm{enc}_{\vecb\phi}(\vecb x_n)$
    \item Set $\vecb z_n=\vecb\mu_n+\vecb\sigma_n\odot\vecb\epsilon_n$ and ascend $\grad_{(\vecb\theta,\vecb\phi)}\hat{\mathcal L}$
  \end{enumerate}
\end{cscard}

\begin{cscard}{csE}{VARIANCE \& BOUNDS CHEAT-LINE}
  \begin{csbul}
    \item \textbf{Tightness hierarchy:} $\mathrm{ELBO}=\mathcal L_1\le\mathcal L_S\le\log p_{\vecb\theta}(\vecb x)$
    \item \textbf{Mean-field precision:} $\Var_{q_j}(z_j)=(\mat\Sigma\inv)_{jj}\inv \le \Sigma_{jj}$
    \item \textbf{IWAE SNR trap:} encoder SNR scales as $\mathcal O(S^{-1/2})$ without DReG
  \end{csbul}
\end{cscard}

\begin{cscard}{csG}{PITFALL CHECKLIST}
  \begin{csbul}
    \item \textbf{Passing $\grad_{\vecb\phi}$ inside $\E_{q_{\vecb\phi}}$ directly:} invalid! Must reparameterize $\vecb z=g_{\vecb\phi}(\vecb\epsilon)$ or use score identity $\grad\log q_{\vecb\phi}$
    \item \textbf{Ignoring $\log q_{\vecb\phi}$ pathwise term:} if stopping gradient on $\vecb\phi$ in $\log q_{\vecb\phi}(\vecb z)$, ensure $\Entropy(q)$ or $\KL{q}{p}$ is differentiated properly
    \item \textbf{Mean-field uncertainty underestimation:} factorized $q$ locks onto conditional precision $(\mat\Sigma\inv)_{jj}$; never trust raw MF-VI error bars on correlated parameters
    \item \textbf{Posterior collapse in VAEs:} autoregressive decoder ignores $\vecb z$; use KL warmup ($\beta$: $0\to 1$) or free bits
    \item \textbf{Undamped EP divergence:} EP has no global Lyapunov guarantee; damp site updates and check positive-definiteness
  \end{csbul}
\end{cscard}

\begin{cscard}{csD}{10-SECOND VISUAL RECAP}
  \sffamily\footnotesize
  Because $\log p(\vecb x) = \mathrm{ELBO}(q) + \KL{q}{p}$, maximizing the \textbf{ELBO} minimizes reverse KL without ever computing the evidence $p(\vecb x)$.
  Use \textbf{CAVI/SVI} when conditionally conjugate, \textbf{Pathwise Reparameterization} for continuous deep models, \textbf{BBVI+baselines} for discrete latents, and \textbf{EP} when mode-covering moments matter.
\end{cscard}
