\csbranch{1}{A}{1\;\; WHAT $-$ minimax game \& JSD equilibrium}{%
  \csfm{\min_\theta\max_\phi V(\phi,\theta)=\E_{p_{\mathrm{data}}}[\log D_\phi(\vecb x)]+\E_{p_Z}[\log(1-D_\phi(G_\theta(\vecb z)))]}
  \begin{csbul}
    \item \textbf{Optimal discriminator} (fixed $q_\theta=(G_\theta)_\# p_Z$):
      \csfm{D^*_\theta(\vecb x)=\frac{p_{\mathrm{data}}(\vecb x)}{p_{\mathrm{data}}(\vecb x)+q_\theta(\vecb x)},\quad \frac{D^*_\theta(\vecb x)}{1-D^*_\theta(\vecb x)}=\frac{p_{\mathrm{data}}(\vecb x)}{q_\theta(\vecb x)}}
    \item \textbf{Plug-in value:} $V(G_\theta,D^*_\theta)=2\,\JS{p_{\mathrm{data}}}{q_\theta}-\log 4$ (min $-\log 4$ at $q_\theta=p_{\mathrm{data}}$, $D^*\equiv\frac{1}{2}$)
    \item \textbf{Non-saturating loss:} $L_G^{\mathrm{NS}}=-\E_{p_Z}[\log D_\phi(G_\theta(\vecb z))]$ fixes $\nabla_\theta\to\zeros$ when $D_\phi\approx 0$
  \end{csbul}}

\csbranch{2}{B}{2\;\; $f$-GANs $-$ Fenchel conjugate duality}{%
  \csfm{D_f(p_{\mathrm{data}}\|q_\theta)\ge\sup_{T_\phi}\Bigl\{\E_{\vecb x\sim p_{\mathrm{data}}}[T_\phi(\vecb x)]-\E_{\tilde{\vecb x}\sim q_\theta}[f^\dagger(T_\phi(\tilde{\vecb x}))]\Bigr\}}
  \renewcommand{\arraystretch}{1.12}\scriptsize
  \begin{tabular}{@{}>{\raggedright\arraybackslash}p{1.6cm}>{\raggedright\arraybackslash}p{2.0cm}>{\raggedright\arraybackslash}p{1.65cm}>{\raggedright\arraybackslash}p{1.55cm}@{}}
    \textbf{\textcolor{csB}{Divergence}} & \textbf{\textcolor{csB}{$f(u)$}} & \textbf{\textcolor{csB}{$f^\dagger(t)$}} & \textbf{\textcolor{csB}{$T^*(r)$}}\\ \hline
    Forward KL & $u\log u$ & $\exp(t-1)$ & $1+\log r$\\
    Reverse KL & $-\log u$ & $-1-\log(-t)$ & $-1/r$\\
    JSD & $u\log u-(u{+}1)\log\frac{u+1}{2}$ & $-\log(2-e^t)$ & $\log\frac{2r}{r+1}$\\
    Pearson $\chi^2$ & $(u-1)^2$ & $\frac{1}{4}t^2+t$ & $2(r-1)$\\
  \end{tabular}}

\csbranch{3}{C}{3\;\; KEY THEOREM $-$ disjoint supports \& $W_1$}{%
  \begin{minipage}[t]{0.60\linewidth}
    \csfm{\Wass[1]{p_{\mathrm{data}}}{q_\theta}=\sup_{\norm{f}_{\mathrm{Lip}}\le 1}\bigl\{\E_{p_{\mathrm{data}}}[f(\vecb x)]-\E_{q_\theta}[f(\tilde{\vecb x})]\bigr\}}
    \begin{csbul}
      \item \textbf{Parallel lines} $P=(0,U)$, $Q_\theta=(\theta,U)$: $\mathrm{KL}=\infty$, $\mathrm{JSD}=\log 2\cdot\mathbf 1_{\theta\ne 0}$ ($\nabla_\theta=0$!), but $\Wass[1]{P}{Q_\theta}=\abs{\theta}$ ($\nabla_\theta=\pm 1$)
      \item \textbf{WGAN-GP:} $\eta\E_{\hat{\vecb x}}[(\norm{\nabla_{\hat{\vecb x}}f_\phi(\hat{\vecb x})}_2-1)^2]$
      \item \textbf{SN-GAN:} $\mat W_l^{\mathrm{SN}}=\mat W_l/\sigma_{\max}(\mat W_l)$
    \end{csbul}
  \end{minipage}\hfill

\csbranch{4}{D}{4\;\; GAME DYNAMICS $-$ DiracGAN \& collapse}{%
  \begin{minipage}[t]{0.60\linewidth}
    \csfm{\mat J(0,0)=\begin{bmatrix}0&\ell'(0)\\-\ell'(0)&0\end{bmatrix},\quad \lambda_\pm=\pm i\ell'(0)}
    \begin{csbul}
      \item \textbf{DiracGAN} ($D_\phi(x)=\phi x$, $G_\theta=\theta$): $\dd{}{t}(\theta^2+\phi^2)=0\implies$ undamped circular orbits in continuous time; discrete Euler spirals outward!

      \item \textbf{Fix:} Adam $\beta_1\in[0,0.5]$, $R_1$ grad penalty, AdaIN (StyleGAN)
    \end{csbul}

\csbranch{6}{F}{6\;\; CONNECTIONS $-$ ratios, IPMs \& StyleGAN}{%
  \begin{csbul}
    \item \textbf{Density-ratio vs.\ IPM:} $f$-divergences test ratios $p_{\mathrm{data}}/q_\theta$ (fail on thin manifolds); Integral Probability Metrics ($\sup_{f\in\mathcal F}\abs{\E_P f-\E_Q f}$) test smooth differences
    \item \textbf{Minibatch Partial OT ($\mathrm{m\text{-}POT}_s$):} transports fraction $s\in(0,1]$ in $O(k(m+1)^2)$ to drop cross-mode outliers
    \item \textbf{Convolution Sliced-OT:} replaces flat $\R^d$ random projections with strided/dilated conv slicers
    \item \textbf{Truncation trick} ($w'=\bar w+\tau(w-\bar w)$): trades diversity for fidelity
  \end{csbul}}

\csnote{7}{G}{HOOK $-$ say it in one breath}{%
  \centering\normalsize\sffamily\bfseries\textcolor{csG}{Density Ratios Saturate on Disjoint Manifolds $\cdot$ $1$-Lipschitz Critics Give Smooth Gradients}\\[1pt]
  \footnotesize\mdseries \textbf{When $\mathrm{supp}(p_{\mathrm{data}})\cap\mathrm{supp}(q_\theta)=\emptyset$, $\mathrm{JSD}=\log 2$ has zero slope, while $W_1=\abs{\theta}$ pulls $q_\theta\to p_{\mathrm{data}}$ at unit speed.}}

\csnote{8}{A}{TRAPS $-$ before you trust the output}{%
  \begin{csbul}
    \item \textbf{No BatchNorm in WGAN-GP:} corrupts $\norm{\nabla_{\hat{\vecb x}}f_\phi}_2\to$ LayerNorm / SN
    \item \textbf{Weight clipping $[-c,c]$:} forces weights to $\pm c$, collapsing capacity
    \item \textbf{Cycles \& mOT bias:} use Adam $\beta_1\le 0.5$ and $\mathrm{m\text{-}POT}_s$
  \end{csbul}}

\begin{cscard}{csA}{FORMULAS TO REMEMBER}
  \begin{csbul}
    \item Optimal Discriminator \& JSD Minimax Equilibrium:
      \csfm{D^*_\theta(\vecb x)=\frac{p_{\mathrm{data}}(\vecb x)}{p_{\mathrm{data}}(\vecb x)+q_\theta(\vecb x)},\quad V(G_\theta,D^*_\theta)=2\,\JS{p_{\mathrm{data}}}{q_\theta}-\log 4}
    \item $f$-GAN Variational Lower Bound ($f^\dagger(t)=\sup_u\{ut-f(u)\}$):
      \csfm{D_f(p_{\mathrm{data}}\|q_\theta)\ge\sup_{T_\phi}\Bigl\{\E_{p_{\mathrm{data}}}[T_\phi(\vecb x)]-\E_{q_\theta}[f^\dagger(T_\phi(\tilde{\vecb x}))]\Bigr\}}
    \item Kantorovich--Rubinstein $W_1$ Dual:
      \csfm{\Wass[1]{p_{\mathrm{data}}}{q_\theta}=\inf_{\pi\in\Pi}\E_\pi[\norm{\vecb x-\vecb y}_2]=\sup_{\norm{f}_{\mathrm{Lip}}\le 1}\bigl\{\E_{p_{\mathrm{data}}}[f(\vecb x)]-\E_{q_\theta}[f(\tilde{\vecb x})]\bigr\}}
    \item WGAN-GP Critic Objective ($\hat{\vecb x}=\epsilon\vecb x+(1-\epsilon)\tilde{\vecb x}$, $\eta=10$):
      \csfm{L_D^{\mathrm{GP}}=\E_{q_\theta}[f_\phi(\tilde{\vecb x})]-\E_{p_{\mathrm{data}}}[f_\phi(\vecb x)]+\eta\,\E_{\hat{\vecb x}}\!\bigl[(\norm{\nabla_{\hat{\vecb x}}f_\phi(\hat{\vecb x})}_2-1)^2\bigr]}
    \item Spectral Normalization \& Minibatch OT:
      \csfm{\mat W_l^{\mathrm{SN}}=\frac{\mat W_l}{\sigma_{\max}(\mat W_l)},\qquad L_{\mathrm{mOT}}(\theta)=\frac{1}{k}\sum_{i=1}^k\mathrm{OT}_c\bigl((G_\theta)_\# P_{z_i^m},P_{X_i^m}\bigr)}
  \end{csbul}
\end{cscard}

\begin{cscard}{csB}{RECIPE $-$ WGAN-GP / SN-GAN TRAINING}
  \begin{enumerate}[leftmargin=1.2em, itemsep=1pt, topsep=1pt, parsep=0pt]
    \item \textbf{Critic loop ($n_{\mathrm{critic}}=5$ steps):} sample $\vecb x\sim p_{\mathrm{data}}$, $\tilde{\vecb x}=G_\theta(\vecb z)$, $\epsilon\sim\Uniform(0,1)$, form $\hat{\vecb x}=\epsilon\vecb x+(1-\epsilon)\tilde{\vecb x}$
    \item \textbf{Enforce $1$-Lip:} add GP $\eta(\norm{\nabla_{\hat{\vecb x}}f_\phi}_2-1)^2$ (\emph{no BatchNorm!}) or apply 1-step power iteration $\mat W_l\leftarrow\mat W_l/\sigma_{\max}(\mat W_l)$ (SN-GAN)
    \item \textbf{Generator step:} minimize $L_G=-\E_{p_Z}[f_\phi(G_\theta(\vecb z))]$ using Adam with low $\beta_1\in[0,0.5]$
  \end{enumerate}
\end{cscard}

\begin{cscard}{csC}{DECISION GUIDE}
  \renewcommand{\arraystretch}{1.22}\small
  \begin{tabular}{@{}>{\raggedright\arraybackslash}p{3.1cm}>{\raggedright\arraybackslash}p{4.6cm}@{}}
    \textbf{If you have\ldots} & \textbf{use\ldots}\\ \hline
    vanishing grad on thin manifolds & WGAN-GP or SN-GAN ($W_1$ / Hinge)\\
    2nd-order GP is too slow & Spectral Norm (SN-GAN) + $R_1$ reg.\\
    custom ground cost $c(\vecb x,\vecb y)$ & Primal mOT-GAN with Sinkhorn\\

  \end{tabular}
\end{cscard}

\begin{cscard}{csE}{CONTINUITY \& SCALING CHEAT-LINE}
  \begin{csbul}
    \item \textbf{Arjovsky Theorem:} if $G_\theta$ is continuous in $\theta$, $\Wass[1]{p_{\mathrm{data}}}{q_\theta}$ is continuous everywhere (unlike $\mathrm{KL},\mathrm{JSD}$)
    \item \textbf{OT scaling:} exact LP $O(m^3)\to$ Sinkhorn $O(km^2)\to$ 1D Sliced OT $O(Lm\log m)$
  \end{csbul}
\end{cscard}

\begin{cscard}{csG}{PITFALL CHECKLIST}
  \begin{csbul}
    \item \textbf{Saturating generator loss:} minimizing $\E[\log(1-D_\phi(G_\theta(\vecb z)))]$ gives $\nabla_\theta\approx\zeros$ when $D_\phi\to 0$; use $-\E[\log D_\phi(G_\theta(\vecb z))]$ or $W_1$
    \item \textbf{Symplectic game rotation:}skew-symmetric off-diagonal blocks in $\mat J(\phi,\theta)$ yield $\lambda_\pm=\pm i\ell'(0)$, causing Euler steps to spiral outward
    \item \textbf{Mode dropping vs.\ quality:} reverse-KL-like dynamics prefer high-precision samples on a few modes over full support coverage
    \item \textbf{BatchNorm inside WGAN-GP critic:} turns a pointwise gradient norm constraint into a batch-coupled penalty; use LayerNorm/Spectral Norm
    \item \textbf{Weight clipping pathology:} $\phi\in[-c,c]$ forces weights to binary extremes $\pm c$ and ruins high-order moments
    \item \textbf{False minibatch OT pairings:} small batch $m$ pairs unrelated modes; increase $m,k$ or use partial transport $\mathrm{m\text{-}POT}_s$
  \end{csbul}
\end{cscard}

\begin{cscard}{csD}{10-SECOND VISUAL RECAP}
  \sffamily\footnotesize
  \textbf{GANs} pit a generator $G_\theta(\vecb z)$ against a critic without ever evaluating $q_\theta(\vecb x)$.
  \textbf{JSD / $f$-GANs} estimate density ratios $p_{\mathrm{data}}/q_\theta$, which saturate ($\nabla_\theta=0$) when low-dimensional manifolds do not overlap.
  \textbf{WGANs \& OT-GANs} enforce $1$-Lipschitz critics (GP / Spectral Norm) or solve primal transport to provide smooth geometry-aware gradients everywhere.
\end{cscard}
