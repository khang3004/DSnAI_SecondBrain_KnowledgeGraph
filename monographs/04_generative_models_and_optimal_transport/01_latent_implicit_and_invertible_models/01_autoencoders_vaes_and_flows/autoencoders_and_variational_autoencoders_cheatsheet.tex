\csbranch{1}{A}{1\;\; WHAT $-$ deterministic vs.\ variational}{%
  \begin{minipage}[t]{0.59\linewidth}
    \csfm{p_\theta(\vecb x,\vecb z)=p(\vecb z)\,p_\theta(\vecb x\mid\vecb z)}
    \begin{csbul}
      \item \textbf{Linear AE} ($\mat W_2\mat W_1\vecb x$): spans exact $M$-dim \textbf{PCA} subspace
      \item \textbf{DAE} ($\tilde{\vecb x}=\vecb x+\sigma\vecb\epsilon$): Tweedie score $r^*(\tilde{\vecb x})-\tilde{\vecb x}=\sigma^2\nabla\log p_\sigma(\tilde{\vecb x})$
      \item \textbf{CAE:} penalizes $\lambda\norm{\mat J_e(\vecb x)}_F^2$
      \item \textbf{VAE:} amortized $q_\phi(\vecb z\mid\vecb x)$ + prior $p(\vecb z)=\Normal(\zeros,\eye_M)$
    \end{csbul}
  \end{minipage}\hfill
  \begin{minipage}[t]{0.38\linewidth}\centering\vspace{4pt}

\csbranch{2}{B}{2\;\; ELBO $-$ likelihood lower bound \& gap}{%
  \csfm{\mathcal L_{\mathrm{ELBO}}(\theta,\phi;\vecb x)=\underbrace{\E_{q_\phi(\vecb z\mid\vecb x)}[\log p_\theta(\vecb x\mid\vecb z)]}_{\text{Reconstruction }(-D)}-\underbrace{\KL{q_\phi(\vecb z\mid\vecb x)}{p(\vecb z)}}_{\text{Prior Matching / Rate }R}}
  \begin{csbul}
    \item \textbf{Exact gap:} $\log p_\theta(\vecb x)=\mathcal L_{\mathrm{ELBO}}+\KL{q_\phi(\vecb z\mid\vecb x)}{p_\theta(\vecb z\mid\vecb x)}$
    \item \textbf{Joint KL:} $\max_{\theta,\phi}\E_{q_{\mathcal D}}[\mathcal L_{\mathrm{ELBO}}]\iff\min_{\theta,\phi}\KL{q_{\mathcal D,\phi}(\vecb x,\vecb z)}{p_\theta(\vecb x,\vecb z)}$
    \item \textbf{Diagonal Gaussian KL} ($p(\vecb z)=\Normal(\zeros,\eye_M)$):
      $\frac{1}{2}\sum_{j=1}^M\bigl(\mu_{\phi,j}^2+\sigma_{\phi,j}^2-1-\log\sigma_{\phi,j}^2\bigr)$
  \end{csbul}}

\csbranch{3}{C}{3\;\; HOW $-$ pathwise reparameterization trick}{%
  \begin{minipage}[t]{0.58\linewidth}
    \csfm{\vecb z=g_\phi(\vecb\epsilon,\vecb x)=\vecb\mu_\phi(\vecb x)+\vecb\sigma_\phi(\vecb x)\odot\vecb\epsilon}
    \begin{csbul}
      \item $\vecb\epsilon\sim\Normal(\zeros,\eye_M)$ moves randomness \emph{outside} the backprop path
      \item \textbf{Pathwise grad:} low variance ($L=1$ sample/item suffices)
      \item \textbf{REINFORCE} $f(\vecb z)\nabla_\phi\log q_\phi$: unbiased, huge variance
    \end{csbul}
  \end{minipage}\hfill
  \begin{minipage}[t]{0.39\linewidth}\centering\vspace{2pt}

\csbranch{4}{D}{4\;\; $\beta$-VAE \& VQ-VAE $-$ rate vs.\ discrete codes}{%
  \begin{csbul}
    \item \textbf{$\beta$-VAE:} $\mathcal L_\beta=D+\beta R$ ($\beta>1$ suppresses $\mathrm{TC}(\vecb z)$, disentangles at cost of blur; $\beta\to 0$ gives noisy AE)
    \item \textbf{VQ-VAE lookup:} $k^*=\argmin_k\norm{\vecb z_e(\vecb x)-\vecb e_k}_2$, $\;\vecb z_q(\vecb x)=\vecb e_{k^*}$
    \item \textbf{Straight-Through (STE):} $\vecb z_q\leftarrow\vecb z_e+\operatorname{sg}[\vecb e_{k^*}-\vecb z_e]$ copies $\nabla_{\vecb z_q}\mathcal L\to\nabla_{\vecb z_e}\mathcal L$
    \item \textbf{VQ loss:} $-\log p_\theta(\vecb x\mid\vecb z_q)+\norm{\operatorname{sg}[\vecb z_e]-\vecb e_{k^*}}_2^2+\beta_{\mathrm{com}}\norm{\vecb z_e-\operatorname{sg}[\vecb e_{k^*}]}_2^2$
    \item \textbf{EMA codebook:} $N_k\leftarrow\gamma N_k+(1-\gamma)n_k$, $\;\vecb e_k\leftarrow\vecb m_k/N_k$ ($\gamma\approx 0.99$)
  \end{csbul}}

\begin{cscard}{csA}{FORMULAS TO REMEMBER}
  \begin{csbul}
    \item ELBO \& Marginal Identity:
      \csfm{\log p_\theta(\vecb x)=\mathcal L_{\mathrm{ELBO}}(\theta,\phi;\vecb x)+\KL{q_\phi(\vecb z\mid\vecb x)}{p_\theta(\vecb z\mid\vecb x)}}
    \item Diagonal Gaussian KL ($q_\phi=\Normal(\vecb\mu_\phi,\diag(\vecb\sigma_\phi^2))$, $p=\Normal(\zeros,\eye_M)$):
      \csfm{\KL{q_\phi}{p}=\frac{1}{2}\sum_{j=1}^M\bigl(\mu_{\phi,j}^2+\sigma_{\phi,j}^2-1-\log\sigma_{\phi,j}^2\bigr)}
    \item Reparameterized Gradient ($g_\phi(\vecb\epsilon,\vecb x)=\vecb\mu_\phi(\vecb x)+\vecb\sigma_\phi(\vecb x)\odot\vecb\epsilon$):
      \csfm{\nabla_\phi\E_{q_\phi}[f(\vecb z)]=\E_{\vecb\epsilon\sim\Normal(\zeros,\eye)}\!\bigl[\nabla_\phi f(g_\phi(\vecb\epsilon,\vecb x))\bigr]}
    \item DAE Tweedie Score Identity ($\tilde{\vecb x}\sim\Normal(\vecb x,\sigma^2\eye)$):
      \csfm{r^*(\tilde{\vecb x})-\tilde{\vecb x}=\sigma^2\nabla_{\tilde{\vecb x}}\log p_\sigma(\tilde{\vecb x})+o(\sigma^2)}
    \item VQ-VAE Objective ($\beta_{\mathrm{com}}\approx 0.25$):
      \csfm{\mathcal L_{\mathrm{VQ}}=-\log p_\theta(\vecb x\mid\vecb z_q)+\norm{\operatorname{sg}[\vecb z_e]-\vecb e_{k^*}}_2^2+\beta_{\mathrm{com}}\norm{\vecb z_e-\operatorname{sg}[\vecb e_{k^*}]}_2^2}
  \end{csbul}
\end{cscard}

\begin{cscard}{csB}{RECIPE $-$ SGVB ($\beta$-VAE) \& VQ-VAE STEP}
  \begin{enumerate}[leftmargin=1.2em, itemsep=1pt, topsep=1pt, parsep=0pt]
    \item \textbf{VAE forward:} $(\vecb\mu,\log\vecb\sigma^2)\leftarrow e_\phi(\vecb x)$; clamp $\log\vecb\sigma^2\in[-20,10]$
    \item \textbf{Reparameterize:} draw $\vecb\epsilon\sim\Normal(\zeros,\eye_M)$, set $\vecb z=\vecb\mu+\exp(\tfrac{1}{2}\log\vecb\sigma^2)\odot\vecb\epsilon$
    \item \textbf{VAE loss:} $\mathcal L_\beta=-\log p_\theta(\vecb x\mid d_\theta(\vecb z))+\beta\,\KL{q_\phi}{p}$
    \item \textbf{VQ-VAE alternative:} $k^*=\argmin_k\norm{\vecb z_e-\vecb e_k}_2$; set $\vecb z_q=\vecb z_e+\operatorname{sg}[\vecb e_{k^*}-\vecb z_e]$ and update $\vecb e_k$ via EMA ($\gamma=0.99$)
  \end{enumerate}
\end{cscard}

\begin{cscard}{csE}{RATE--DISTORTION SCALING CHEAT-LINE}

    \item \textbf{Prior mismatch:} sampling $\vecb z\sim p(\vecb z)$ vs.\ aggregated $q_\phi(\vecb z)=\frac{1}{N}\sum_n q_\phi(\vecb z\mid\vecb x_n)$ leaves holes of size $\KL{q_\phi(\vecb z)}{p(\vecb z)}$
  \end{csbul}
\end{cscard}

\begin{cscard}{csG}{PITFALL CHECKLIST}
  \begin{csbul}
    \item \textbf{Posterior collapse ($\KL{q_\phi}{p}\to 0$):} powerful decoders ignore $\vecb z$; apply cyclical $\beta$-annealing ($0\to 1$), free bits $\max(\lambda,\mathrm{KL}_j)$, or skip connections
    \item \textbf{Blurry reconstructions:} pixel-wise Gaussian MSE averages over posterior uncertainty; use perceptual/adversarial losses or latent diffusion
    \item \textbf{Prior hole problem:} $p(\vecb z)=\Normal(\zeros,\eye)$ fails to match $q_\phi(\vecb z)$; fit a post-hoc prior (VampPrior, PixelCNN, Transformer)
    \item \textbf{VQ codebook collapse:} dead embeddings receive zero STE gradients; use EMA cluster updates and re-seed unused codes from batch $\vecb z_e$
    \item \textbf{Unidentifiability:} nonlinear $\beta$-VAE cannot provably disentangle without structural inductive biases
    \item \textbf{Float16 overflow:} always clamp encoder $\log\sigma^2\in[-20,10]$ and compute KL in float32
  \end{csbul}
\end{cscard}

\begin{cscard}{csD}{10-SECOND VISUAL RECAP}
  \sffamily\footnotesize
  \textbf{Deterministic AE} maps $\vecb x\to\vecb z\to\hat{\vecb x}$ (spans PCA if linear, estimates $\nabla\log p_\sigma$ if denoising) but leaves holes in $\mathcal Z$.
  \textbf{VAE} maps $\vecb x$ to a cloud $\Normal(\vecb\mu_\phi,\vecb\sigma_\phi^2)$, pulls it toward $\Normal(\zeros,\eye)$ via KL rate $R$, and backpropagates through $\vecb z=\vecb\mu_\phi+\vecb\sigma_\phi\odot\vecb\epsilon$.
  \textbf{VQ-VAE} snaps $\vecb z_e$ to discrete codebook vectors $\vecb e_{k^*}$ with straight-through gradients.
\end{cscard}
