\csbranch{1}{A}{1\;\; WHAT $-$ exact change of variables}{%
  \csfm{\log p_X(\vecb x)=\log p_Z\!\bigl(f_\theta\inv(\vecb x)\bigr)+\log\abs{\det\jac_{f_\theta\inv}(\vecb x)}}
  \begin{csbul}
    \item \textbf{Inverse Jacobian identity:} $\log\abs{\det\jac_{f_\theta\inv}(\vecb x)}=-\log\abs{\det\jac_{f_\theta}(\vecb z)}$
    \item \textbf{$L$-layer composition} ($f=f_L\circ\dots\circ f_1$, $\vecb h_0=\vecb z$, $\vecb h_L=\vecb x$):
      \csfm{\log p_X(\vecb x)=\log p_Z(\vecb z)-\sum_{k=1}^L\log\abs{\det\jac_{f_k}(\vecb h_{k-1})}}
    \item \textbf{Volume conservation:} expansion ($\abs{\det\jac_f}>1$) dilutes $p_X$; compression ($\abs{\det\jac_f}<1$) concentrates $p_X$
  \end{csbul}}

\csbranch{2}{B}{2\;\; COUPLING \& GLOW $-$ $O(1)$ both ways}{%
  \begin{minipage}[t]{0.60\linewidth}
    \csfm{\vecb x_A=\vecb z_A,\quad \vecb x_B=e^{\vecb s(\vecb z_A)}\odot\vecb z_B+\vecb t(\vecb z_A)}
    \begin{csbul}
      \item \textbf{Inverse:} $\vecb z_B=e^{-\vecb s(\vecb x_A)}\odot(\vecb x_B-\vecb t(\vecb x_A))$
      \item $\vecb s,\vecb t$ can be \textbf{any} neural net (never inverted!)
      \item \textbf{Log-det:} $\sum_{j=1}^{D-d}s_j(\vecb z_A)$ in $O(D)$
      \item \textbf{Glow $1{\times}1$ conv:} $\mat W=\mat P\mat L(\mat U+\diag(\vecb s))$, $\log\abs{\det}=HW\sum_c\log\abs{s_c}$ in $O(C)$
    \end{csbul}
  \end{minipage}\hfill

\csbranch{3}{C}{3\;\; AUTOREGRESSIVE $-$ MAF vs.\ IAF duality}{%
  \csfm{\text{\textbf{MAF}: }x_i=z_i e^{\alpha_i(\vecb x_{1:i-1})}+\mu_i(\vecb x_{1:i-1})\quad\text{\textbf{IAF}: }x_i=z_i e^{\alpha_i(\vecb z_{1:i-1})}+\mu_i(\vecb z_{1:i-1})}
  \begin{csbul}
    \item Lower-triangular $\jac_f\implies\log\abs{\det\jac_f}=\sum_{i=1}^D\alpha_i$ in $O(D)$
    \item \textbf{MAF (MADE):} $O(1)$ parallel density $\vecb x\to\vecb z$ (\textbf{fast MLE}); $O(D)$ sequential sampling $\vecb z\to\vecb x$
    \item \textbf{IAF:} $O(1)$ parallel sampling $\vecb z\to\vecb x$ (\textbf{fast VI / distillation}); $O(D)$ sequential density $\vecb x\to\vecb z$
  \end{csbul}}

\csbranch{4}{D}{4\;\; NEURAL ODEs \& CNFs $-$ continuous limit}{%
  \begin{minipage}[t]{0.60\linewidth}
    \csfm{\dd{\vecb z(t)}{t}=\vecb v_\theta(\vecb z(t),t),\quad \dd{}{t}\log p_t(\vecb z_t)=-\grad_{\vecb z}\cdot\vecb v_\theta}
    \begin{csbul}
      \item \textbf{Exact log-density:} $\log p_X(\vecb x)=\log p_Z(\vecb z_0)-\int_0^T\Tr(\jac_{\vecb v_\theta})\,\mathrm{d}t$
      \item \textbf{FFJORD Hutchinson trace:} $\Tr(\jac_{\vecb v_\theta})=\E_{\vecb\epsilon}[\vecb\epsilon\T\jac_{\vecb v_\theta}\vecb\epsilon]$ in $O(D)$ via 1 VJP
      \item \textbf{Adjoint ODE:} $\dd{\vecb a}{t}=-\vecb a\T\jac_{\vecb v_\theta}$, $O(1)$ memory
    \end{csbul}
  \end{minipage}\hfill

\csbranch{6}{F}{6\;\; CONNECTIONS $-$ physics, KL \& flow matching}{%
  \begin{csbul}
    \item \textbf{Continuity Equation:} $\partial_t p_t+\grad\cdot(p_t\vecb v_\theta)=0$ (zero-diffusion $g(t)=0$ limit of Fokker--Planck PDE $\implies$ Diffusion PF-ODE)
    \item \textbf{Forward KL (MLE):} $\min\KL{p_{\mathrm{data}}}{p_\theta}$ needs fast $\vecb x\to\vecb z$ (RealNVP/MAF)
    \item \textbf{Reverse KL (VI/Distill):} $\min\KL{p_\theta}{q}$ needs fast $\vecb z\to\vecb x$ (IAF)
    \item \textbf{Flow Matching:} regresses $\vecb v_\theta(\vecb z_t,t)$ directly without training ODE solves!
  \end{csbul}}

\csnote{8}{A}{TRAPS $-$ before you trust the output}{%

\begin{cscard}{csA}{FORMULAS TO REMEMBER}
  \begin{csbul}
    \item Discrete Change-of-Variables ($g_\theta=f_\theta\inv$):
      \csfm{\log p_X(\vecb x)=\log p_Z(g_\theta(\vecb x))+\log\abs{\det\jac_{g_\theta}(\vecb x)}}
    \item RealNVP Affine Coupling ($\vecb z=(\vecb z_A,\vecb z_B)$):
      \csfm{\vecb x_A=\vecb z_A,\;\vecb x_B=e^{\vecb s(\vecb z_A)}\odot\vecb z_B+\vecb t(\vecb z_A),\;\log\abs{\det\jac_f}=\sum_{j=1}^{D-d}s_j(\vecb z_A)}
    \item Glow $1\times 1$ LU Invertible Convolution ($\mat W\in\R^{C\times C}$):
      \csfm{\mat W=\mat P\mat L(\mat U+\diag(\vecb s)),\quad \log\abs{\det\jac}=HW\sum_{c=1}^C\log\abs{s_c}}
    \item Instantaneous Change of Variables \& Continuity PDE:
      \csfm{\dd{}{t}\log p_t(\vecb z_t)=-\grad_{\vecb z}\cdot\vecb v_\theta(\vecb z_t,t),\quad \partial_t p_t+\grad\cdot(p_t\vecb v_\theta)=0}
    \item Hutchinson Unbiased Trace Estimator ($\E[\vecb\epsilon]=\zeros,\Cov(\vecb\epsilon)=\eye$):
      \csfm{\Tr(\jac_{\vecb v_\theta})=\E_{\vecb\epsilon}\!\bigl[\vecb\epsilon\T\jac_{\vecb v_\theta}(\vecb z,t)\vecb\epsilon\bigr]}
  \end{csbul}
\end{cscard}

\begin{cscard}{csB}{RECIPE $-$ FFJORD CNF TRAINING STEP}
  \begin{enumerate}[leftmargin=1.2em, itemsep=1pt, topsep=1pt, parsep=0pt]
    \item \textbf{Dequantize} discrete inputs $\vecb x\leftarrow\vecb x+\vecb u$, $\vecb u\sim\Uniform([0,1)^D)$
    \item \textbf{Sample probe} $\vecb\epsilon\sim\Normal(\zeros,\eye_D)$ (or Rademacher $\{\pm 1\}^D$)
    \item \textbf{Integrate backward} $t:T\to 0$ from $[\vecb x,0]$ with augmented ODE $[\vecb v_\theta,\;-\vecb\epsilon\T\jac_{\vecb v_\theta}\vecb\epsilon]$ to get $[\vecb z(0),\Delta\ell]$
    \item \textbf{Evaluate NLL} $-\log p_Z(\vecb z(0))+\Delta\ell$; solve adjoint ODE for $\nabla_\theta\mathcal L$
  \end{enumerate}
\end{cscard}

\begin{cscard}{csE}{JACOBIAN SCALING CHEAT-LINE}
  \begin{csbul}
    \item Dense $\det\jac$: $O(D^3)$ $\;\to\;$ Triangular / Coupling: $O(D)$
    \item Exact $\Tr(\jac_{\vecb v_\theta})$: $D$ VJPs ($O(D^2)$) $\;\to\;$ Hutchinson: $1$ VJP ($O(D)$)
    \item Backprop through ODE solver: $O(\text{NFE})$ RAM $\;\to\;$ Adjoint: $O(1)$ RAM
  \end{csbul}
\end{cscard}

\begin{cscard}{csG}{PITFALL CHECKLIST}
  \begin{csbul}
    \item \textbf{Missing dequantization:} fitting continuous $p_X(\vecb x)$ to discrete pixels $\{0..255\}^D$ collapses density onto Dirac spikes ($\log p_X\to+\infty$); always apply uniform or variational dequantization
    \item \textbf{Floating-point non-invertibility:} large $\vecb s(\vecb z_A)$ causes $\exp(\vecb s)$ overflow/underflow so $f_\theta\inv(f_\theta(\vecb z))\ne\vecb z$; bound scale via $\tanh$ or $\sigmoid$
    \item \textbf{Forgotten channel mixing:} a single coupling layer leaves $\vecb z_A$ untouched; always alternate splits or insert $1\times 1$ LU convolutions
    \item \textbf{Topological obstruction:} diffeomorphisms preserve homotopy; splitting 1 Gaussian into disconnected modes causes $\norm{\jac_f}$ to explode in bridges
    \item \textbf{OOD likelihood paradox:} smooth backgrounds yield huge $\log\abs{\det\jac_{f\inv}}$; use typicality or likelihood ratios for OOD detection
  \end{csbul}
\end{cscard}

\begin{cscard}{csD}{10-SECOND VISUAL RECAP}
  \sffamily\footnotesize
  \textbf{Normalizing flows} turn a simple Gaussian $\vecb z\sim\Normal(\zeros,\eye_D)$ into complex data $\vecb x=f_\theta(\vecb z)$ via bijections.
  Conservation of probability scales density inversely with volume stretch: $p_X(\vecb x)=p_Z(\vecb z)/\abs{\det\jac_{f_\theta}(\vecb z)}$.
  Coupling/autoregressive layers make $\jac_{f_\theta}$ triangular, while Neural ODEs replace determinants with the divergence integral $-\int_0^T\nabla\cdot\vecb v_\theta\,\mathrm{d}t$.
\end{cscard}
