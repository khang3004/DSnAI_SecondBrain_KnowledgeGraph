\csbranch{1}{A}{1\;\; GUIDANCE $-$ CG, CFG \& Diffusion-DPO}{%
  \begin{minipage}[t]{0.59\linewidth}
    \csfm{\vecb s_\phi^{\mathrm{CFG}}(\vecb x_t,t,\vecb c;\omega)=\omega\vecb s_\phi(\vecb x_t,t,\vecb c)+(1-\omega)\vecb s_\phi(\vecb x_t,t,\emptyset)}
    \begin{csbul}
      \item \textbf{Bayes:} $\grad\log p_t(\vecb x_t\mid\vecb c)=\grad\log p_t(\vecb x_t)+\grad\log p_t(\vecb c\mid\vecb x_t)$
      \item \textbf{CFG:} drop $\vecb c\gets\emptyset$ ($10$--$20\%$) during training; $\omega>1$ extrapolates toward $\vecb c$ without external classifier $p_\psi$
      \item \textbf{Diffusion-DPO:} implicit reward $r_\pi=\beta\log\frac{\pi}{\pi_{\mathrm{ref}}}+\beta\log Z(\vecb c)$ cancels $Z(\vecb c)$ on $(\vecb x^w,\vecb x^l)$ pairs
    \end{csbul}

\csbranch{2}{B}{2\;\; SEMILINEAR ODE $-$ DDIM = Exponential Euler}{%
  \csfm{\dd{\vecb x_t}{t}=\underbrace{\frac{\alpha_t'}{\alpha_t}\vecb x_t}_{\text{linear drift}}+\underbrace{\frac{g^2(t)}{2\sigma_t}\boldsymbol\epsilon_\phi(\vecb x_t,t)}_{\text{nonlinear residual}}}
  \begin{csbul}
    \item Integrating factor $\alpha_t/\alpha_s$ integrates linear drift \textbf{analytically}
    \item Constant $\boldsymbol\epsilon_\phi(\tilde{\vecb x}_s,s)$ on $[t,s]$ yields \textbf{DDIM}:
      \csfm{\tilde{\vecb x}_t=\frac{\alpha_t}{\alpha_s}\tilde{\vecb x}_s-\alpha_t\!\left(\frac{\sigma_s}{\alpha_s}-\frac{\sigma_t}{\alpha_t}\right)\boldsymbol\epsilon_\phi(\tilde{\vecb x}_s,s)=\alpha_t\hat{\vecb x}_0+\sigma_t\boldsymbol\epsilon_\phi}
    \item $\vecb v$-pred: Euler $=$ DDIM; $\boldsymbol\epsilon/\vecb x/\vecb s$-pred: \textbf{Exp.\ Euler $=$ DDIM $\ne$ Euler}!
  \end{csbul}}

\csbranch{3}{C}{3\;\; HIGH-ORDER SOLVERS $-$ Log-SNR \& DPM-Solver++}{%
  \begin{csbul}
    \item \textbf{Half-log SNR change of variables:} $\lambda_t=\log(\alpha_t/\sigma_t)$ simplifies the exact ODE integral to
      \csfm{\tilde\Psi_{s\to t}(\vecb x_s)=\frac{\alpha_t}{\alpha_s}\vecb x_s-\alpha_t\int_{\lambda_s}^{\lambda_t}e^{-\lambda}\hat{\boldsymbol\epsilon}_\phi(\hat{\vecb x}_\lambda,\lambda)\,\mathrm{d}\lambda}
    \item \textbf{DPM-Solver-$n$:} $(n-1)$-th Taylor expansion in $\lambda$ gives global error $\mathcal O(h^n)$ ($h=\lambda_t-\lambda_s$)
    \item \textbf{DPM-Solver++ ($\vecb x$-pred + dynamic thresholding):} stabilizes large CFG $\omega\gg 1$:
      \csfm{\tilde\Psi_{s\to t}(\vecb x_s)=\frac{\sigma_t}{\sigma_s}\vecb x_s+\sigma_t\int_{\lambda_s}^{\lambda_t}e^{\lambda}\hat{\vecb x}_\phi(\hat{\vecb x}_\lambda,\lambda)\,\mathrm{d}\lambda}
  \end{csbul}}

\csbranch{4}{D}{4\;\; DISTILLATION $-$ Progressive Distillation \& Consistency}{%
  \begin{minipage}[t]{0.59\linewidth}
    \begin{csbul}
      \item \textbf{Progressive Distill.\ (PD):} match 2 teacher steps ($s\to u\to t$) in 1 step via $\tilde{\vecb x}=\frac{\sigma_s\tilde{\vecb x}_t-\sigma_t\vecb x_s}{\alpha_t\sigma_s-\alpha_s\sigma_t}$
      \item \textbf{Consistency (CM):} $f_\theta(\vecb x_s,s)\approx\Psi_{s\to 0}(\vecb x_s)$, $f_\theta(\vecb x_0,0)=\vecb x_0$
      \item \textbf{CD vs.\ CT:} $\mathcal L_{\mathrm{CM}}=\mathcal L_{\mathrm{CT}}+\mathcal O(\Delta s^2)$
      \item \textbf{sCM ($\Delta s\to 0$):} $\vecb x_s=\cos(s)\vecb x_0+\sin(s)\vecb z$
    \end{csbul}
  \end{minipage}\hfill

\csbranch{5}{E}{5\;\; GENERAL FLOW MAPS $-$ Semigroup, CTM \& Mean Flow}{%
  \csfm{\Psi_{u\to t}\circ\Psi_{s\to u}=\Psi_{s\to t},\qquad \Psi_{s\to s}=\mathrm{Id}\qquad(\forall\, s\ge u\ge t\ge 0)}
  \begin{csbul}
    \item \textbf{Two-time flow map ($s\to t$):} bridges instantaneous velocity ($s=t$) and 1-step jump ($t=0$):
      \csfm{\Psi_{s\to t}(\vecb x_s)\approx\underbrace{\frac{t}{s}\vecb x_s+\frac{s-t}{s}g_\theta(\vecb x_s,s,t)}_{\text{CTM}}=\underbrace{\vecb x_s+(t-s)h_\theta(\vecb x_s,s,t)}_{\text{Mean Flow (MF)}}}
    \item \textbf{Discrete state spaces (D3PM):} categorical transition matrices $q(\vecb x_t\mid\vecb x_0)=\Cat(\vecb x_0\bar{\mat Q}_t)$
  \end{csbul}}

\begin{cscard}{csA}{FORMULAS TO REMEMBER}
  \begin{csbul}
    \item Classifier-Free Guidance (with $w=\omega-1$):
      \csfm{\vecb s_\phi^{\mathrm{CFG}}(\vecb x_t,t,\vecb c;\omega)=\vecb s_\phi(\vecb x_t,t,\emptyset)+\omega\bigl(\vecb s_\phi(\vecb x_t,t,\vecb c)-\vecb s_\phi(\vecb x_t,t,\emptyset)\bigr)}
    \item DDIM update ($s\to t$, exact on linear drift $\frac{\alpha_t'}{\alpha_t}\vecb x_t$):
      \csfm{\tilde{\vecb x}_t=\frac{\alpha_t}{\alpha_s}\tilde{\vecb x}_s-\alpha_t\!\left(\frac{\sigma_s}{\alpha_s}-\frac{\sigma_t}{\alpha_t}\right)\boldsymbol\epsilon_\phi(\tilde{\vecb x}_s,s)=\alpha_t\hat{\vecb x}_0+\sigma_t\boldsymbol\epsilon_\phi}
    \item DPM-Solver-$n$ Taylor expansion in $\lambda_t=\log(\alpha_t/\sigma_t)$ ($h=\lambda_t-\lambda_s$):
      \csfm{\tilde{\vecb x}_t=\frac{\alpha_t}{\alpha_s}\tilde{\vecb x}_s-\alpha_t\sum_{k=0}^{n-1}\hat{\boldsymbol\epsilon}_\phi^{(k)}(\hat{\vecb x}_{\lambda_s},\lambda_s)\int_{\lambda_s}^{\lambda_t}e^{-\lambda}\frac{(\lambda-\lambda_s)^k}{k!}\mathrm{d}\lambda}
    \item Progressive Distillation 2-in-1 pseudo-clean target:
      \csfm{\tilde{\vecb x}=\frac{\sigma_s\tilde{\vecb x}_t-\sigma_t\vecb x_s}{\alpha_t\sigma_s-\alpha_s\sigma_t}}
  \end{csbul}
\end{cscard}

\begin{cscard}{csB}{RECIPE $-$ DPM-SOLVER++(2S) \& CONSISTENCY SAMPLING}
  \textbf{DPM-Solver++(2S) Midpoint Step} ($t_{i-1}\to t_i$, $h_i=\lambda_{t_i}-\lambda_{t_{i-1}}$):
  \begin{enumerate}[leftmargin=1.2em, itemsep=1pt, topsep=1pt, parsep=0pt]
    \item Midpoint $s_{\mathrm{mid}}=t_\lambda(\frac{1}{2}(\lambda_{t_{i-1}}+\lambda_{t_i}))$; threshold $\hat{\vecb x}_0=\hat{\vecb x}_\phi(\tilde{\vecb x}_{t_{i-1}},t_{i-1})$
    \item $\vecb u_i\gets\frac{\sigma_{s_{\mathrm{mid}}}}{\sigma_{t_{i-1}}}\tilde{\vecb x}_{t_{i-1}}-\alpha_{s_{\mathrm{mid}}}(e^{-h_i/2}-1)\hat{\vecb x}_0$
    \item $\tilde{\vecb x}_{t_i}\gets\frac{\sigma_{t_i}}{\sigma_{t_{i-1}}}\tilde{\vecb x}_{t_{i-1}}-\alpha_{t_i}(e^{-h_i}-1)\hat{\vecb x}_\phi(\vecb u_i,s_{\mathrm{mid}})$
  \end{enumerate}
  \vspace{2pt}
  \textbf{Multi-Step Consistency Sampling} ($\tau_1>\dots>\tau_{K-1}$):
  \begin{enumerate}[leftmargin=1.2em, itemsep=1pt, topsep=1pt, parsep=0pt]
    \item Draw $\hat{\vecb x}_T\sim\Normal(\zeros,\sigma_T^2\eye)$, predict $\hat{\vecb x}_0\gets f_\theta(\hat{\vecb x}_T,T)$
    \item Repeat: re-noise $\hat{\vecb x}_{\tau_k}\gets\alpha_{\tau_k}\hat{\vecb x}_0+\sigma_{\tau_k}\boldsymbol\epsilon$, denoise $\hat{\vecb x}_0\gets f_\theta(\hat{\vecb x}_{\tau_k},\tau_k)$
  \end{enumerate}
\end{cscard}

\begin{cscard}{csE}{CONVERGENCE \& EQUIVALENCE CHEAT-LINE}
  \begin{csbul}
    \item Solver global error: DDPM $\mathcal O(h^{1/2})$, DDIM $\mathcal O(h)$, DPM-Solver++(2S/2M) $\mathcal O(h^2)$
    \item CD vs.\ CT gap: $\mathcal L_{\mathrm{CM}}(\theta,\theta^-)=\mathcal L_{\mathrm{CT}}(\theta,\theta^-)+\mathcal O(\Delta s^2)$ (via shared $(\vecb x_0,\boldsymbol\epsilon)$)
    \item Continuous CM tangent: $\dd{}{s}f_{\theta^-}=\partial_s f_{\theta^-}+\partial_{\vecb x}f_{\theta^-}\,\vecb v^*$
    \item CTM $\leftrightarrow$ Mean Flow link: $g_\theta(\vecb x_s,s,t)=\vecb x_s-s\,h_\theta(\vecb x_s,s,t)$ (losses differ by $1/s^2$)
  \end{csbul}
\end{cscard}

\begin{cscard}{csG}{PITFALL CHECKLIST}
  \begin{csbul}
    \item \textbf{Plain Euler $\ne$ DDIM in $\boldsymbol\epsilon$-space:} naive Euler discretizes the stiff linear term $\frac{\alpha_t'}{\alpha_t}\vecb x_t$ numerically and diverges at $N\le 50$; always use exponential integrators
    \item \textbf{CFG doubles per-step NFE:} evaluating $\vecb s_\phi(\vecb x_t,t,\vecb c)$ and $\vecb s_\phi(\vecb x_t,t,\emptyset)$ requires $2\times$ batch pass unless distilled into an $\omega$-conditioned student
    \item \textbf{High-order overshoot at large $\omega$:} Taylor derivatives $\hat{\boldsymbol\epsilon}_\phi^{(k)}$ explode when $\omega\gg 1$ $\to$ switch to $\vecb x$-prediction (DPM-Solver++) with dynamic percentile clipping
    \item \textbf{Consistency collapse without stop-grad:} omitting $\mathrm{sg}(\theta^-)$ lets $f_\theta(\vecb x,s)$ collapse to a constant vector in 1 step
    \item \textbf{Trigonometric sCM scaling:} continuous-time CT requires bounded $c_{\mathrm{noise}}(s)=s$ and trig schedule $\cos(s)\vecb x_0+\sin(s)\vecb z$ to prevent tangent explosion
  \end{csbul}
\end{cscard}

\begin{cscard}{csD}{10-SECOND VISUAL RECAP}
