\csbranch{2}{B}{2\;\; HOW $-$ Variational ELBO vs.\ Denoising Score Matching}{%
  \begin{csbul}
    \item \textbf{ELBO decomposition:} $-\log p_\theta(\vecb x_0)\le \mathcal L_T + \sum_{t=2}^T \mathcal L_{t-1} + \mathcal L_0$, where $\mathcal L_{t-1}=\KL{q(\vecb x_{t-1}\mid\vecb x_t,\vecb x_0)}{p_\theta(\vecb x_{t-1}\mid\vecb x_t)}$
    \item \textbf{Ho et al.\ unweighted loss:} drop time prefactors $\Rightarrow$ predict noise $\boldsymbol\epsilon\sim\Normal(\zeros,\eye)$:
    \csfm{\mathcal L_{\mathrm{simple}}(\theta)=\E_{t,\vecb x_0,\boldsymbol\epsilon}\!\Bigl[\norm{\boldsymbol\epsilon-\boldsymbol\epsilon_\theta(\sqrt{\bar\alpha_t}\vecb x_0+\sqrt{1-\bar\alpha_t}\boldsymbol\epsilon,\,t)}_2^2\Bigr]}
    \item \textbf{Vincent's DSM theorem:} replacing intractable $\grad_{\tilde{\vecb x}}\log q_\sigma(\tilde{\vecb x})$ by conditional $\grad_{\tilde{\vecb x}}\log q_\sigma(\tilde{\vecb x}\mid\vecb x_0)=-\boldsymbol\epsilon/\sigma$ changes loss only by a $\theta$-constant: $J_{\mathrm{DSM}}(\theta;\sigma)=J_{\mathrm{ESM}}(\theta;\sigma)+C$
  \end{csbul}}

\csbranch{3}{C}{3\;\; KEY THEOREM $-$ Tweedie's Formula \& 4-Way Equivalence}{%
  \begin{minipage}[t]{0.58\linewidth}
    \csfm{\grad_{\vecb x_t}\log q_t(\vecb x_t)=-\frac{\E[\boldsymbol\epsilon\mid\vecb x_t]}{\sqrt{1-\bar\alpha_t}}=\frac{\sqrt{\bar\alpha_t}\E[\vecb x_0\mid\vecb x_t]-\vecb x_t}{1-\bar\alpha_t}}
    \begin{csbul}
      \item Affine kernel $\vecb x_t=\alpha_t^{\mathrm{aff}}\vecb x_0+\sigma_t\boldsymbol\epsilon$
      \item $\vecb s_\theta \equiv -\boldsymbol\epsilon_\theta/\sigma_t \equiv (\alpha_t^{\mathrm{aff}}\vecb x_\theta-\vecb x_t)/\sigma_t^2$
      \item Velocity: $\vecb v_\theta = (\alpha_t^{\mathrm{aff}})'\vecb x_\theta + \sigma_t'\boldsymbol\epsilon_\theta$
      \item All 4 squared losses match up to scalar weights $w(t)$!
    \end{csbul}
  \end{minipage}\hfill
  \begin{minipage}[t]{0.39\linewidth}\centering\vspace{2pt}

\csbranch{6}{F}{6\;\; USE-CASES $-$ Why Multi-Scale Noise Bridges Work}{%
  \begin{csbul}
    \item \textbf{Bypasses intractable partition $Z_\theta$:} scores $\grad_{\vecb x}\log p_\theta(\vecb x)=-\grad_{\vecb x}E_\theta(\vecb x)$ drop $Z_\theta$ completely
    \item \textbf{Heals manifold blindness:} Gaussian smoothing $\sigma_i>0$ spreads support over all $\R^D$
    \item \textbf{Fixes mode-weight blindness:} large $\sigma_1$ bridges disjoint modes $\pi p_0+(1-\pi)p_1$ so Langevin dynamics recovers true mixture weights $\pi$
    \item \textbf{Foundation for controllable synthesis:} plug-and-play posterior score $\grad_{\vecb x_t}\log p(\vecb y\mid\vecb x_t)$
  \end{csbul}}

\begin{cscard}{csA}{FORMULAS TO REMEMBER}
  \begin{csbul}
    \item Forward marginal ($t$-step jump):
      \csfm{\vecb x_t=\sqrt{\bar\alpha_t}\,\vecb x_0+\sqrt{1-\bar\alpha_t}\,\boldsymbol\epsilon,\quad \bar\alpha_t=\prod_{s=1}^t(1-\beta_s)}
    \item Tractable reverse conditional posterior $q(\vecb x_{t-1}\mid\vecb x_t,\vecb x_0)$:
      \csfm{\tilde{\boldsymbol\mu}_t=\frac{\sqrt{\bar\alpha_{t-1}}\beta_t}{1-\bar\alpha_t}\vecb x_0+\frac{\sqrt{\alpha_t}(1-\bar\alpha_{t-1})}{1-\bar\alpha_t}\vecb x_t,\quad \tilde\beta_t=\frac{1-\bar\alpha_{t-1}}{1-\bar\alpha_t}\beta_t}
    \item Posterior mean in $\boldsymbol\epsilon$-form:
      \csfm{\vecb\mu_\theta(\vecb x_t,t)=\frac{1}{\sqrt{\alpha_t}}\!\left(\vecb x_t-\frac{\beta_t}{\sqrt{1-\bar\alpha_t}}\boldsymbol\epsilon_\theta(\vecb x_t,t)\right)}
    \item Tweedie \& 4-way head conversions ($\vecb x_t=\alpha_t^{\mathrm{aff}}\vecb x_0+\sigma_t\boldsymbol\epsilon$):
      \csfm{\vecb s_\theta = -\frac{\boldsymbol\epsilon_\theta}{\sigma_t} = \frac{\alpha_t^{\mathrm{aff}}\vecb x_\theta-\vecb x_t}{\sigma_t^2},\quad \vecb v_\theta=(\alpha_t^{\mathrm{aff}})'\vecb x_\theta+\sigma_t'\boldsymbol\epsilon_\theta}
  \end{csbul}
\end{cscard}

\begin{cscard}{csB}{RECIPE $-$ DDPM vs.\ NCSN (ALD) SAMPLING}
  \textbf{DDPM Ancestral Sampling} ($\vecb x_T\sim\Normal(\zeros,\eye)$):
  \begin{enumerate}[leftmargin=1.2em, itemsep=1pt, topsep=1pt, parsep=0pt]
    \item For $t=T,T-1,\dots,1$: draw $\vecb z\sim\Normal(\zeros,\eye)$ if $t>1$ else $\zeros$
    \item Update $\vecb x_{t-1}\gets\frac{1}{\sqrt{\alpha_t}}\bigl(\vecb x_t-\frac{\beta_t}{\sqrt{1-\bar\alpha_t}}\boldsymbol\epsilon_\theta(\vecb x_t,t)\bigr)+\sigma_t\vecb z$
  \end{enumerate}
  \vspace{2pt}
  \textbf{Annealed Langevin Dynamics} ($\sigma_1>\dots>\sigma_L$):
  \begin{enumerate}[leftmargin=1.2em, itemsep=1pt, topsep=1pt, parsep=0pt]
    \item For $i=1,\dots,L$: set step size $\tau_i=\epsilon_0(\sigma_i/\sigma_L)^2$
    \item Run $K$ steps: $\tilde{\vecb x}_k\gets\tilde{\vecb x}_{k-1}+\frac{\tau_i}{2}\vecb s_\theta(\tilde{\vecb x}_{k-1},\sigma_i)+\sqrt{\tau_i}\,\vecb z_k$
  \end{enumerate}
\end{cscard}

\begin{cscard}{csE}{LOSS WEIGHTING \& SCALING CHEAT-LINE}
  \begin{csbul}
    \item $\norm{\vecb s_\theta-\vecb s^*}_2^2 = \frac{1}{\sigma_t^2}\norm{\boldsymbol\epsilon_\theta-\boldsymbol\epsilon^*}_2^2 = \frac{(\alpha_t^{\mathrm{aff}})^2}{\sigma_t^4}\norm{\vecb x_\theta-\vecb x_0^*}_2^2$
    \item Multi-scale DSM weight: $\lambda(\sigma_i)=\sigma_i^2 \Rightarrow$ balanced unit-scale target $\norm{\sigma_i\vecb s_\theta(\tilde{\vecb x},\sigma_i)+\boldsymbol\epsilon}_2^2$
    \item Reverse variance bounds: $\tilde\beta_t \le \sigma_t^2 \le \beta_t$ (identical as $T\to\infty$)
  \end{csbul}
\end{cscard}

\begin{cscard}{csG}{PITFALL CHECKLIST}
  \begin{csbul}
    \item \textbf{Manifold \& low-density blindness:} never train score matching on clean data $\vecb x_0$ alone; $\grad_{\vecb x}\log p_{\mathrm{data}}$ is undefined off-manifold and noisy in low-density voids
    \item \textbf{Disjoint mode mixing:} $\grad_{\vecb x}\log(\pi p_0+(1-\pi)p_1)$ drops $\pi$ when supports are disjoint $\to$ require $\sigma_1$ large enough to overlap modes
    \item \textbf{Endpoint blow-up:} $\vecb s_\theta=-\boldsymbol\epsilon_\theta/\sigma_t$ diverges as $\sigma_t\to 0$, while $\vecb x_\theta$ loses signal as $\alpha_t^{\mathrm{aff}}\to 0$ $\to$ use $\vecb v_\theta$ or clip $\sigma_t\ge\epsilon$
    \item \textbf{Final step noise:} always set $\vecb z=\zeros$ at $t=1\to 0$ in ancestral sampling so output returns the clean posterior mean
    \item \textbf{Non-Gaussian large steps:} skipping $t$ in standard DDPM fails because $q(\vecb x_{t-k}\mid\vecb x_t)$ is multimodal for large $k$
  \end{csbul}
\end{cscard}

\begin{cscard}{csD}{10-SECOND VISUAL RECAP}
  \sffamily\footnotesize
  \textbf{DDPM \& NCSN are two faces of the same Gaussian bridge.}
  Forward diffusion jumps in one shot via $\vecb x_t=\sqrt{\bar\alpha_t}\vecb x_0+\sqrt{1-\bar\alpha_t}\boldsymbol\epsilon$.
  Conditioning on $\vecb x_0$ makes the reverse step an exact Gaussian, and \textbf{Tweedie's identity} $\vecb s^*=-\boldsymbol\epsilon^*/\sigma_t$ proves that predicting added noise is identical to learning the Stein score across noise scales.
\end{cscard}
