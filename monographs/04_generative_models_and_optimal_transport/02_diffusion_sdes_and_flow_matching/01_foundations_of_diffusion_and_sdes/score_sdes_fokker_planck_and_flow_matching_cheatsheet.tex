\csbranch{1}{A}{1\;\; WHAT $-$ Continuum Limit: VP-SDE \& VE-SDE}{%
  \csfm{\mathrm{d}\vecb x_t=\vecb f(\vecb x_t,t)\,\mathrm{d}t+g(t)\,\mathrm{d}\vecb w_t,\quad \mathrm{d}\vecb w_t\sim\Normal(\zeros,\mathrm{d}t\,\eye)}
  \begin{csbul}
    \item \textbf{DDPM $\to$ VP-SDE:} $\mathrm{d}\vecb x_t=-\frac{1}{2}\beta(t)\vecb x_t\,\mathrm{d}t+\sqrt{\beta(t)}\,\mathrm{d}\vecb w_t$
      \\ Kernel: $\Normal\bigl(e^{-\frac{1}{2}B(t)}\vecb x_0,\,(1-e^{-B(t)})\eye\bigr)$, $B(t)=\int_0^t\beta(s)\,\mathrm{d}s$
    \item \textbf{NCSN $\to$ VE-SDE:} $\mathrm{d}\vecb x_t=\sqrt{\frac{\mathrm{d}[\sigma^2(t)]}{\mathrm{d}t}}\,\mathrm{d}\vecb w_t$
      \\ Kernel: $\Normal\bigl(\vecb x_0,\,(\sigma^2(t)-\sigma^2(0))\eye\bigr)$
    \item Linear drift $\vecb f(\vecb x,t)=f(t)\vecb x \Rightarrow$ exact Gaussian transition $p_t(\vecb x_t\mid\vecb x_0)$
  \end{csbul}}

\csbranch{2}{B}{2\;\; HOW $-$ It\^o's Lemma \& Fokker-Planck PDE}{%
  \begin{minipage}[t]{0.60\linewidth}
    \csfm{(\mathrm{d}w_{t,i})(\mathrm{d}w_{t,j})=\delta_{ij}\,\mathrm{d}t}
    \begin{csbul}
      \item \textbf{It\^o's Formula:} $\mathrm{d}\varphi = \bigl(\partial_t\varphi + (\grad\varphi)\T\vecb f + \frac{1}{2}g^2\Delta\varphi\bigr)\mathrm{d}t + g(\grad\varphi)\T\mathrm{d}\vecb w_t$
      \item \textbf{Fokker-Planck (IBP on $\E[\mathrm{d}\varphi]$):}
      \csfm{\partial_t p_t = -\nabla\!\cdot\!(\vecb f p_t)+\tfrac{1}{2}g^2\Delta p_t}
      \item Using $\Delta p_t=\nabla\!\cdot\!(p_t\grad\log p_t)$ converts diffusion into a transport continuity PDE!
    \end{csbul}
  \end{minipage}\hfill
  \begin{minipage}[t]{0.37\linewidth}\centering\vspace{4pt}
    \begin{tikzpicture}[scale=0.58, every node/.style={font=\tiny}]
      \node[draw=csB, fill=csB!12, rounded corners=2pt, inner sep=2.5pt, align=center] (sde) at (0,3.0) {Sample path SDE\\$\mathrm{d}\vecb x_t=\vecb f\mathrm{d}t+g\mathrm{d}\vecb w_t$};
      \node[draw=csB, fill=white, rounded corners=2pt, inner sep=2.5pt, align=center] (ito) at (0,1.5) {It\^o + $\E[\cdot]$ + IBP\\$\frac{1}{2}g^2\Delta\varphi\to\frac{1}{2}g^2\Delta p_t$};
      \node[draw=csB, fill=csB!12, rounded corners=2pt, inner sep=2.5pt, align=center] (fp) at (0,0) {Continuity PDE\\$\partial_t p_t+\nabla\!\cdot\!(p_t\tilde{\vecb v}_t)=0$};

    \end{tikzpicture}
  \end{minipage}}

\csbranch{3}{C}{3\;\; KEY THEOREM $-$ Anderson Reverse SDE \& PF-ODE}{%
  \begin{csbul}
    \item \textbf{Reverse-Time SDE} ($\bar{\vecb w}_t$ reverse Wiener, $t:T\to 0$):
      \csfm{\mathrm{d}\bar{\vecb x}_t=\bigl[\vecb f(\bar{\vecb x}_t,t)-g^2(t)\grad_{\vecb x}\log p_t(\bar{\vecb x}_t)\bigr]\mathrm{d}t+g(t)\,\mathrm{d}\bar{\vecb w}_t}
    \item \textbf{Probability Flow ODE} (deterministic, shares \textbf{exact} $p_t(\vecb x)$):
      \csfm{\mathrm{d}\tilde{\vecb x}_t=\underbrace{\bigl[\vecb f(\tilde{\vecb x}_t,t)-\tfrac{1}{2}g^2(t)\grad_{\vecb x}\log p_t(\tilde{\vecb x}_t)\bigr]}_{\tilde{\vecb v}_t(\tilde{\vecb x}_t)}\mathrm{d}t}
    \item \textbf{Why $\frac{1}{2}g^2$ vs.\ $g^2$?} Rev.\ SDE = PF-ODE $+$ time-varying Langevin corrector $\bigl(-\frac{1}{2}g^2\grad\log p_t\,\mathrm{d}t+g\,\mathrm{d}\bar{\vecb w}_t\bigr)$ that damps score error
  \end{csbul}}

\csbranch{4}{D}{4\;\; CONNECTIONS $-$ Girsanov Path KL \& Exact NLL}{%
  \begin{csbul}
    \item \textbf{Girsanov Theorem:} for same $g(t)$ and drift gap $g^2(\vecb s_\phi-\grad\log p_t)$, path KL is weighted DSM:
      \csfm{\KL{\mathbb P}{\mathbb P_\phi}=\frac{1}{2}\E_{\mathbb P}\!\left[\int_0^T g^2(t)\norm{\vecb s_\phi(\vecb x_t,t)-\grad\log p_t(\vecb x_t)}_2^2\mathrm{d}t\right]}
    \item Data processing: $\KL{p_{\mathrm{data}}}{p_\phi(\vecb x_0)}\le\KL{\mathbb P}{\mathbb P_\phi}$ ($\lambda(t)=g^2(t)$ = NLL weight)
    \item \textbf{PF-ODE exact NLL:} $\log p_0(\vecb x_0)=\log p_T(\vecb x_T)+\int_0^T \nabla_{\vecb x}\cdot\tilde{\vecb v}_t(\vecb x_t)\,\mathrm{d}t$
  \end{csbul}}

\csbranch{5}{E}{5\;\; FLOW MATCHING $-$ CFM, Rectified Flow \& Reflow}{%
  \begin{minipage}[t]{0.59\linewidth}
    \begin{csbul}
      \item Learn $\dot{\vecb x}_t=\vecb v_\phi(\vecb x_t,t)$ pushing $p_{\mathrm{src}}\to p_{\mathrm{tgt}}$ directly
      \item \textbf{CFM:} $\grad_\phi\mathcal L_{\mathrm{FM}}=\grad_\phi\mathcal L_{\mathrm{CFM}}$ for $\vecb v_t(\vecb x)=\E[\vecb v_t(\vecb x\mid\vecb z)\mid\vecb x_t=\vecb x]$
      \item \textbf{Rectified Flow} ($\vecb x_t=(1-t)\vecb x_0+t\vecb x_1$):
        \csfm{\mathcal L_{\mathrm{RF}}=\E_{t,(\vecb x_0,\vecb x_1)}\!\bigl[\norm{\vecb v_\phi(\vecb x_t,t)-(\vecb x_1-\vecb x_0)}_2^2\bigr]}
      \item \textbf{Reflow} $(\vecb x_0,\Phi_{0\to 1}^{(k)}(\vecb x_0))$ straightens paths: $\mathcal S\le \mathcal O(1/K)$
    \end{csbul}
  \end{minipage}\hfill

\begin{cscard}{csA}{FORMULAS TO REMEMBER}
  \begin{csbul}
    \item Linear SDE moment ODEs ($\vecb f=f(t)\vecb x$, $p_t(\vecb x_t\mid\vecb x_0)=\Normal(\vecb m_t,P_t\eye)$):
      \csfm{\dot{\vecb m}_t=f(t)\vecb m_t,\qquad \dot P_t=2f(t)P_t+g^2(t)}
    \item Multidimensional It\^o's Lemma:
      \csfm{\mathrm{d}\varphi(\vecb x_t,t)=\Bigl(\partial_t\varphi+(\grad\varphi)\T\vecb f+\tfrac{1}{2}g^2(t)\Tr(\grad^2\varphi)\Bigr)\mathrm{d}t+g(t)(\grad\varphi)\T\mathrm{d}\vecb w_t}
    \item Fokker-Planck $\Longleftrightarrow$ Continuity Equation:
      \csfm{\partial_t p_t=-\nabla\!\cdot\!(p_t\vecb f)+\tfrac{1}{2}g^2\Delta p_t=-\nabla\!\cdot\!\Bigl(p_t\bigl[\vecb f-\tfrac{1}{2}g^2\grad\log p_t\bigr]\Bigr)}
    \item Gaussian Bridge Conditional Velocity ($p_t(\vecb x\mid\vecb z)=\Normal(\vecb\mu_t,\sigma_t^2\eye)$):
      \csfm{\vecb v_t(\vecb x\mid\vecb z)=\frac{\sigma_t'(\vecb z)}{\sigma_t(\vecb z)}\bigl(\vecb x-\vecb\mu_t(\vecb z)\bigr)+\vecb\mu_t'(\vecb z)}
  \end{csbul}
\end{cscard}

\begin{cscard}{csB}{RECIPE $-$ SIMULATION-FREE CFM \& REVERSE SAMPLING}
  \textbf{CFM / Rectified Flow Training:}
  \begin{enumerate}[leftmargin=1.2em, itemsep=1pt, topsep=1pt, parsep=0pt]
    \item Sample $t\sim\Uniform[0,1]$ and pair $(\vecb x_0,\vecb x_1)\sim\pi$ (e.g.\ $p_{\mathrm{src}}\otimes p_{\mathrm{tgt}}$)
    \item Interpolate $\vecb x_t=(1-t)\vecb x_0+t\vecb x_1$ and step on $\norm{\vecb v_\phi(\vecb x_t,t)-(\vecb x_1-\vecb x_0)}_2^2$
  \end{enumerate}
  \vspace{2pt}
  \textbf{Reverse Euler-Maruyama vs.\ PF-ODE} ($\Delta t=T/N$, $t_k=k\Delta t$):
  \begin{enumerate}[leftmargin=1.2em, itemsep=1pt, topsep=1pt, parsep=0pt]
    \item \textbf{PF-ODE:} $\vecb x\gets\vecb x-\bigl[\vecb f(\vecb x,t)-\frac{1}{2}g^2(t)\vecb s_\phi(\vecb x,t)\bigr]\Delta t$
    \item \textbf{Rev-SDE:} $\vecb x\gets\vecb x-\bigl[\vecb f(\vecb x,t)-g^2(t)\vecb s_\phi(\vecb x,t)\bigr]\Delta t+g(t)\sqrt{\Delta t}\,\vecb z$
  \end{enumerate}
\end{cscard}

\begin{cscard}{csE}{GIRSANOV \& REFLOW BOUNDS CHEAT-LINE}
  \begin{csbul}
    \item Girsanov Radon-Nikodym ratio ($\vecb\delta_t=\vecb f-\tilde{\vecb f}$):
      $\log\frac{\mathrm{d}\mathbb P_{\vecb f}}{\mathrm{d}\mathbb P_{\tilde{\vecb f}}}=\int_0^T\frac{\vecb\delta_t\T}{g}\mathrm{d}\vecb w_t-\frac{1}{2}\int_0^T\frac{\norm{\vecb\delta_t}_2^2}{g^2}\mathrm{d}t$
    \item Reflow convex transport cost never increases: $\E[c(\vecb x_1^{(k+1)}-\vecb x_0^{(k+1)})]\le\E[c(\vecb x_1^{(k)}-\vecb x_0^{(k)})]$
    \item Straightness decay after $K$ reflows: $\min_{k\le K}\mathcal S(\vecb Z^{(k)})\le\frac{1}{K}\E[\norm{\vecb x_1-\vecb x_0}_2^2]$
  \end{csbul}
\end{cscard}

\begin{cscard}{csG}{PITFALL CHECKLIST}
  \begin{csbul}
    \item \textbf{Missing the $\frac{1}{2}$ in PF-ODE:} using $-\vecb f+g^2\vecb s_\phi$ without Brownian noise $g\mathrm{d}\bar{\vecb w}_t$ changes the Fokker-Planck PDE and corrupts the generated marginals $p_t(\vecb x)$
    \item \textbf{Classical chain rule on SDEs:} $\mathrm{d}\vecb w_t=\mathcal O(\sqrt{\mathrm{d}t})$ means second-order Taylor terms $\frac{1}{2}g^2\Delta\varphi\,\mathrm{d}t$ are first-order in $\mathrm{d}t$ and can never be dropped
    \item \textbf{Score singularity at $t\to 0$:} $\sigma_t\to 0$ causes $-\boldsymbol\epsilon/\sigma_t$ to explode on manifold data $\to$ truncate integration at $t=\varepsilon\in[10^{-5},10^{-3}]$
    \item \textbf{VE-SDE prior bias:} $\sigma(T)\gg\sigma(0)$ is required so $p_T(\vecb x_T)\approx\Normal(\zeros,\sigma^2(T)\eye)$ forgets $\vecb x_0$
    \item \textbf{Straight $\ne$ OT:} a rotated coupling $\vecb x_1=\mat A\vecb x_0$ produces straight lines $\vecb x_t=((1-t)\eye+t\mat A)\vecb x_0$ but has suboptimal transport cost
  \end{csbul}
\end{cscard}

\begin{cscard}{csD}{10-SECOND VISUAL RECAP}
  \sffamily\footnotesize
  \textbf{One marginal path $\{p_t\}$, three generative engines:}
  \textbf{It\^o + Fokker-Planck} proves that stochastic Reverse SDEs ($-g^2\vecb s_\phi\,\mathrm{d}t+g\mathrm{d}\bar{\vecb w}_t$) and deterministic PF-ODEs ($-\frac{1}{2}g^2\vecb s_\phi\,\mathrm{d}t$) traverse the exact same density bridge, while \textbf{Conditional Flow Matching} and \textbf{Rectified Flow} straighten trajectories into $\vecb x_t=(1-t)\vecb x_0+t\vecb x_1$ for simulation-free, few-step transport.
\end{cscard}
