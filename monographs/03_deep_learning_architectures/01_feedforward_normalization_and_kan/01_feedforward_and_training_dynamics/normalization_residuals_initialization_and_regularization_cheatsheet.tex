\csbranch{1}{A}{1\;\; INIT $-$ Variance Preservation at $t=0$}{%
  \begin{minipage}[t]{0.59\linewidth}
    \csfm{\Var(f_i^{(l)})=D_{l-1}\sigma_l^2\,\E\bigl[(h_j^{(l-1)})^2\bigr]}
    \begin{csbul}
      \item \textbf{Xavier ($\tanh$/linear):} $\E[h^2]\approx\Var(f)$
        $\Rightarrow \sigma^2=\frac{2}{D_{l-1}+D_l}$
      \item \textbf{Kaiming/He ($\ReLU$):} clips negative half $\Rightarrow \E[h^2]=\frac{1}{2}\Var(f)\Rightarrow \sigma^2=\frac{2}{D_{l-1}}$
      \item Without $\times 2$, variance decays as $2^{-L}$!
    \end{csbul}
  \end{minipage}\hfill

\csbranch{2}{B}{2\;\; NORMALIZATION $-$ Which Axes to Slice?}{%
  \renewcommand{\arraystretch}{1.14}\scriptsize
  \begin{tabular}{@{}>{\raggedright\arraybackslash}p{1.6cm}>{\raggedright\arraybackslash}p{1.75cm}>{\raggedright\arraybackslash}p{1.15cm}>{\raggedright\arraybackslash}p{2.5cm}@{}}
    \textbf{\textcolor{csB}{Method}} & \textbf{\textcolor{csB}{Axes $\mathcal{S}_i$}} & \textbf{\textcolor{csB}{Gain}} & \textbf{\textcolor{csB}{Key Property}}\\ \hline
    BatchNorm & $(N,H,W)$ & $2C$ & couples batch; EMA at eval\\
    LayerNorm & $(C,H,W)$ or $D$ & $2D$ & batch-free; exact at eval\\
    RMSNorm & feature $D$ & $D$ & no mean $\mu$; fast LLM norm\\
    GroupNorm & $(C/G,H,W)$ & $2C$ & $G{=}1\to\text{LN}$, $G{=}C\to\text{IN}$\\
  \end{tabular}\par\vspace{2pt}
  \small\textbf{RMSNorm:} $\bar x_i = x_i/\sqrt{\frac{1}{D}\sum_{j=1}^D x_j^2+\epsilon},\; y_i=\gamma_i\bar x_i$.}

\csbranch{3}{C}{3\;\; RESIDUALS $-$ The $\eye$ Gradient Highway}{%
  \begin{minipage}[t]{0.62\linewidth}
    \csfm{\pd{\mathcal{L}}{\vecb x^{(l)}}=\pd{\mathcal{L}}{\vecb x^{(L)}}\!\left(\eye+\pd{}{\vecb x^{(l)}}\sum_{k=l}^{L-1}\mathcal{F}_k\right)}
    \begin{csbul}
      \item Fixes \textbf{shattered gradients} \& smooths loss landscape
      \item Implicit ensemble of $2^L$ paths
      \item \textbf{Unnormalized ResNet:} scale branch by $1/\sqrt{L}$ so $\Var(\vecb x^{(L)})=\mathcal{O}(1)$ instead of $2^L$
    \end{csbul}
  \end{minipage}\hfill

\csbranch{4}{D}{4\;\; PRE-LN vs.\ POST-LN $-$ Transformer Conditioning}{%
  \begin{csbul}
    \item \textbf{Post-LN:} $\vecb x^{(l+1)}=\mathrm{Norm}(\vecb x^{(l)}+\mathcal{F}_l(\vecb x^{(l)}))$ $\Rightarrow$ skip path passes \emph{through} Norm Jacobian at every layer; gradients attenuate with depth $\Rightarrow$ requires LR warmup
    \item \textbf{Pre-LN:} $\vecb x^{(l+1)}=\vecb x^{(l)}+\mathcal{F}_l(\mathrm{Norm}(\vecb x^{(l)}))$ $\Rightarrow$ clean identity $\eye$ highway from input to output; stable gradients at $t=0$ without warmup (standard in GPT/LLaMA)
    \item \textbf{Final Norm:} Pre-LN requires one final $\mathrm{Norm}$ right before the readout head
  \end{csbul}}

\csbranch{5}{E}{5\;\; REGULARIZATION $-$ Curvature \& Ensembles}{%
  \begin{csbul}
    \item \textbf{$L_2$ Weight Decay ($\frac{\lambda}{2}\norm{\vecb w}_2^2$):} $\tilde w_i'=\frac{\Lambda_{ii}}{\Lambda_{ii}+\lambda}(w_i^*)'$ in Hessian eigenbasis $\mat H=\mat Q\mat\Lambda\mat Q\T$; shrinks flat directions ($\Lambda_{ii}\ll\lambda$), keeps sharp ones ($\Lambda_{ii}\gg\lambda$)
    \item \textbf{Inverted Dropout:} $\tilde{\vecb h}=\frac{1}{p}(\vecb m\odot\vecb h)$, $m_j\sim\Bernoulli(p)$; test time uses untouched weights; exact geometric-mean ensemble of $2^D$ sub-networks for softmax
    \item \textbf{Early Stopping ($T$ steps):} acts like $L_2$ penalty with $\lambda_{\text{eff}}\propto (\eta T)^{-1}$
  \end{csbul}}

\csbranch{6}{F}{6\;\; DATA \& TARGET REGULARIZERS}{%
  \begin{minipage}[t]{0.59\linewidth}
    \begin{csbul}
      \item \textbf{Label Smoothing:} $\tilde{\vecb y}=(1-\varepsilon)\vecb y+\frac{\varepsilon}{K}\ones$ prevents logit blow-up $\norm{\vecb z}\to\infty$
      \item \textbf{Mixup:} $\tilde{\vecb x}=\alpha\vecb x_i+(1-\alpha)\vecb x_j$, $\tilde{\vecb y}=\alpha\vecb y_i+(1-\alpha)\vecb y_j$ ($\alpha\sim\mathrm{Beta}(\nu,\nu)$) linearizes between manifolds
    \end{csbul}
  \end{minipage}\hfill
  \begin{minipage}[t]{0.38\linewidth}\centering\vspace{2pt}

\csnote{8}{A}{TRAPS $-$ before you trust the output}{%
  \begin{csbul}
    \item \textbf{Xavier on ReLU:} misses $1/2$ clipping factor $\Rightarrow 2^{-L}$ collapse
    \item \textbf{BatchNorm ($N\le 4$) / no \texttt{eval()}:} noisy stats or test leakage (use GN/LN)
    \item \textbf{Dropout before BN:} $1/p$ train var inflates $\hat\sigma_{\text{run}}^2$, breaking eval
  \end{csbul}}

\begin{cscard}{csA}{FORMULAS TO REMEMBER}
  \begin{csbul}
    \item Xavier vs.\ Kaiming/He Init:\csfm{\sigma_{\text{Xavier}}^2=\frac{2}{D_{l-1}+D_l},\qquad \sigma_{\text{He}}^2=\frac{2}{D_{l-1}}}
    \item General Norm \& RMSNorm:\csfm{Y_i=\gamma_c\frac{X_i-\mu_i}{\sqrt{\sigma_i^2+\epsilon}}+\beta_c,\qquad y_i=\gamma_i\frac{x_i}{\sqrt{\frac{1}{D}\sum_{j=1}^D x_j^2+\epsilon}}}
    \item BatchNorm Coupled Backward Gradient:\csfm{\pd{\mathcal{L}}{x_n}=\frac{1}{\sqrt{\sigma_B^2+\epsilon}}\!\left[\pd{\mathcal{L}}{\hat x_n}-\frac{1}{N}\sum_{m=1}^N\pd{\mathcal{L}}{\hat x_m}-\frac{\hat x_n}{N}\sum_{m=1}^N\pd{\mathcal{L}}{\hat x_m}\hat x_m\right]}
    \item Pre-LN Residual Block:\csfm{\vecb x^{(l+1)}=\vecb x^{(l)}+\mathcal{F}_l\!\left(\mathrm{Norm}(\vecb x^{(l)})\right)}
    \item $L_2$ Weight Decay Spectral Filter:\csfm{\tilde w_i'=\frac{\Lambda_{ii}}{\Lambda_{ii}+\lambda}(w_i^*)'\quad (\mat H=\mat Q\mat\Lambda\mat Q\T)}
  \end{csbul}
\end{cscard}

\begin{cscard}{csB}{RECIPE: PRE-LN RMSNORM + BN TRAIN/EVAL}
  \begin{enumerate}[leftmargin=1.2em, itemsep=1pt, topsep=1pt, parsep=0pt]
    \item \textbf{Pre-LN RMSNorm:} compute $\mathrm{rms}=\sqrt{\frac{1}{D}\sum_j (x_j^{(l)})^2+\epsilon}$, scale $\bar{\vecb x}=\vecb\gamma\odot(\vecb x^{(l)}/\mathrm{rms})$, and add to clean highway $\vecb x^{(l+1)}=\vecb x^{(l)}+\mathcal{F}_l(\bar{\vecb x})$.
    \item \textbf{BN Train:} compute $\vecb\mu_B,\vecb\sigma_B^2$ over $(N,H,W)$, normalize, and update EMA buffers $\hat{\vecb\mu}\leftarrow(1-\rho)\hat{\vecb\mu}+\rho\vecb\mu_B$, $\hat{\vecb\sigma}^2\leftarrow(1-\rho)\hat{\vecb\sigma}^2+\frac{N}{N-1}\rho\vecb\sigma_B^2$.
    \item \textbf{BN Eval:} freeze running $(\hat{\vecb\mu},\hat{\vecb\sigma}^2)$ so inference is deterministic per sample.
  \end{enumerate}
\end{cscard}

\begin{cscard}{csC}{DECISION GUIDE}
  \renewcommand{\arraystretch}{1.22}\small
  \begin{tabular}{@{}>{\raggedright\arraybackslash}p{3.2cm}>{\raggedright\arraybackslash}p{4.6cm}@{}}
    \textbf{If you have\ldots} & \textbf{use\ldots}\\ \hline
    ReLU / GELU / SiLU MLP or CNN & Kaiming/He init ($\sigma^2=2/D_{\text{in}}$)\\
    $\tanh$ / sigmoid layer & Xavier/Glorot init ($\sigma^2=2/(D_{\text{in}}{+}D_{\text{out}})$)\\
    LLM / Transformer backbone & Pre-LN with RMSNorm\\
    CNN with small batch $N\le 4$ & GroupNorm ($G=32$), not BatchNorm\\
    overconfident logits & Label smoothing $\varepsilon\approx 0.1$ + Weight Decay\\
  \end{tabular}
\end{cscard}

\begin{cscard}{csE}{VARIANCE \& SCALING CHEAT-LINE}
  \begin{csbul}
    \item \textbf{ReLU second moment:} $\E[\ReLU(f)^2]=\frac{1}{2}\Var(f)$ for symmetric zero-mean $p(f)$
    \item \textbf{Norm-free ResNet scaling:} multiply $\mathcal{F}_l$ by $1/\sqrt{L}$ (or $1/\sqrt{2L}$) to stop $2^L$ blowup
    \item \textbf{Inverted dropout:} scale by $1/p$ in \texttt{train()}, multiply by $1$ in \texttt{eval()}
  \end{csbul}
\end{cscard}

\begin{cscard}{csG}{PITFALL CHECKLIST}
  \begin{csbul}
    \item \textbf{Missing $\sqrt{2}$ in ReLU init:} shrinks variance by $2^{-30}\approx 10^{-9}$ at $L=30$
    \item \textbf{Post-LN without warmup:} large output gradients destabilize early Adam steps
    \item \textbf{Forgetting \texttt{model.eval()}:} BatchNorm \& Dropout corrupt test predictions
    \item \textbf{Dropout $\to$ BatchNorm disharmony:} train variance $(1/p)$ mismatches eval variance
    \item \textbf{Penalizing biases/norm gains:} apply weight decay only to weight matrices $\mat W^{(l)}$, never to $\vecb b,\vecb\gamma,\vecb\beta$
  \end{csbul}
\end{cscard}

\begin{cscard}{csD}{10-SECOND VISUAL RECAP}
  \sffamily\footnotesize
  Deep networks train reliably when four mechanisms lock together: \textbf{He initialization} ($\sigma^2=2/D_{\text{in}}$) preserves unit variance at step $0$; \textbf{Normalization} (BN/LN/RMSNorm) conditions feature scales; \textbf{Pre-LN Residual connections} preserve an unmodulated $\eye$ gradient highway across $2^L$ paths; and \textbf{Weight Decay + Dropout} damp low-curvature noise directions.
\end{cscard}
