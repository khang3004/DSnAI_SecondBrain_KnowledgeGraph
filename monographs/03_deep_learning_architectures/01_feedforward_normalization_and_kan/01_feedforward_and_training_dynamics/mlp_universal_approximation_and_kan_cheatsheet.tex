\csbranch{1}{A}{1\;\; WHAT $-$ Node vs.\ Edge Duality}{%
  \begin{minipage}[t]{0.59\linewidth}
    \begin{csbul}
      \item \textbf{MLP:} linear weights on edges, fixed $\sigma(\cdot)$ on nodes:
        $\vecb h^{(\ell)}=\sigma(\mat W^{(\ell)}\vecb h^{(\ell-1)}+\vecb b^{(\ell)})$
      \item \textbf{KAN:} learnable 1D splines $\phi_{\ell,q,p}$ on edges, pure $\sum$ on nodes:
        $h_q^{(\ell)}=\sum_{p=1}^{n_{\ell-1}}\phi_{\ell,q,p}(h_p^{(\ell-1)})$
      \item \textbf{KAN Edge:} $\phi(x)=w_b\,\mathrm{SiLU}(x)+w_s\sum_m c_m B_{m,k}(x)$
    \end{csbul}

\csbranch{3}{C}{3\;\; KEY THEOREMS $-$ UAT, Regions \& K--A}{%
  \begin{minipage}[t]{0.61\linewidth}

      \item \textbf{K--A Theorem:} $f(\vecb x)=\sum_{q=0}^{2n}\Phi_q(\sum_{p=1}^n\phi_{q,p}(x_p))$

\csbranch{4}{D}{4\;\; CONNECTIONS $-$ Space Folding vs.\ Grid Extension}{%
  \begin{csbul}
    \item \textbf{Fixed Basis vs.\ Adaptive:} Polynomial basis needs $O(D_i^M)$ coeffs; MLPs adapt basis directions $\vecb w_j$
    \item \textbf{Input-Space Folding:} Each ReLU layer folds $\R^{D_i}$ along hyperplanes $\vecb w_j\T\vecb h+b_j=0$, multiplying linear regions across depth ($11^5=161{,}051$ regions with $471$ params for $K=5,D=10$)
    \item \textbf{Coarse-to-Fine KAN Grid Extension:} Refine $G_1\to G_2$ without retraining from scratch via local least-squares B-spline projection $\min_{\vecb c'}\sum_s(\sum_j c'_j B_{j,k}^{\text{fine}}(x_s)-\sum_m c_m B_{m,k}^{\text{coarse}}(x_s))^2$
  \end{csbul}}

\begin{cscard}{csA}{FORMULAS TO REMEMBER}
  \begin{csbul}
    \item MLP Layer \& Readout:\csfm{\vecb h^{(\ell)}=\sigma\!\left(\mat W^{(\ell)}\vecb h^{(\ell-1)}+\vecb b^{(\ell)}\right)}
    \item KAN Layer \& Edge Parametrization:\csfm{h_q^{(\ell)}=\sum_{p=1}^{n_{\ell-1}}\!\left[w_b\,\mathrm{SiLU}(h_p^{(\ell-1)})+w_s\sum_{m=0}^{G+k-1}c_m B_{m,k}(h_p^{(\ell-1)})\right]}
    \item Cox--de Boor Recursion ($r=1,\dots,k$):\csfm{B_{m,r}(x)=\frac{x-t_m}{t_{m+r}-t_m}B_{m,r-1}(x)+\frac{t_{m+r+1}-x}{t_{m+r+1}-t_{m+1}}B_{m+1,r-1}(x)}
    \item Linear Regions (1D vs.\ Mont\'ufar $d$-D):\csfm{R_{\max}(1,K,D)=(D{+}1)^K,\quad R_{\max}(d,l,n)=\Omega\!\left(\Bigl(\frac{n}{d}\Bigr)^{(l-1)d}n^d\right)}
    \item Kolmogorov--Arnold Superposition:\csfm{f(x_1,\dots,x_n)=\sum_{q=0}^{2n}\Phi_q\!\left(\sum_{p=1}^n\phi_{q,p}(x_p)\right)}
  \end{csbul}
\end{cscard}

\begin{cscard}{csB}{RECIPE: KAN FORWARD \& GRID EXTENSION}
  \begin{enumerate}[leftmargin=1.2em, itemsep=1pt, topsep=1pt, parsep=0pt]
    \item \textbf{Base indicators:} evaluate $B_{m,0}(x)=\mathbb{I}[t_m\le x<t_{m+1}]$ for knot vector $\vecb t\in\R^{G+2k+1}$.
    \item \textbf{Upward sweep:} run Cox--de Boor for $r=1\dots k$; only $k{+}1$ bases are non-zero per input $x$.
    \item \textbf{Edge combine:} add residual $w_b\,\mathrm{SiLU}(x)$ to spline sum $w_s\sum_m c_m B_{m,k}(x)$ and sum over $p$.
    \item \textbf{Grid extension ($G\to G_{\text{new}}$):} sample collocation points $\{x_s\}\subset[t_k,t_{G+k}]$ and solve least-squares $\argmin_{\vecb c'}\sum_s(\sum_j c'_j B_{j,k}^{\text{fine}}(x_s)-\sum_m c_m B_{m,k}(x_s))^2$.
  \end{enumerate}
\end{cscard}

\begin{cscard}{csG}{PITFALL CHECKLIST}
  \begin{csbul}
    \item \textbf{Linear collapse:} stacking linear layers without nonlinear $\sigma$ collapses to a single matrix $\mat W'\vecb x+\vecb b'$
    \item \textbf{Saturating sigmoids:} $\sigma(a)=(1+e^{-a})^{-1}$ and $\tanh(a)$ kill gradients for $|a|\gg 1$
    \item \textbf{Dying ReLU:} negative bias shift zeroes out both forward activation and backward gradient forever
    \item \textbf{KAN knot overflow:} inputs outside $[t_k,t_{G+k}]$ evaluate $B_{m,k}(x)=0$, falling back solely to $w_b\,\mathrm{SiLU}(x)$
    \item \textbf{Non-GEMM bottleneck:} B-spline evaluation takes $5\times$--$10\times$ higher wall-clock time per parameter than dense matmul
  \end{csbul}
\end{cscard}

\begin{cscard}{csD}{10-SECOND VISUAL RECAP}
  \sffamily\footnotesize
  \textbf{MLPs} place learnable weights on edges and fixed nonlinear hinges on nodes; stacking layers \emph{folds} input space to create $(D{+}1)^K$ polyhedral regions.
  \textbf{KANs} dualize the graph: learnable B-splines $\phi(x)=w_b\,\mathrm{SiLU}(x)+w_s\sum_m c_m B_{m,k}(x)$ live on edges and nodes simply sum, trading dense GEMM speed for local grid extension and symbolic interpretability.
\end{cscard}
