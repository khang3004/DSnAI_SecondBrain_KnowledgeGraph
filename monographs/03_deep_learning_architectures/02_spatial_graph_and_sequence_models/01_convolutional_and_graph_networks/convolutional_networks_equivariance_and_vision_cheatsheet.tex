\csbranch{2}{B}{2\;\; EQUIVARIANCE vs.\ POOLING INVARIANCE}{%
  \begin{csbul}
    \item \textbf{Translation Equivariance ($s=1$):} $\mathcal{F}_{\mat W}(T_\Delta\mat X)=T_\Delta\mathcal{F}_{\mat W}(\mat X)$; shifting input by $\Delta$ shifts features by $\Delta$
    \item \textbf{Pooling Local Invariance:} max/avg pooling $P_k(T_\delta\mat H)\approx P_k(\mat H)$ absorbs small shifts $\delta$; Global Average Pooling (GAP) gives global shift invariance
    \item \textbf{Bayesian View:} CNN = MLP with an \emph{infinitely strong prior} (zero weights outside $k\times k$, tied weights across space)
  \end{csbul}}

\csbranch{3}{C}{3\;\; RECEPTIVE FIELD \& DEPTHWISE-SEPARABLE}{%
  \begin{minipage}[t]{0.61\linewidth}
    \csfm{R_l = R_{l-1} + d_l(k_l-1)\prod_{i=1}^{l-1}s_i}
    \begin{csbul}
      \item Base $R_0=1$; $m$ stride-1 $3\times 3$: $R_m=2m+1$ ($18C^2$ vs.\ $25C^2$ for $5\times 5$)
      \item Exponential dilation $d_l=2^{l-1}\Rightarrow R_L=2^{L+1}-1$
      \item \textbf{Depthwise-Sep.\ ratio:} $\frac{1}{C_{\mathrm{out}}}+\frac{1}{k^2}$ ($\approx 8.7\times$ cheaper for $k=3,C_{\mathrm{out}}=256$)

\csbranch{4}{D}{4\;\; VISION BACKBONES $-$ VGG $\to$ ResNet $\to$ U-Net}{%
  \begin{csbul}
    \item \textbf{AlexNet $\to$ VGG:} replaced $11\times 11, 5\times 5$ kernels with homogeneous stacks of $3\times 3$ convs + $2\times 2$ pooling
    \item \textbf{GoogLeNet / ResNet Bottleneck:} $1\times 1$ compress ($C\to C/4$) $\to$ $3\times 3$ spatial $\to$ $1\times 1$ expand ($C/4\to C$) + identity skip $\mat H_{l+1}=\mat H_l+\mathcal{F}(\mat H_l)$
    \item \textbf{FCN $\to$ U-Net:} symmetric encoder--decoder with multi-scale skip concatenations to preserve sharp segmentation boundaries
  \end{csbul}}

\begin{cscard}{csA}{FORMULAS TO REMEMBER}
  \begin{csbul}
    \item Spatial Output Shape:\csfm{H_{\mathrm{out}}=\left\lfloor\frac{H+2p_h-d_h(k_h-1)-1}{s_h}\right\rfloor+1}
    \item 2D Cross-Correlation:\csfm{\mat Z[d,i,j]=b_d+\sum_{c,u,v}\mat X[c,s_hi{+}d_hu{-}p_h,\,s_wj{+}d_wv{-}p_w]\,\mat W[d,c,u,v]}
    \item Translation Equivariance ($s=1$):\csfm{\mathcal{F}_{\mat W}(T_\Delta\mat X)=T_\Delta\mathcal{F}_{\mat W}(\mat X)}
    \item Receptive Field Recurrence ($J_{l-1}=\prod_{i<l}s_i$):\csfm{R_l=R_{l-1}+d_l(k_l-1)J_{l-1}=1+\sum_{l'=1}^l d_{l'}(k_{l'}-1)\prod_{i=1}^{l'-1}s_i}
    \item Depthwise-Separable Cost Ratio:\csfm{\frac{\mathrm{Params}_{\mathrm{DS}}}{\mathrm{Params}_{\mathrm{Std}}}=\frac{\mathrm{MACs}_{\mathrm{DS}}}{\mathrm{MACs}_{\mathrm{Std}}}=\frac{k^2C_{\mathrm{in}}+C_{\mathrm{in}}C_{\mathrm{out}}}{k^2C_{\mathrm{in}}C_{\mathrm{out}}}=\frac{1}{C_{\mathrm{out}}}+\frac{1}{k^2}}
  \end{csbul}
\end{cscard}

\begin{cscard}{csB}{RECIPE: IM2COL GEMM vs.\ DEPTHWISE-SEPARABLE}
  \begin{enumerate}[leftmargin=1.2em, itemsep=1pt, topsep=1pt, parsep=0pt]
    \item \textbf{Im2Col (Standard):} extract each $C_{\mathrm{in}}\times k\times k$ patch into column matrix $\mat M_{\mathrm{col}}\in\R^{(C_{\mathrm{in}}k^2)\times(H_{\mathrm{out}}W_{\mathrm{out}})}$; multiply by reshaped weights $\mat W_{\mathrm{row}}\in\R^{C_{\mathrm{out}}\times(C_{\mathrm{in}}k^2)}$.
    \item \textbf{Depthwise step:} apply one spatial $k\times k$ filter per input channel ($k^2 C_{\mathrm{in}}$ params).
    \item \textbf{Pointwise step:} apply $1\times 1$ conv across channels ($C_{\mathrm{in}}C_{\mathrm{out}}$ params) via dense GEMM.
  \end{enumerate}
\end{cscard}

\begin{cscard}{csE}{SCALING \& BACKBONE CHEAT-LINE}
  \begin{csbul}
    \item \textbf{Two $3\times 3$ vs.\ one $5\times 5$:} same $5\times 5$ RF, $18C^2$ vs.\ $25C^2$ params ($-28\%$) + extra ReLU
    \item \textbf{ResNet Bottleneck:} $1\times 1$ ($C\!\to\! C/4$) $\to 3\times 3 \to 1\times 1$ ($C/4\!\to\! C$) cuts $3\times 3$ cost by $16\times$
    \item \textbf{ERF vs.\ theoretical RF:} effective sensitivity scales as $O(\sqrt{L})$ Gaussian around center
  \end{csbul}
\end{cscard}

\begin{cscard}{csG}{PITFALL CHECKLIST}
  \begin{csbul}
    \item \textbf{Strided aliasing:} $s\ge 2$ violates Nyquist--Shannon when high frequencies are present, breaking shift invariance
    \item \textbf{Transposed conv checkerboards:} uneven kernel overlap when $k\bmod s\neq 0$ (e.g., $k=3, s=2$)
    \item \textbf{Dilated gridding artifact:} stacking identical dilation rates leaves blind checkerboard holes in the input
    \item \textbf{Coord-sensitive tasks:} pure CNN + GAP discards absolute $(x,y)$ coordinates (concat CoordConv channels if position matters)
    \item \textbf{Im2Col memory blow-up:} materializing $\mat M_{\mathrm{col}}$ duplicates input pixels $k^2$ times in RAM
  \end{csbul}
\end{cscard}

\begin{cscard}{csD}{10-SECOND VISUAL RECAP}
  \sffamily\footnotesize
  A \textbf{CNN layer} is a doubly block-Toeplitz linear map that shares a $k\times k$ kernel across space so translations commute: $\mathcal{F}(T_\Delta\mat X)=T_\Delta\mathcal{F}(\mat X)$.
  Stacking $3\times 3$ filters with striding or dilation grows the receptive field $R_l$, while \textbf{depthwise-separable} ($3\times 3\text{ DW}+1\times 1\text{ PW}$) and \textbf{ResNet/U-Net skips} scale vision backbones efficiently.
\end{cscard}
