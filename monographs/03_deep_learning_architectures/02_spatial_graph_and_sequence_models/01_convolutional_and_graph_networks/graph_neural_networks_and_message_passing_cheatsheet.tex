\csbranch{4}{D}{4\;\; CONNECTIONS $-$ Fourier, Chebyshev \& attention}{%
  \begin{csbul}
    \item \textbf{Graph Laplacian:} $\mat L=\mat D-\mat A$, $\mat L_{\mathrm{sym}}=\eye_N-\mat D^{-1/2}\mat A\mat D^{-1/2}=\mat U\mat\Lambda\mat U\T$
    \item \textbf{Dirichlet energy:} $\vecb x\T\mat L\vecb x = \frac{1}{2}\sum_{i,j}A_{ij}(x_i-x_j)^2 \ge 0$ (smoothness)
    \item \textbf{ChebNet ($K=1$):} $\sum_{k=0}^K \theta_k T_k(\tilde{\mat L})\vecb x \;\Rightarrow\;$ 1-hop spatial GCN
    \item \textbf{Transformer = GAT} on a complete graph ($\mathcal N(i)=\mathcal V$) + positional encodings
    \item \textbf{CNN = GNN} on a regular 2D grid with directional edge types
  \end{csbul}}

\begin{cscard}{csA}{FORMULAS TO REMEMBER}
  \begin{csbul}
    \item Symmetric Normalized Laplacian:\csfm{\mat L_{\mathrm{sym}}=\eye_N-\mat D^{-1/2}\mat A\mat D^{-1/2},\quad \lambda_i\in[0,2]}
    \item Kipf--Welling GCN ($\tilde{\mat A}=\mat A+\eye_N$):\csfm{\mat H^{(\ell+1)}=\sigma\!\bigl(\tilde{\mat D}^{-1/2}\tilde{\mat A}\tilde{\mat D}^{-1/2}\mat H^{(\ell)}\mat W^{(\ell)}\bigr)}
    \item GAT Anisotropic Attention ($j\in\mathcal N(i)\cup\{i\}$):\csfm{\alpha_{ij}^{(\ell)}=\softmax_j\!\Bigl(\mathrm{LeakyReLU}\bigl(\vecb a\T[\mat W\vecb h_i^{(\ell)}\|\mat W\vecb h_j^{(\ell)}]\bigr)\Bigr)}
    \item GraphSAGE Inductive Update:\csfm{\vecb h_i^{(\ell+1)}=\sigma\!\Bigl(\mat W_1^{(\ell)}\vecb h_i^{(\ell)}+\mat W_2^{(\ell)}\mathrm{AGG}_{j\in\mathcal N_S(i)}\bigl(\vecb h_j^{(\ell)}\bigr)\Bigr)}
    \item GIN (1-WL Optimal Injectivity):\csfm{\vecb h_i^{(\ell+1)}=\mathrm{MLP}^{(\ell)}\!\Bigl((1+\epsilon^{(\ell)})\vecb h_i^{(\ell)}+\sum_{j\in\mathcal N(i)}\vecb h_j^{(\ell)}\Bigr)}
  \end{csbul}
\end{cscard}

\begin{cscard}{csB}{MINI-BATCH GRAPHSAGE RECIPE}
  \begin{enumerate}[leftmargin=1.2em, itemsep=1pt, topsep=1pt, parsep=0pt]
    \item Set target batch $\mathcal B^{(L)} \leftarrow \mathcal B \subseteq \mathcal V$
    \item \textbf{Top-down sample} ($\ell=L\dots 1$): for $i\in\mathcal B^{(\ell)}$, draw $q_\ell$ neighbors $\mathcal N_S^{(\ell)}(i)$ without replacement; $\mathcal B^{(\ell-1)} \leftarrow \mathcal B^{(\ell)}\cup\mathcal N_S^{(\ell)}(i)$
    \item Initialize $\vecb h_i^{(0)}\leftarrow\vecb x_i$ for all $i\in\mathcal B^{(0)}$
    \item \textbf{Bottom-up pass} ($\ell=0\dots L-1$): aggregate $\vecb m_i^{(\ell+1)}$ over $\mathcal N_S^{(\ell+1)}(i)$, apply $\sigma(\mat W_1\vecb h_i+\mat W_2\vecb m_i)$, and $\ell_2$-normalize $\vecb h_i^{(\ell+1)}$
  \end{enumerate}
\end{cscard}

\begin{cscard}{csE}{EXPRESSIVITY \& SPECTRAL BOUNDS}
  \begin{csbul}
    \item \textbf{Aggregator rank:} $\mathrm{Sum}$ (all multisets) $\succ \mathrm{Mean}$ (proportions) $\succ \mathrm{Max}$ (set support)
    \item \textbf{Dirichlet decay:} $E(\mat H^{(\ell)})\le (1-\lambda_2)^2 s_{\max}^2 E(\mat H^{(\ell-1)})$
    \item \textbf{Sampled footprint:} $O(B\prod_{\ell=1}^L q_\ell)$ vs full-batch $O(N)$
  \end{csbul}
\end{cscard}

\begin{cscard}{csG}{PITFALL CHECKLIST}
  \begin{csbul}
    \item \textbf{Missing self-loops ($\mat A$ vs $\tilde{\mat A}$):} omitting $+\eye_N$ discards node $i$'s own features in 1-hop GCNs
    \item \textbf{Unnormalized sum explosion:} $\tilde{\mat A}\mat H\mat W$ scales linearly with degree $d_i$; use $\tilde{\mat D}^{-1/2}\tilde{\mat A}\tilde{\mat D}^{-1/2}$ or MLP+BN (GIN)
    \item \textbf{Stacking $>4$ vanilla GCN layers:} features collapse to $\vecb u_1 \propto \sqrt{\tilde d_i}$ ($\ker(\tilde{\mat L}_{\mathrm{sym}})$); always add residual connections $\vecb h^{(\ell+1)} = f(\cdot)+\vecb h^{(\ell)}$
    \item \textbf{Tree/bridge bottlenecks (Over-squashing):} $\norm{\partial\vecb h_i^{(L)}/\partial\vecb x_j}$ decays exponentially with distance; add a virtual super-node or graph rewiring
    \item \textbf{Data leakage in transductive splits:} test nodes $\mathcal V_{\mathrm{trans}}$ may pass messages during training, but their \emph{labels} must never enter the loss
  \end{csbul}
\end{cscard}

\begin{cscard}{csD}{10-SECOND VISUAL RECAP}
  \sffamily\footnotesize
  \textbf{A GNN layer is local message passing}: each node collects neighbor vectors, aggregates them with a \textbf{permutation-invariant} $\bigoplus$, and updates its state with shared weights ($\Rightarrow$ \textbf{permutation-equivariant} node embeddings). \textbf{GCN} uses fixed degree-normalized smoothing, \textbf{GAT} learns attention weights, \textbf{GraphSAGE} samples bounded fanouts for scale, and \textbf{GIN} uses injective sum+MLP to match the \textbf{1-WL isomorphism ceiling}.
\end{cscard}
