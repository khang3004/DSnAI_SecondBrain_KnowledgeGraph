\csbranch{2}{B}{2\;\; KALMAN \& RTS $-$ exact Bayesian state inference}{%
  \begin{csbul}
    \item \textbf{Predict:} $\vecb\mu_{t|t-1}=\mat F_t\vecb\mu_{t-1|t-1}+\mat B_t\vecb u_t$, \; $\mat\Sigma_{t|t-1}=\mat F_t\mat\Sigma_{t-1|t-1}\mat F_t\T+\mat Q_t$
    \item \textbf{Kalman Gain (SNR gate):} $\mat S_t=\mat H_t\mat\Sigma_{t|t-1}\mat H_t\T+\mat R_t$, \; $\mat K_t=\mat\Sigma_{t|t-1}\mat H_t\T\mat S_t\inv$
    \item \textbf{Update:} $\vecb\mu_{t|t}=\vecb\mu_{t|t-1}+\mat K_t(\vecb y_t-\hat{\vecb y}_t)$, \; $\mat\Sigma_{t|t}=\mat\Sigma_{t|t-1}-\mat K_t\mat S_t\mat K_t\T$
    \item \textbf{RTS Smoother ($t=T{-}1\dots 1$):} $\mat J_t=\mat\Sigma_{t|t}\mat F_t\T\mat\Sigma_{t+1|t}\inv$, \; $\vecb\mu_{t|T}=\vecb\mu_{t|t}+\mat J_t(\vecb\mu_{t+1|T}-\vecb\mu_{t+1|t})$
    \item High sensor noise $\mat R_t\gg\mat\Sigma \Rightarrow \mat K_t\to\zeros$ (trust dynamics); $\mat R_t\to\zeros \Rightarrow$ snap to $\vecb y_t$
  \end{csbul}}

\csbranch{3}{C}{3\;\; KEY ADVANCES $-$ ZOH, HiPPO/S4 \& Selective $S6$}{%
  \csfm{\bar{\mat A}_t = \exp(\Delta_t\mat A),\qquad \bar{\mat B}_t = (\Delta_t\mat A)\inv\bigl(\exp(\Delta_t\mat A)-\mat I\bigr)\Delta_t\mat B_t}
  \begin{csbul}
    \item \textbf{HiPPO-LegS:} $A_{nk}=-(2n{+}1)^{1/2}(2k{+}1)^{1/2}$ ($n>k$) projects history onto Legendre polynomials; \textbf{S4} uses NPLR + Woodbury/Cauchy
    \item \textbf{LTI Failure:} fixed $(\bar{\mat A},\bar{\mat B},\mat C)$ cannot do content-based selection / induction heads
    \item \textbf{Mamba ($S6$):} input-dependent $\mat B_t=\mathrm{Lin}_N(x_t)$, $\mat C_t=\mathrm{Lin}_N(x_t)$, $\Delta_t=\softplus(\mathrm{Lin}_1(x_t))$
  \end{csbul}}

\csbranch{4}{D}{4\;\; CONNECTIONS $-$ Parallel Scan, SRAM \& Gating}{%
  \begin{minipage}[t]{0.59\linewidth}
    \csfm{(\mat M_2,\vecb v_2)\bullet(\mat M_1,\vecb v_1)=(\mat M_2\mat M_1,\;\mat M_2\vecb v_1+\vecb v_2)}
    \begin{csbul}
      \item \textbf{Associative Scan:} solves time-varying $\vecb h_t=\bar{\mat A}_t\vecb h_{t-1}+\vecb b_t$ (and Kalman!) in $O(\log L)$ span
      \item \textbf{$\Delta_t \leftrightarrow$ RNN Gate:} $\Delta_t\to\infty$ resets state to $x_t$; $\Delta_t\to 0$ persists $\vecb h_{t-1}$
      \item \textbf{SRAM Fusion:} keep $(B,L,D,N)$ states in fast SRAM, never HBM
    \end{csbul}
  \end{minipage}\hfill

\begin{cscard}{csA}{FORMULAS TO REMEMBER}
  \begin{csbul}
    \item Kalman Predict \& Gain:\csfm{\mat\Sigma_{t|t-1}=\mat F_t\mat\Sigma_{t-1|t-1}\mat F_t\T+\mat Q_t,\quad \mat K_t=\mat\Sigma_{t|t-1}\mat H_t\T(\mat H_t\mat\Sigma_{t|t-1}\mat H_t\T+\mat R_t)\inv}
    \item Joseph-Form Covariance Update:\csfm{\mat\Sigma_{t|t}=(\mat I-\mat K_t\mat H_t)\mat\Sigma_{t|t-1}(\mat I-\mat K_t\mat H_t)\T+\mat K_t\mat R_t\mat K_t\T}
    \item RTS Backward Smoother:\csfm{\mat J_t=\mat\Sigma_{t|t}\mat F_t\T\mat\Sigma_{t+1|t}\inv,\quad \vecb\mu_{t|T}=\vecb\mu_{t|t}+\mat J_t(\vecb\mu_{t+1|T}-\vecb\mu_{t+1|t})}
    \item ZOH Discretization:\csfm{\bar{\mat A}_t=\exp(\Delta_t\mat A),\quad \bar{\mat B}_t=(\Delta_t\mat A)\inv(e^{\Delta_t\mat A}-\mat I)\Delta_t\mat B_t\approx \Delta_t\mat B_t}
    \item Associative Parallel Scan Operator:\csfm{(\mat M_2,\vecb v_2)\bullet(\mat M_1,\vecb v_1)=(\mat M_2\mat M_1,\;\mat M_2\vecb v_1+\vecb v_2)}
  \end{csbul}
\end{cscard}

\begin{cscard}{csB}{MAMBA ($S6$) HARDWARE-AWARE SCAN RECIPE}
  \begin{enumerate}[leftmargin=1.2em, itemsep=1pt, topsep=1pt, parsep=0pt]
    \item Project input $\mat X\in\R^{B\times L\times D}$ to $\mat B=\mathrm{Lin}_N(\mat X)$, $\mat C=\mathrm{Lin}_N(\mat X)$, and $\mat\Delta=\softplus(\vecb b_\Delta + \mathrm{Lin}_1(\mat X))$ in HBM
    \item \textbf{Load tiles to SRAM:} read $(\mat A, \mat\Delta, \mat B, \mat C, \mat X)$ into on-chip SRAM
    \item \textbf{Discretize \& Scan in SRAM:} form $\bar{\mat A}=\exp(\mat\Delta\mat A)$, $\bar{\mat B}=\mat\Delta\mat B$; run parallel associative scan $\bullet$ over $L$ in $O(\log L)$ depth
    \item \textbf{Project \& Write:} compute $y_t = \mat C_t\vecb h_t$ in SRAM; write only $\mat Y\in\R^{B\times L\times D}$ to HBM (recompute $\vecb h$ on backward pass)
  \end{enumerate}
\end{cscard}

\begin{cscard}{csG}{PITFALL CHECKLIST}
  \begin{csbul}
    \item \textbf{Naive $\mat\Sigma_{t|t}$ subtraction:} $(\mat I-\mat K_t\mat H_t)\mat\Sigma_{t|t-1}$ loses symmetry/PD in float32; always use the symmetric Joseph update or Cholesky/QR factors
    \item \textbf{Explicit matrix inverse $\mat S_t\inv$:} never invert $\mat S_t$ or $\mat\Sigma_{t+1|t}$; solve linear systems $\mat S_t\T\mat K_t\T = \mat H_t\mat\Sigma_{t|t-1}$
    \item \textbf{Positive eigenvalues in $\mat A$:} if $\mathrm{Re}(\lambda_i(\mat A))>0$, discrete $|\bar\lambda_i|>1$ explodes; parameterize $\mat A = -\exp(\mat A_{\log})$ and $\Delta_t = \softplus(\cdot) > 0$
    \item \textbf{Using FFT convolution on Mamba:} input-dependent $(\Delta_t,\mat B_t,\mat C_t)$ break time-invariance, so $\vecb y\ne\bar{\mat K}*\vecb x$; must use the associative parallel scan $\bullet$
    \item \textbf{Saving 4D state $\vecb h\in\R^{B\times L\times D\times N}$ in HBM:} blows up GPU memory $N\times$; recompute in SRAM during backprop
  \end{csbul}
\end{cscard}

\begin{cscard}{csD}{10-SECOND VISUAL RECAP}
  \sffamily\footnotesize
  \textbf{All state-space models compress the past into a Markovian state.} Classical \textbf{Kalman/RTS} propagates Gaussian beliefs $(\vecb\mu_t,\mat\Sigma_t)$ gated by inverse SNR ($\mat K_t$). Deep \textbf{LTI SSMs (S4)} discretize HiPPO ODEs into fast $O(L\log L)$ FFT convolutions but cannot select content. \textbf{Mamba ($S6$)} makes $(\Delta_t,\mat B_t,\mat C_t)$ functions of token $x_t$ and fuses an $O(\log L)$ \textbf{associative scan in GPU SRAM}, achieving linear $O(L)$ training and constant $O(1)$ per-token inference.
\end{cscard}
