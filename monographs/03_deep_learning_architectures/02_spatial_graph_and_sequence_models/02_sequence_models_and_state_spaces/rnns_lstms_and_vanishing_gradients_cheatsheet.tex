\csbranch{2}{B}{2\;\; SPECTRAL RADIUS $-$ vanish vs explode}{%
  \begin{minipage}[t]{0.60\linewidth}
    \csfm{\norm[\Big]{\pd{\vecb h_t}{\vecb h_k}}_2 \le \bigl(\gamma\,\sigma_{\max}(\mat W_{hh})\bigr)^{t-k}}
    \begin{csbul}
      \item $\gamma=\sup_z|\phi'(z)|$: $\gamma=1$ ($\tanh$), $\gamma=\tfrac{1}{4}$ ($\sigma$)
      \item $|\lambda|<1 \Rightarrow |\lambda|^{t-k}\to 0$ (\textbf{vanishing}: forgets past)
      \item $|\lambda|>1 \Rightarrow |\lambda|^{t-k}\to \infty$ (\textbf{exploding}: loss cliffs)
      \item Vanilla $\vecb h_t$ cannot both \emph{compute locally} \& \emph{store globally}
    \end{csbul}
  \end{minipage}\hfill
  \begin{minipage}[t]{0.37\linewidth}\centering\vspace{2pt}

\csbranch{3}{C}{3\;\; KEY MECHANISM $-$ LSTM \& CEC highway}{%
  \csfm{\vecb c_t = \vecb f_t \odot \vecb c_{t-1} + \vecb i_t \odot \tilde{\vecb c}_t,\qquad \vecb h_t = \vecb o_t \odot \tanh(\vecb c_t)}
  \begin{csbul}
    \item \textbf{3 sigmoid gates:} $\vecb f_t$ (forget), $\vecb i_t$ (input/add), $\vecb o_t$ (output) $\in (0,1)^{d_h}$
    \item \textbf{Candidate memory:} $\tilde{\vecb c}_t = \tanh(\mat W_{cx}\vecb x_t + \mat W_{ch}\vecb h_{t-1} + \vecb b_c) \in (-1,1)^{d_h}$
    \item \textbf{Constant Error Carousel (CEC):} $\left.\pd{\vecb c_t}{\vecb c_k}\right|_{\mathrm{direct}} = \diag\bigl(\prod_{j=k+1}^t \vecb f_j\bigr)$
    \item When $\vecb f_j \approx \ones$, gradients flow across $t-k$ steps with \textbf{no $\mat W_{hh}$ multiplication}!
  \end{csbul}}

\csbranch{4}{D}{4\;\; CONNECTIONS $-$ GRU, ESNs \& leaky units}{%
  \begin{csbul}
    \item \textbf{GRU (2 gates, 1 state):} $\vecb h_t = \vecb z_t\odot\vecb h_{t-1} + (\ones-\vecb z_t)\odot\tanh(\mat W_{hx}\vecb x_t + \mat W_{hh}(\vecb r_t\odot\vecb h_{t-1}))$
    \item \textbf{ResNet = unrolled LSTM:} $\vecb c_t = \vecb c_{t-1} + \text{gate}\odot\tilde{\vecb c}_t$ is a temporal residual connection
    \item \textbf{Echo State Network (ESN):} fix random $\mat W_{hh}$ at $\rho(\mat W_{hh})\approx 1$; train only $\mat W_{oh}$
    \item \textbf{Leaky units / skip edges:} $\mu_t=\alpha\mu_{t-1}+(1-\alpha)v_t$ or $d$-step jumps ($T/d$ path)
    \item \textbf{Bidirectional RNN:} forward $\overrightarrow{\vecb h}_t \oplus$ backward $\overleftarrow{\vecb h}_t$ for non-causal tagging
  \end{csbul}}

\csbranch{5}{E}{5\;\; COST $-$ SRN vs LSTM vs GRU vs TBPTT}{%

    \textbf{\textcolor{csE}{Cell}} & \textbf{\textcolor{csE}{Params}} & \textbf{\textcolor{csE}{BPTT Mem}} & \textbf{\textcolor{csE}{Highway}}\\ \hline
    Elman SRN & $d_h(d_x{+}d_h{+}1)$ & $O(Kd_h)$ & None ($\prod\jac_j$)\\

\csnote{8}{A}{TRAPS $-$ before you trust the output}{%
  \begin{csbul}
    \item \textbf{Zero forget bias:} $f_t{\approx}0.5\Rightarrow 0.5^{20}$ decay; init $\vecb b_f\in[1,2]$
    \item \textbf{Elementwise clip:} clip global $\norm{\vecb g}_2>\nu$ to keep direction
    \item \textbf{Exposure bias:} train $\vecb y_{t-1}$ vs test $\hat{\vecb y}_{t-1}$ (scheduled sampling)
    \item \textbf{State leakage:} reset $\vecb h_0=\zeros, \vecb c_0=\zeros$ at TBPTT boundaries
  \end{csbul}}

\begin{cscard}{csA}{FORMULAS TO REMEMBER}
  \begin{csbul}
    \item BPTT Recursion ($\vecb\delta_t^o = \hat{\vecb y}_t - \vecb y_t$):\csfm{\nabla_{\vecb h_t}\mathcal L = \mat W_{hh}\T\diag\bigl(\phi'(\vecb a_{t+1})\bigr)\nabla_{\vecb h_{t+1}}\mathcal L + \mat W_{oh}\T\vecb\delta_t^o}
    \item Parameter Gradients:\csfm{\nabla_{\mat W_{hh}}\mathcal L = \sum_{t=1}^T \diag\bigl(\phi'(\vecb a_t)\bigr)(\nabla_{\vecb h_t}\mathcal L)\vecb h_{t-1}\T}
    \item LSTM Gates \& Cell Update:\csfm{[\vecb f_t,\vecb i_t,\vecb o_t] = \sigma(\mat W_x\vecb x_t + \mat W_h\vecb h_{t-1} + \vecb b),\quad \vecb c_t = \vecb f_t\odot\vecb c_{t-1} + \vecb i_t\odot\tilde{\vecb c}_t}
    \item GRU Interpolation Update:\csfm{\vecb h_t = \vecb z_t\odot\vecb h_{t-1} + (\ones-\vecb z_t)\odot\tanh\!\bigl(\mat W_{hx}\vecb x_t + \mat W_{hh}(\vecb r_t\odot\vecb h_{t-1})+\vecb b_h\bigr)}
    \item Global $L_2$-Norm Clipping:\csfm{\vecb g \leftarrow \min\!\Bigl(1,\, \frac{\nu}{\norm{\vecb g}_2}\Bigr)\vecb g}
  \end{csbul}
\end{cscard}

\begin{cscard}{csB}{TBPTT($K$) + GRADIENT CLIPPING RECIPE}
  \begin{enumerate}[leftmargin=1.2em, itemsep=1pt, topsep=1pt, parsep=0pt]
    \item Init $\vecb h_0\leftarrow\zeros, \vecb c_0\leftarrow\zeros$ at sequence start ($b_f\leftarrow +1$)
    \item For chunk $s=1, K{+}1, 2K{+}1,\dots$ to $e=\min(s{+}K{-}1,T)$:
    \item \textbf{Detach} $\vecb h_{s-1}, \vecb c_{s-1}$ from prior autograd graph (carry values forward, stop gradients beyond $K$)
    \item Unroll forward $t=s\dots e$; compute $\mathcal L_{\mathrm{chunk}}$ and backprop $\vecb g = \nabla_\theta\mathcal L_{\mathrm{chunk}}$ from $e$ down to $s$
    \item If $\norm{\vecb g}_2 > \nu$, rescale $\vecb g \leftarrow (\nu/\norm{\vecb g}_2)\vecb g$; step $\theta \leftarrow \theta - \eta\vecb g$
  \end{enumerate}
\end{cscard}

  \end{tabular}
\end{cscard}

\begin{cscard}{csE}{STABILITY \& SCALING CHEAT-LINE}
  \begin{csbul}
    \item \textbf{Sufficient vanishing:} $\sigma_{\max}(\mat W_{hh}) < 1/\gamma$ ($\gamma{=}1$ for $\tanh$)
    \item \textbf{Necessary exploding:} $\rho(\jac_t) > 1$ along non-saturated modes
    \item \textbf{CEC gain:} $\prod_{j=k+1}^t f_j \approx 1$ when $f_j\to 1$ (additive highway)
  \end{csbul}
\end{cscard}

\begin{cscard}{csG}{PITFALL CHECKLIST}
  \begin{csbul}
    \item \textbf{Forget bias $\vecb b_f = \zeros$:} gives $f_t = \sigma(0) = 0.5$, killing cell gradients in $\sim 15$ steps; always set $\vecb b_f \in [1, 2]$
    \item \textbf{Sigmoid recurrent activation:} $\sup|\sigma'(z)| = 0.25$ accelerates vanishing $4\times$ per step; use $\tanh$ for candidates
    \item \textbf{Un-detached hidden carryover:} forgetting `.detach()` at TBPTT chunk boundaries backpropagates to $t=1$ and causes OOM
    \item \textbf{Teacher forcing exposure bias:} model never sees its own mistakes during training; anneal with scheduled sampling or train with sequence-level RL/beam search
    \item \textbf{Orthogonal init forgotten on $\mat W_{hh}$:} Gaussian $\mat W_{hh}$ has spread singular values; initialize $\mat W_{hh}$ orthogonal so $\sigma_i(\mat W_{hh})=1$
  \end{csbul}
\end{cscard}

\begin{cscard}{csD}{10-SECOND VISUAL RECAP}
  \sffamily\footnotesize
  \textbf{Vanilla RNNs multiply by $\mat W_{hh}$ every step}, so backpropagated gradients scale as $\jac^{t-k}$---exploding when $|\lambda|>1$ (fix: \textbf{norm clipping}) and vanishing when $|\lambda|<1$ (fix: \textbf{additive gating}). \textbf{LSTMs} and \textbf{GRUs} replace multiplicative state transitions with a gated linear self-loop ($\vecb c_t = \vecb f_t\odot\vecb c_{t-1} + \dots$), creating a \textbf{Constant Error Carousel} where gradients travel unimpeded whenever $\vecb f_t\approx \ones$.
\end{cscard}
