\begin{cscard}{csA}{FORMULAS TO REMEMBER}
  \begin{csbul}
    \item Okapi BM25 lexical score ($k\in[1.2,2.0]$, $b\approx 0.75$):
      \csfm{\mathrm{BM25}(q,d)=\sum_{t\in q}\log\!\left(\frac{N}{\mathrm{df}_t}\right)\frac{\mathrm{tf}_{t,d}(k+1)}{\mathrm{tf}_{t,d}+k\bigl(1-b+b\frac{\abs{d}}{\abs{d_{\mathrm{avg}}}}\bigr)}}
    \item Bi-Encoder vs ColBERT late-interaction MaxSim:
      \csfm{s_{\mathrm{bi}}(q,d)=\inner{\vecb z_q}{\vecb z_d},\quad s_{\mathrm{ColBERT}}(q,d)=\sum_{i=1}^{N_q}\max_{j\le m_d}\inner{\vecb E_{q,i}}{\vecb E_{d,j}}}
    \item Reciprocal Rank Fusion ($\kappa=60$, calibration-free):
      \csfm{\mathrm{RRF}(d;q)=\sum_{r\in\{\mathrm{BM25},\mathrm{Dense}\}}\frac{1}{\kappa+\mathrm{rank}_r(d)}}
    \item IR evaluation metrics ($\mathrm{MRR}$ \& $\mathrm{nDCG}@k$):
      \csfm{\mathrm{MRR}=\frac{1}{\abs{Q}}\sum_{q\in Q}\frac{1}{\mathrm{rank}_1(q)},\quad \mathrm{nDCG}@k=\sum_{i=1}^{k}\frac{2^{\mathrm{rel}_i}-1}{\mathrm{IDCG}_k\log_2(i+1)}}
  \end{csbul}
\end{cscard}

\begin{cscard}{csB}{HYBRID RAG + MEMORY ASSEMBLY RECIPE}
  \begin{enumerate}[leftmargin=1.2em, itemsep=1pt, topsep=1pt, parsep=0pt]
    \item \textbf{Rewrite \& Route:} expand coreferences $q'_t\leftarrow\mathrm{Rewrite}(q_t,\mathcal{S}_{t-1})$; route to VectorDB, SQL, or Web Search.
    \item \textbf{Parallel Retrieve:} fetch top-$K$ from BM25 inverted index and HNSW graph ($E_Q(q'_t),\mathrm{efSearch}$).
    \item \textbf{RRF + Rerank:} fuse via $\mathrm{RRF}(\kappa=60)$ to top-$2k$, then rescore with cross-encoder $s_{\mathrm{cross}}(q_t,d)$ to top-$k$.
    \item \textbf{Compact \& Assemble:} if $\abs{\mathcal{S}_{t-1}}_{\mathrm{tok}}>\tau_{\mathrm{tok}}$, summarize older turns keeping overlap $o$; pack $C_t\le B_{\max}-B_{\mathrm{out}}$.
    \item \textbf{Async Consolidate:} enqueue $\mathcal{M}_t\leftarrow\mathcal{U}_{\mathrm{cons}}(\mathcal{M}_{t-1},\mathcal{E}_\theta(\mathcal{S}_t))$.
  \end{enumerate}
\end{cscard}

\begin{cscard}{csG}{PITFALL CHECKLIST}
  \begin{csbul}
    \item \textbf{Chunk boundary severance:} fixed chunking without $10\text{--}20\%$ overlap cuts sentences; large chunks dilute mean-pooled embeddings
    \item \textbf{HNSW pre/post-filter collapse:} post-filtering drops recall when selectivity is low; naive pre-filtering disconnects the HNSW graph
    \item \textbf{Dialogue injection:} injecting $\mathcal{R}_{\mathrm{mem}}$ into user turns makes the LLM hallucinate past utterances; always inject into system instructions with provenance timestamps
    \item \textbf{Stale memory contradictions:} append-only memory stores accumulate conflicting facts; run $\mathcal{U}_{\mathrm{cons}}$ ($\mathrm{Merge}/\mathrm{Delete}$) asynchronously
    \item \textbf{Context saturation \& TTFT explosion:} stuffing unreranked chunks inflates prefill $O(\abs{C_t}^2)$ and buries evidence in the middle
  \end{csbul}
\end{cscard}

\begin{cscard}{csD}{10-SECOND VISUAL RECAP}
  \sffamily\footnotesize
  \textbf{BM25 catches exact tokens; Bi-Encoder HNSW catches semantics.}
  Fuse both with calibration-free \textbf{RRF ($\kappa{=}60$)}, rescore top candidates with a \textbf{Cross-Encoder}, and pack RAG + compacted session $\mathcal{K}_{\mathrm{comp}}(\mathcal{S}_t)$ + consolidated memory $\mathcal{M}_t$ inside your token budget $B_{\max}$.
\end{cscard}
