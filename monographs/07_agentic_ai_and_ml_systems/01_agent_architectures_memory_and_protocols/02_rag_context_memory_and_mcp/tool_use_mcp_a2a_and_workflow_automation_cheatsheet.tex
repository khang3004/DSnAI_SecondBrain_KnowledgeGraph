\csbranch{2}{B}{2\;\; HOW $-$ Constrained Decoding \& Post-Training}{%
  \begin{csbul}

\csbranch{6}{F}{6\;\; SECURITY $-$ Confused Deputy \& Shadowing}{%
  \begin{csbul}
    \item \textbf{Confused Deputy:} privileged MCP Server executes injected prompt on behalf of unprivileged user $\to$ enforce per-user scoped OAuth 2.0 tokens via side channel (never in LLM prompt!)
    \item \textbf{Dynamic Capability Injection / Tool Shadowing:} server mutates \texttt{tools/list} or poisons descriptions $\to$ pin manifest hashes $H_{\text{pin}}$ + API Gateway allowlist $\mathcal{A}_{\text{allow}}$
    \item \textbf{Untrusted Hints:} never trust \texttt{readOnlyHint} / \texttt{destructiveHint} alone; require explicit user consent on tainted data flows
  \end{csbul}}

\begin{cscard}{csA}{FORMULAS TO REMEMBER}
  \begin{csbul}
    \item Connector complexity reduction ($M$ hosts, $N$ tools/agents):
      \csfm{C_{\text{bespoke}}=\kappa_{\text{glue}}MN,\quad C_{\text{proto}}=\kappa_{\text{cli}}M+\kappa_{\text{srv}}N,\quad \rho=\frac{MN}{M+N}}
    \item Schema-constrained logit masking ($\mathcal{G}_{\text{JSON}}(\sigma_t)$):
      \csfm{\tilde{z}_s(v)=\begin{cases}z_s(v)&v\in\mathcal{V}(w_{<s})\\-\infty&\text{otherwise}\end{cases},\quad \pi_\theta(v\mid w_{<s})=\mathrm{softmax}(\tilde{z}_s)_v}
    \item RL post-training objective over multi-step tool trajectories:
      \csfm{\mathcal{J}(\theta)=\E_{\tau\sim\pi_\theta}[R(\tau)]-\beta\,\KL(\pi_\theta\,\|\,\pi_{\text{ref}})}
    \item Guarded hybrid DAG reliability ($r_j$ retries, fallback $q_{\text{fb},j}$):
      \csfm{P_{\text{success}}=(1-\epsilon_{\text{det}})^{L_{\text{det}}}\prod_{j=1}^{L_{\text{agent}}}\Bigl(1-p_{\text{fail},j}^{\,r_j+1}\bigl(1-q_{\text{fb},j}\bigr)\Bigr)}
  \end{csbul}
\end{cscard}

\begin{cscard}{csB}{GUARDED MCP EXECUTION LOOP RECIPE}
  \begin{enumerate}[leftmargin=1.2em, itemsep=1pt, topsep=1pt, parsep=0pt]
    \item \textbf{Handshake \& Discover:} open \texttt{stdio} or \texttt{Streamable HTTP}; send JSON-RPC \texttt{initialize} then \texttt{tools/list}.
    \item \textbf{Pin \& Sanitize:} verify $\mathrm{Hash}(\mathcal{M}_j)=H_{\text{pin}}[j]$, intersect with gateway allowlist $\mathcal{A}_{\text{allow}}$, and strip prompt-injection strings.
    \item \textbf{Constrained Decode:} sample $a_k\sim\pi_\theta(\cdot\mid x_{k-1})$ under $\mathcal{G}_{\text{JSON}}(\sigma_t)$.
    \item \textbf{Scoped Dispatch:} check user's OAuth 2.0 scope + consent if tainted/destructive; call \texttt{tools/call} with side-channel credentials.
    \item \textbf{Validate / Self-Correct:} if $o_k.\text{isError}$, pass actionable error to $\pi_\theta$; else validate $\text{outputSchema}$ and paginate into $x_k$.
  \end{enumerate}
\end{cscard}

\begin{cscard}{csG}{PITFALL CHECKLIST}
  \begin{csbul}
    \item \textbf{Confused Deputy privilege escalation:} MCP server using shared admin credentials executes injected prompts for unprivileged users $\to$ bind short-lived per-user OAuth 2.0 tokens
    \item \textbf{Dynamic Capability Injection \& Tool Shadowing:} malicious server swaps tools after vetting or hijacks calls via poisoned docstrings $\to$ pin cryptographic hashes $H_{\text{pin}}$
    \item \textbf{Trusting \texttt{readOnlyHint} / \texttt{destructiveHint}:} tool annotations are unverified hints, never security boundaries
    \item \textbf{1:1 REST wrapper anti-pattern:} dumping raw REST endpoints overwhelms tool selection $\to$ aggregate into task-oriented tools with actionable \texttt{isError:true} messages
    \item \textbf{Unbounded tool outputs \& unguarded AI nodes:} paginate bulky responses and wrap stochastic n8n nodes with schema validators and Error Trigger branches
  \end{csbul}
\end{cscard}

\begin{cscard}{csD}{10-SECOND VISUAL RECAP}
  \sffamily\footnotesize
  \textbf{Constrained decoding} guarantees syntactic JSON validity; \textbf{MCP} standardizes vertical Host$\to$Tool plumbing in $\mathcal{O}(M+N)$; \textbf{A2A} standardizes horizontal Agent$\leftrightarrow$Agent delegation via Agent Cards; and \textbf{hybrid DAGs} wrap stochastic agent steps in deterministic validation and retries.
\end{cscard}
