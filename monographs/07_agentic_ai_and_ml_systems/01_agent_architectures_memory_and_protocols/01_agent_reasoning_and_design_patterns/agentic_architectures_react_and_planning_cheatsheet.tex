\csbranch{6}{F}{6\;\; USE-CASES $-$ when an agent is warranted}{%
  \begin{csbul}
    \item \textbf{Build an agent when:} (i) nuanced context-sensitive decisions, (ii) brittle deterministic rule trees, or (iii) heavy unstructured inputs
    \item \textbf{Do NOT use an agent for:} static single-turn classifiers or fixed pipelines
    \item \textbf{Guardrail stack:} input $\mathcal{G}_{\mathrm{in}}(x)$ (PII, relevance, regex) $\to$ risk check $\mathrm{Risk}(g)$ $\to$ HITL approval gate $\to$ output $\mathcal{G}_{\mathrm{out}}(y)$
    \item \textbf{Small-model rule:} prompting PaLM-8B/62B on ReAct collapses; SFT on $3{,}000$ bootstrapped traces beats 540B prompting
  \end{csbul}}

\csnote{8}{A}{TRAPS $-$ before you trust the output}{%

\begin{cscard}{csA}{FORMULAS TO REMEMBER}
  \begin{csbul}
    \item Augmented action space \& trajectory context:
      \csfm{\hat{\mathcal{A}}=\mathcal{A}\cup\mathcal{L},\quad c_t=(x,o_1,\hat{a}_1,a_1,\dots,o_t)}
    \item Autoregressive ReAct trajectory factorization:
      \csfm{\Prob(\tau\mid x)=\prod_{t=1}^{T}\pi(\hat{a}_t\mid c_t)\,\pi(a_t\mid c_t,\hat{a}_t)\,\mathcal{P}(o_{t+1}\mid s_t,a_t)}
    \item ReAct $\leftrightarrow$ CoT-SC hybrid switching rules ($n=21$):
      \csfm{t > T_{\max}\implies\text{CoT-SC};\quad \max_{y}\sum_{i=1}^{n}\mathbb{I}[y^{(i)}{=}y]<\frac{n}{2}\implies\text{ReAct}}
    \item Iterative refinement stopping criterion:
      \csfm{q(y^{(k)}) = \pi_{\mathrm{eval}}(y^{(k)},x) \ge \tau \;\;\text{or}\;\; k = K_{\max}}
  \end{csbul}
\end{cscard}

\begin{cscard}{csB}{BOUNDED REACT + REFLECTION RECIPE}
  \begin{enumerate}[leftmargin=1.2em, itemsep=0.5pt, topsep=1pt, parsep=0pt]
    \item \textbf{Input Guardrail:} verify $\mathcal{G}_{\mathrm{in}}(x)$; abort/escalate if tripwire fires.
    \item \textbf{Init Context:} $c_1\leftarrow(I,x,\text{LongTermMemory.fetch}(x))$, $t\leftarrow 1$.
    \item \textbf{Think \& Act:} sample $\hat{a}_t\sim\pi(\cdot\mid c_t)$, then $a_t\sim\pi(\cdot\mid c_t,\hat{a}_t)$.
    \item \textbf{Tool Gate:} if $a_t=\texttt{ToolCall}(g,\theta)$, enforce HITL when $\mathrm{Risk}(g)=\mathrm{High}$, execute $o_{t+1}\leftarrow g(\theta)$ in sandbox, append to $c_{t+1}$.
    \item \textbf{Critic Gate:} if $a_t$ emits answer $y$, check $\pi_{\mathrm{eval}}(y,x)\ge\tau$ and $\mathcal{G}_{\mathrm{out}}(y)$; else append critique $f_t$ and loop until $T_{\max}$.
  \end{enumerate}
\end{cscard}

\begin{cscard}{csG}{PITFALL CHECKLIST}
  \begin{csbul}
    \item \textbf{Infinite / repetitive $(a_t,o_t)$ loops:} hash action-observation pairs, enforce $T_{\max}$, and back off to CoT-SC or human escalation
    \item \textbf{Dense Inner Monologue pollution:} forcing thoughts on every step wastes context tokens and drops ALFWorld SR from $71\%$ to $53\%$
    \item \textbf{Non-informative tool returns:} empty search results derail reasoning; return structured ``similar entity'' suggestions
    \item \textbf{Premature multi-agent splitting:} maximize single-agent prompt \& tool clarity before adding multi-agent coordination latency
    \item \textbf{Few-shot ReAct on small models:} fails without trajectory SFT
    \item \textbf{Missing HITL gates:} never allow autonomous execution of high-risk write, deletion, or payment tools without explicit approval
  \end{csbul}
\end{cscard}

\begin{cscard}{csD}{10-SECOND VISUAL RECAP}
  \sffamily\footnotesize
  \textbf{Pure CoT hallucinates; pure Act is myopic.}
  \textbf{ReAct} unifies $\hat{\mathcal{A}}=\mathcal{A}\cup\mathcal{L}$ so thoughts decompose goals and observations ground facts.
  Wrap the loop in \textbf{Reflection} ($q\ge\tau$), \textbf{Guardrails} ($\mathcal{G}_{\mathrm{in}},\mathcal{G}_{\mathrm{out}}$), and \textbf{Manager/Handoff graphs} $G=(V,E)$ to scale safely to production.
\end{cscard}
