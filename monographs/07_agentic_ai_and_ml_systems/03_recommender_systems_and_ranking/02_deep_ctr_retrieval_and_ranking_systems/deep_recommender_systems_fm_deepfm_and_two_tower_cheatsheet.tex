\csbranch{1}{A}{1\;\; FACTORIZATION MACHINES ($O(kd)$)}{%
  \csfm{\hat y_{\mathrm{FM}}=w_0+\sum_i w_i x_i + \frac{1}{2}\sum_{f=1}^k\Big[\big(\sum_i v_{i,f}x_i\big)^2-\sum_i v_{i,f}^2 x_i^2\Big]}
  \begin{csbul}
    \item \textbf{Linear-time trick:} computes all $\binom{d}{2}$ feature crosses in $O(k\cdot\mathrm{nnz}(\vecb x))$
    \item \textbf{Sparse generalization:} estimates unseen $(i,j)$ crosses via shared latent $\vecb v_i, \vecb v_j$
  \end{csbul}}

\csbranch{2}{B}{2\;\; DEEPFM \& NEUMF $-$ Low + High-Order}{%
  \csfm{\hat y_{\mathrm{DeepFM}} = \sigma\bigl(y_{\mathrm{FM}}(\vecb x) + y_{\mathrm{DNN}}(\vecb e)\bigr)}
  \begin{csbul}
    \item \textbf{Shared embeddings:} FM (degree-2) + MLP (high-order) share $\mat V$ without manual crosses
    \item \textbf{NeuMF:} fuses GMF ($\vecb p_u^{\mathrm{G}}\odot\vecb q_i^{\mathrm{G}}$) with MLP branch on $[\vecb p_u^{\mathrm{M}};\vecb q_i^{\mathrm{M}}]$
  \end{csbul}}

\csbranch{3}{C}{3\;\; TWO-TOWER RETRIEVAL \& LOG-Q CORRECTION}{%
  \csfm{s(u,i)=\frac{\inner{u(\vecb x_u)}{v(\vecb x_i)}}{\tau},\quad s_{\mathrm{corr}}(u,j)=s(u,j)-\log Q(j)}
  \begin{csbul}
    \item \textbf{Decoupled towers:} item embeddings indexed offline in HNSW / IVF-PQ ($O(\log N)$)
    \item \textbf{Log-Q fix:} corrects in-batch softmax so popular items are not over-penalized
  \end{csbul}}

\csbranch{4}{D}{4\;\; MULTI-STAGE PRODUCTION FUNNEL}{%
  \begin{csbul}
    \item \textbf{1. Retrieval ($10^7\to 10^3$):} Two-Tower ANN + item-item CF + trending fallback
    \item \textbf{2. Ranking ($10^3\to 10^2$):} DeepFM / DCN multi-task $P(\text{click}), P(\text{buy}), \E[\text{dwell}]$
    \item \textbf{3. Re-ranking ($10^2\to 10^1$):} MMR/DPP diversity, freshness, business rules
  \end{csbul}}

\csbranch{5}{E}{5\;\; SEQUENTIAL RS \& COLD-START}{%
  \begin{csbul}
    \item \textbf{SASRec / Transformers:} causal self-attention over user interaction sequence $(i_1,\dots,i_t)$
    \item \textbf{Cold-start:} content metadata / text/vision embeddings in Two-Tower bypass missing IDs
  \end{csbul}}

\csbranch{6}{F}{6\;\; RANKING METRICS $-$ NDCG@K \& CALIBRATION}{%
  \csfm{\mathrm{DCG}@K=\sum_{r=1}^K \frac{2^{\mathrm{rel}_r}-1}{\log_2(r+1)},\qquad \mathrm{NDCG}@K=\frac{\mathrm{DCG}@K}{\mathrm{IDCG}@K}}
  \begin{csbul}
    \item \textbf{Calibration (ECE / RIG):} ad auctions need true probabilities $\hat p$, not just rank order
  \end{csbul}}

\begin{cscard}{csA}{FORMULAS TO REMEMBER}
  \begin{csbul}
    \item \textbf{FM Trick:} $\sum_{i<j}\inner{\vecb v_i}{\vecb v_j}x_i x_j = \frac{1}{2}\sum_{f=1}^k[(\sum_i v_{i,f}x_i)^2-\sum_i v_{i,f}^2 x_i^2]$
    \item \textbf{Log-Q In-Batch Loss:} $-\log\frac{e^{s(u,i^+)-\log Q(i^+)}}{e^{s(u,i^+)-\log Q(i^+)}+\sum_{j\in\mathcal B^-}e^{s(u,j)-\log Q(j)}}$
    \item \textbf{DCG@K:} $\sum_{r=1}^K \frac{2^{\mathrm{rel}_r}-1}{\log_2(r+1)}$ (logarithmic rank discount)
    \item \textbf{IPS Debiasing:} $\hat R_{\mathrm{IPS}}=\frac{1}{N}\sum_{t}\frac{\mathbb{I}(\text{click}_t)\,\ell_t}{p(\text{expose at pos}_t)}$
  \end{csbul}
\end{cscard}

\begin{cscard}{csB}{PRODUCTION RANKING PIPELINE RECIPE}
  \begin{enumerate}[leftmargin=1.2em, itemsep=1pt, topsep=1pt, parsep=0pt]
    \item \textbf{Retrieve:} embed user/context online, query HNSW/ScaNN for top-$10^3$ items.
    \item \textbf{Cross-score:} hydrate dense+sparse features from Feature Store and score with DeepFM/DCN.
    \item \textbf{Multi-objective combine:} $S = w_1\hat p_{\mathrm{click}} + w_2\hat p_{\mathrm{buy}} + w_3\hat t_{\mathrm{dwell}}$.
    \item \textbf{Diversify \& Debias:} apply DPP/MMR re-ranking and log random exploration propensities.
  \end{enumerate}
\end{cscard}

\begin{cscard}{csG}{PITFALL CHECKLIST}
  \begin{csbul}
    \item \textbf{Missing log-Q correction:} in-batch negatives sample popular items too often, suppressing their scores.
    \item \textbf{Position bias loop:} top slots get more clicks; include position feature at train time and zero/marginalize at inference.
    \item \textbf{Ranker selection bias:} training only on past ranker winners fails on retrieval tail; log randomized exploration slices.
    \item \textbf{Two-Tower late-interaction ceiling:} pure dot product misses fine $(u,i)$ feature crosses—always follow with DeepFM/DCN.
  \end{csbul}
\end{cscard}
