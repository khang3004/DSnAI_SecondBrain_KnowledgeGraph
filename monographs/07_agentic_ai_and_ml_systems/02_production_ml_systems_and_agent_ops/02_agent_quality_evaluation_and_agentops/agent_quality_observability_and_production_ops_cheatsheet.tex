\csbranch{2}{B}{2\;\; HORIZON DECAY $-$ \texttt{pass@k} vs.\ \texttt{pass\textasciicircum k}}{%
  \begin{csbul}
    \item \textbf{Step compounding:} $p_\tau=\prod_{t=1}^{H}p_t\approx p_{\mathrm{step}}^H$
    \item \textbf{Demo} (\texttt{pass@k}) vs.\ \textbf{SLA consistency} (\texttt{pass\textasciicircum k}):
      \csfm{\mathrm{pass@}k=1-(1-p_\tau)^k,\quad \mathrm{pass\textasciicircum}k=p_\tau^k=p_{\mathrm{step}}^{kH}}
    \item \textbf{Trap ($p_{\mathrm{step}}=0.95, H=10, k=5$):} $p_\tau\approx 0.60 \Rightarrow \mathrm{pass@}5=99.0\%$ (great demo!) but $\mathrm{pass\textasciicircum}5=7.7\%$ ($>92\%$ SLA streak failure)
  \end{csbul}}

\csbranch{3}{C}{3\;\; LLM-AS-A-JUDGE \& HITL ALIGNMENT}{%
  \begin{csbul}
    \item \textbf{Symmetric slot-swapped win rate} (cancels position bias):
      \csfm{\hat w_{A\succ B}(x)=\tfrac{1}{2}\bigl[J_\phi(y_A,y_B\mid x)+\bigl(1-J_\phi(y_B,y_A\mid x)\bigr)\bigr]}
    \item \textbf{Cohen's $\kappa$ vs.\ HITL experts:} $\kappa=(p_o-p_e)/(1-p_e)$
    \item \textbf{CI/CD sample power:} $N\ge\frac{(z_{1-\alpha/2}+z_{1-\beta})^2\sigma_w^2}{(\Delta w_{\min})^2}$
    \item \textbf{Pairwise $>$ Pointwise:} cancels shared query difficulty
  \end{csbul}}

\csbranch{4}{D}{4\;\; SLICING, SIMPSON'S PARADOX \& DEFERRAL}{%
  \begin{csbul}
    \item \textbf{Slice audit:} track $\min_{g\in\mathcal{G}_{\mathrm{slice}}}S_g(\pi)$ across cohorts
    \item \textbf{Simpson's reversal:} $S_g(\pi_B){>}S_g(\pi_A)\,\forall g$ yet aggregate $S(\pi_B){<}S(\pi_A)$ if weights $w_g^{(A)}{\ne}w_g^{(B)}$
    \item \textbf{Bayes HITL deferral} for tool risk $r_a$:
      \csfm{\tau_{\mathrm{conf}}(r_a)=1-\frac{C_{\mathrm{HITL}}}{C_{\mathrm{err}}(r_a)}\quad\bigl(C_{\mathrm{err}}\gg C_{\mathrm{HITL}}\Rightarrow\tau_{\mathrm{conf}}\to 1\bigr)}
  \end{csbul}}

\begin{cscard}{csA}{FORMULAS TO REMEMBER}
  \begin{csbul}
    \item Trajectory horizon decay:\csfm{p_\tau=\prod_{t=1}^{H}p_t\approx p_{\mathrm{step}}^{H}}
    \item Capability (\texttt{pass@k}) vs.\ Consistency (\texttt{pass\textasciicircum k}):\csfm{\mathrm{pass@}k=1-(1-p_\tau)^k,\qquad \mathrm{pass\textasciicircum}k=p_\tau^k=p_{\mathrm{step}}^{kH}}
    \item Position-debiased pairwise judge estimator:\csfm{\hat w_{A\succ B}(x)=\tfrac{1}{2}\bigl[J_\phi(y_A,y_B\mid x)+\bigl(1-J_\phi(y_B,y_A\mid x)\bigr)\bigr]}
    \item Cohen's $\kappa$, Sample Power \& HITL Deferral:\csfm{\kappa=\frac{p_o-p_e}{1-p_e},\quad N\ge\frac{(z_{1-\alpha/2}+z_{1-\beta})^2\sigma_w^2}{(\Delta w_{\min})^2},\quad \tau_{\mathrm{conf}}=1-\frac{C_{\mathrm{HITL}}}{C_{\mathrm{err}}}}
  \end{csbul}

\begin{cscard}{csB}{GUARDED CANARY PROMOTION RECIPE}
  \begin{enumerate}[leftmargin=1.2em, itemsep=1pt, topsep=1pt, parsep=0pt]
    \item \textbf{Sandbox Trace:} run $\pi_{\mathrm{cand}}$ on $\mathcal{D}_{\mathrm{gold}}$ with OTel spans $\to$ output $y_{\mathrm{cand}}$, trace $\tau$
    \item \textbf{Deterministic Gate:} block on schema, regex/PII, or perturbation failure
    \item \textbf{Judge \& Slice Gate:} require $\kappa(J_\phi,\mathrm{HITL})\ge\kappa_{\min}$, $\min_g S_g\ge\tau_{\mathrm{slice}}$, and $\bar w\ge\tau_{\mathrm{win}}$
    \item \textbf{Canary \& Evolve:} route $1\%\to 100\%$ with parallel guardrails; on tripwire spike, rollback and add $\mathcal{G}_{\hat\tau}$ to $\mathcal{D}_{\mathrm{gold}}$
  \end{enumerate}
\end{cscard}

\begin{cscard}{csG}{PITFALL CHECKLIST}
  \begin{csbul}
    \item \textbf{Judge biases:} position (slot 1), verbosity, and self-preference skew ratings $\to$ swap slots \& check Cohen's $\kappa$
    \item \textbf{Demo illusion (\texttt{pass@k}):} high $\mathrm{pass@}5$ hides multi-step decay $\to$ gate CI/CD on $\mathrm{pass\textasciicircum}k$ consistency
    \item \textbf{Harness \& token drift:} sandbox, prompt, or token budget changes swing scores $\to$ lock harness in CI
    \item \textbf{8-gram contamination:} public leaderboards leak into pretraining $\to$ run 8-gram/50-char decontamination
    \item \textbf{Simpson's paradox:} aggregate canary pass masks minority failures $\to$ enforce $\min_g S_g\ge\tau_{\mathrm{slice}}$
    \item \textbf{Duplicate side effects:} retrying write tools without idempotency keys triggers double charges
  \end{csbul}
\end{cscard}

\begin{cscard}{csD}{10-SECOND VISUAL RECAP}

  Debias pairwise judges via slot swapping, gate every slice $\min_g S_g$, run optimistic parallel guardrails, and turn production OTel failures into Golden Set tests.
\end{cscard}
