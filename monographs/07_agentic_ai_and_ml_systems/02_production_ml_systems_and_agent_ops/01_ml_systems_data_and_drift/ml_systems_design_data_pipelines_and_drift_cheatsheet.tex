\csbranch{2}{B}{2\;\; IMBALANCE \& SPLIT-INVARIANT FEATURES}{%
  \begin{csbul}
    \item \textbf{Focal Loss:} suppresses easy majority examples ($p_t\!\to\!1$):
      \csfm{\mathcal{L}_{\mathrm{FL}}(p_t)=-\alpha_t(1-p_t)^\gamma\log(p_t),\quad W_i^{\mathrm{eff}}=\frac{1-\beta}{1-\beta^{N_i}}}
    \item \textbf{Prior shift calibration:} oversampling by $r$ shifts prior $\tilde{\pi}_1=\frac{r\pi_1}{r\pi_1+1-\pi_1}$ $\Rightarrow$ recalibrate via Platt/isotonic scaling
    \item \textbf{Split invariance:} fit scaler/imputer $\psi^*=\mathcal{A}_{\mathrm{fit}}(\mathcal{D}_{\mathrm{train}})$ \emph{only}; join features point-in-time ($\tau\le t$, $t_{\mathrm{train}}<t_{\mathrm{val}}<t_{\mathrm{test}}$)
  \end{csbul}}

\csbranch{3}{C}{3\;\; SHIFT TAXONOMY $-$ Factorizing $P(X,Y)$}{%
  \renewcommand{\arraystretch}{1.15}\scriptsize
  \begin{tabular}{@{}>{\raggedright\arraybackslash}p{1.65cm}>{\raggedright\arraybackslash}p{2.35cm}>{\raggedright\arraybackslash}p{2.3cm}@{}}
    \textbf{\textcolor{csC}{Regime}} & \textbf{\textcolor{csC}{Condition}} & \textbf{\textcolor{csC}{Remedy}}\\ \hline
    Covariate Shift & $P_{\mathrm{tgt}}(X)\ne P_{\mathrm{src}}(X)$, $P(Y|X)$ fixed & Importance weight $w(\vecb x)=\frac{P_{\mathrm{tgt}}(\vecb x)}{P_{\mathrm{src}}(\vecb x)}$\\
    Label Shift & $P_{\mathrm{tgt}}(Y)\ne P_{\mathrm{src}}(Y)$, $P(X|Y)$ fixed & BBSE prior $\hat{\vecb\mu}_{\mathrm{tgt}}=\mat C_{\mathrm{src}}\vecb w$\\
    Concept Drift & $P_{\mathrm{tgt}}(Y|X)\ne P_{\mathrm{src}}(Y|X)$, $P(X)$ fixed & Needs delayed labels $Y$; retrain $\theta$\\
  \end{tabular}\par\vspace{2pt}
  \small\textbf{Identity:} $P(X,Y)=P(Y\mid X)P(X)=P(X\mid Y)P(Y)$.}

\csbranch{4}{D}{4\;\; DRIFT METRICS $-$ KS, PSI \& Kernel MMD}{%

\begin{cscard}{csA}{FORMULAS TO REMEMBER}
  \begin{csbul}
    \item Cost-sensitive \& Focal loss:\csfm{\mathcal{L}_{\mathrm{CS}}(\vecb x,i)=\sum_{j=1}^{K}C_{ij}P(j\mid\vecb x),\quad \mathcal{L}_{\mathrm{FL}}=-\alpha_t(1-p_t)^\gamma\log p_t}
    \item Effective class weight:\csfm{E_{N_i}=\frac{1-\beta^{N_i}}{1-\beta},\qquad W_i^{\mathrm{eff}}=\frac{1-\beta}{1-\beta^{N_i}}}
    \item Covariate-shift target risk:\csfm{\E_{P_{\mathrm{tgt}}}[\ell(\vecb x,y;\theta)]=\E_{P_{\mathrm{src}}}\!\left[\frac{P_{\mathrm{tgt}}(\vecb x)}{P_{\mathrm{src}}(\vecb x)}\,\ell(\vecb x,y;\theta)\right]}
    \item PSI \& Kernel MMD$^2$:\csfm{\mathrm{PSI}=\sum_{b=1}^{B}(q_b-p_b)\ln\frac{q_b}{p_b},\quad \mathrm{MMD}^2=\norm{\mu_P-\mu_Q}_{\mathcal{H}}^2}
  \end{csbul}
\end{cscard}

\begin{cscard}{csB}{CONTINUAL RETRAINING RECIPE}
  \begin{enumerate}[leftmargin=1.2em, itemsep=1pt, topsep=1pt, parsep=0pt]
    \item \textbf{As-of Join:} fetch $\vecb x_t\leftarrow\mathrm{AsOfJoin}(\mathcal{S}_F,\text{entity}_t,\tau\le t)$
    \item \textbf{Log-and-Wait:} serve $\hat y_t=f_{\theta_{\mathrm{champ}}}(\vecb x_t)$; log $(\vecb x_t,\hat y_t,t)$
    \item \textbf{Drift Test:} track $\mathrm{PSI}_j(\mathcal{W}_{\mathrm{ref}},\mathcal{W}_t)$ and $\widehat{\mathrm{MMD}}^2$ on window $\mathcal{W}_t$
    \item \textbf{Stateful Gate:} if $\mathrm{PSI}\ge 0.25$, fine-tune $\theta_{\mathrm{chal}}$ on $\mathcal{D}_{\mathrm{new}}$; promote if holdout gain $>\delta_{\min}$ and reset $\mathcal{W}_{\mathrm{ref}}\leftarrow\mathcal{W}_t$
  \end{enumerate}
\end{cscard}

\begin{cscard}{csG}{PITFALL CHECKLIST}
  \begin{csbul}
    \item \textbf{Preprocessing leakage:} fitting $\mu,\sigma$ or SMOTE before splitting leaks test moments
    \item \textbf{Group \& duplicate leakage:} keep user/patient in one split; deduplicate before splitting
    \item \textbf{Sentinel collision:} never impute NaN with $0$ without indicator $m_j=\mathbb{I}[x_j\text{ is NaN}]$
    \item \textbf{Train--serving skew:} dual SQL/stream codebases diverge $\to$ use feature store or Log-and-Wait
    \item \textbf{Catastrophic forgetting:} sample-by-sample updates overwrite regimes $\to$ gate micro-batches
    \item \textbf{High-$d$ SMOTE / KS failure:} distance concentration breaks $d\gg 100$ SMOTE; 1D KS misses correlation shift
  \end{csbul}
\end{cscard}

\begin{cscard}{csD}{10-SECOND VISUAL RECAP}
  \sffamily\footnotesize
  \textbf{Production ML} \,=\, \textbf{FTI Decoupling} ($\mathcal{P}_F\!\to\!\mathcal{S}_F\!\to\!\mathcal{P}_T\!\to\!\mathcal{R}_M\!\to\!\mathcal{P}_I$) $+$ \textbf{As-Of Joins} ($\tau\le t$).
  Factorize shift into \textbf{Covariate} $P(X)$, \textbf{Label} $P(Y)$, and \textbf{Concept} $P(Y\mid X)$.
  Track sliding-window $\mathrm{PSI}$ and $\mathrm{MMD}^2$; when $\mathrm{PSI}\ge 0.25$, trigger \textbf{stateful continual retraining} and promote via champion--challenger gating.
\end{cscard}
