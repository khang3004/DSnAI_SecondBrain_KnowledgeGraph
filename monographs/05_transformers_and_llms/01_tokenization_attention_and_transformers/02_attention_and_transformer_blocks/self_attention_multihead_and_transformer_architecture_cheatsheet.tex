\csbranch{1}{A}{1\;\; WHAT $-$ differentiable dictionary lookup}{%
  \begin{minipage}[t]{0.60\linewidth}
    \csfm{\mathrm{Attention}(\mat{Q},\mat{K},\mat{V}) = \softmax\!\left(\frac{\mat{Q}\mat{K}\T}{\sqrt{d_k}}\right)\mat{V}}

\csbranch{2}{B}{2\;\; WHY $1/\sqrt{d_k}$ \& PRE-SOFTMAX MASKING}{%
  \csfm{\Var(\vecb{q}\T\vecb{k}) = d_k \;\implies\; \Var\!\left(\frac{\vecb{q}\T\vecb{k}}{\sqrt{d_k}}\right) = 1}
  \begin{csbul}
    \item \textbf{Softmax Jacobian:} $\pd{p_i}{s_j}=p_i(\delta_{ij}-p_j)\to 0$ if large $\Var(s)$ saturates $p_{i^*}\to 1$
    \item \textbf{Causal mask $\mat{M}$:} $M_{ij}=0$ ($j\le i$), $M_{ij}=-\infty$ ($j>i$) added \emph{before} $\softmax$
    \item \textbf{Zero leakage:} $A_{ij} = \frac{\exp(S_{ij})\mathbb{I}[j\le i]}{\sum_{m=1}^i \exp(S_{im})}$ sums strictly over prefix $m\le i$
  \end{csbul}}

\csbranch{3}{C}{3\;\; MULTI-HEAD, GQA \& PRE-LN BLOCK}{%
  \begin{minipage}[t]{0.60\linewidth}
    \csfm{\mathrm{MHA}(\mat{X}) = \mathrm{Concat}(\mat{H}_1,\dots,\mat{H}_H)\mat{W}_O}
    \begin{csbul}
      \item \textbf{Split:} $[B,T,d]\to[B,H,T,d_k]$, $d_k=d/H$
      \item \textbf{GQA:} $H$ query heads share $H_{\mathrm{kv}}$ KV heads
      \item \textbf{Pre-LN:} $\mat{Z}=\mat{X}+\mathrm{MHA}(\mathrm{LN}_1(\mat{X}))$, $\tilde{\mat{X}}=\mat{Z}+\mathrm{MLP}(\mathrm{LN}_2(\mat{Z}))$

\csbranch{6}{F}{6\;\; KERNEL VIEW \& TIED UNEMBEDDING}{%
  \begin{csbul}
    \item \textbf{Nadaraya--Watson smoother:} $A_{ij} = \frac{\kappa(\vecb{q}_i,\vecb{k}_j)}{\sum_m \kappa(\vecb{q}_i,\vecb{k}_m)}$ on simplex $\Delta^{N-1}$
    \item \textbf{Position-wise MLP:} GELU or gated SwiGLU injects non-linearity per token
    \item \textbf{Tied LM head:} logits $\vecb{u}_i = (\vecb{h}_i^{(L)})\T\mat{E}\T\in\R^{|\mathcal{V}|}$ shares input embedding $\mat{E}$
    \item \textbf{$O(1)$ path length:} any token pair interacts in $1$ hop vs.\ $O(N)$ in RNNs
  \end{csbul}}

\csnote{8}{A}{TRAPS $-$ before you trust the output}{%
  \begin{csbul}
    \item omitting $1/\sqrt{d_k}$ gives $\Var(\vecb{q}\T\vecb{k})=d_k\Rightarrow$ vanishing softmax Jacobian
    \item masking \emph{after} $\softmax$ leaks future mass into denominator
    \item all-$-\infty$ row yields $0/0=\texttt{NaN}$; keep $j=i$ unmasked
    \item Post-LN scales residual stream $\Rightarrow$ unstable deep gradients
  \end{csbul}}

\begin{cscard}{csA}{FORMULAS TO REMEMBER}
  \begin{csbul}
    \item Causal Scaled Dot-Product Attention:\csfm{\mat{A}_{\mathrm{causal}} = \softmax\!\left(\frac{\mat{Q}\mat{K}\T}{\sqrt{d_k}} + \mat{M}\right)\mat{V}}
    \item Variance \& Softmax Jacobian:\csfm{\Var\!\left(\frac{\vecb{q}\T\vecb{k}}{\sqrt{d_k}}\right)=1,\qquad \pd{p_i}{s_j} = p_i(\delta_{ij}-p_j)}
    \item Multi-Head Projection ($d_k = d_{\mathrm{model}}/H$):\csfm{\mat{H}_h = \mathrm{Attn}(\mat{X}\mat{W}_{Q,h},\mat{X}\mat{W}_{K,h},\mat{X}\mat{W}_{V,h})}
    \item Pre-LN Transformer Block:\csfm{\mat{Z}^{(\ell)} = \mat{X}^{(\ell)} + \mathrm{MHA}(\mathrm{LN}_1(\mat{X}^{(\ell)})),\quad \mat{X}^{(\ell+1)} = \mat{Z}^{(\ell)} + \mathrm{MLP}(\mathrm{LN}_2(\mat{Z}^{(\ell)}))}
  \end{csbul}
\end{cscard}

\begin{cscard}{csB}{BATCHED MHA TENSOR-SPLIT RECIPE}
  \begin{enumerate}[leftmargin=1.2em, itemsep=1pt, topsep=1pt, parsep=0pt]
    \item Pre-normalize input: $\tilde{\mathcal{X}} = \mathrm{LN}_1(\mathcal{X}) \in \R^{B\times T\times d}$.
    \item Fused QKV projection: $\mathcal{Q},\mathcal{K},\mathcal{V} \in \R^{B\times T\times d}$.
    \item View \& transpose to heads: $[B,T,H,d_k] \to [B,H,T,d_k]$.
    \item Logits $\mathcal{S} = (\mathcal{Q}\mathcal{K}\T)/\sqrt{d_k} \in \R^{B\times H\times T\times T}$; set $S_{b,h,i,j}=-\infty$ for $j>i$.
    \item $\mathcal{H} = \softmax(\mathcal{S})\mathcal{V} \to$ transpose to $[B,T,d] \to \mathcal{H}\mat{W}_O + \mathcal{X}$.
  \end{enumerate}
\end{cscard}

\begin{cscard}{csG}{PITFALL CHECKLIST}
  \begin{csbul}
    \item \textbf{Missing $1/\sqrt{d_k}$:} standard deviation grows as $\sqrt{d_k}$, pushing softmax into one-hot saturation where $\pd{p_i}{s_j}\approx 0$.
    \item \textbf{Post-softmax zeroing:} multiplying $\softmax(\mat{S})$ by a lower-triangular mask leaves rows summing to $<1$ and leaks future magnitude unless renormalized.
    \item \textbf{All-masked padding row:} if padding + causal mask sets an entire row to $-\infty$, $\softmax$ produces $\texttt{NaN}$; always keep $j=i$ unmasked or guard empty rows.
    \item \textbf{FP16 $\exp$ overflow:} subtract row-wise $m_i=\max_j S_{ij}$ before exponentiating and accumulate softmax denominator in FP32.
    \item \textbf{Transpose bug:} reshaping $[B,T,d]\to[B,H,T,d_k]$ directly without \texttt{.view(B,T,H,d\_k).transpose(1,2)} scrambles sequence and head axes.
  \end{csbul}
\end{cscard}

\begin{cscard}{csD}{10-SECOND VISUAL RECAP}
  \sffamily\footnotesize
  Self-attention routes information in $O(1)$ sequential hops by matching \textbf{queries} against \textbf{keys} ($\mat{Q}\mat{K}\T/\sqrt{d_k}$), masking future tokens with $-\infty$ before $\softmax$, and taking a convex combination of \textbf{values} $\mat{V}$. Wrapping $H$ parallel heads and a position-wise MLP inside a \textbf{Pre-LN residual stream} yields the universal building block of GPT, BERT, T5, and ViT.
\end{cscard}
