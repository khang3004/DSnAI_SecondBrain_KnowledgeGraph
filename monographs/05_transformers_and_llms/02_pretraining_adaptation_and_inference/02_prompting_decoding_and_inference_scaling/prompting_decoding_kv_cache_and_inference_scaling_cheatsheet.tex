\csbranch{2}{B}{2\;\; DECODING $-$ Temp $\tau$, Top-$k$ \& Nucleus Top-$p$}{%
  \csfm{p_\tau(w_j) = \frac{\exp(u_j/\tau)}{\sum_v \exp(u_v/\tau)},\qquad m^* = \min\Bigl\{m : \sum_{r=1}^m p_\tau(w_{(r)}) \ge p\Bigr\}}
  \begin{csbul}
    \item \textbf{Limits:} $\tau\to 0^+$ recovers greedy $\argmax$; $\tau\to\infty$ is uniform $1/|\mathcal{V}|$
    \item \textbf{Beam search ($K$):} keeps top-$K$ prefixes; good for MT, dull/repetitive in chat
    \item \textbf{Top-$k$ vs.\ Top-$p$:} fixed $k$ leaks tail tokens when peaked; nucleus $p$ adapts $|\mathcal{V}_p|$ dynamically to local entropy
  \end{csbul}}

\csbranch{3}{C}{3\;\; KV CACHE $-$ Prefill vs.\ Decode \& PagedAttention}{%
  \begin{minipage}[t]{0.61\linewidth}
    \csfm{M_{\mathrm{KV}} = 2\,b\,s\,n_{\mathrm{layers}}\,n_{\mathrm{kv}}\,d_h\,p_{\mathrm{bytes}}}
    \begin{csbul}
      \item \textbf{Prefill ($m$ tokens):} parallel GEMMs, \emph{compute-bound} (sets TTFT)
      \item \textbf{Decode ($1$ token):} arithmetic intensity $\mathcal{I}\sim O(b)$, \emph{HBM-BW-bound} (sets ITL)
      \item \textbf{GQA / MQA:} shrinks $M_{\mathrm{KV}}$ by $n_q/n_{\mathrm{kv}}$
      \item \textbf{PagedAttention:} non-contiguous KV blocks $\Rightarrow 0\%$ external waste
    \end{csbul}
  \end{minipage}\hfill

\csbranch{4}{D}{4\;\; SPECULATIVE DECODING $-$ Lossless Rejection}{%
  \csfm{\text{Accept }\hat{y}\sim q\text{ w.p. }\min\!\left(1,\frac{p(\hat{y})}{q(\hat{y})}\right);\quad p_{\mathrm{res}}(y) = \frac{\max(0,p(y)-q(y))}{1-\sum_v\min(p(v),q(v))}}
  \begin{csbul}
    \item \textbf{Draft--Verify:} small $q$ drafts $\tau$ tokens; target $p$ scores all $\tau{+}1$ in \textbf{1 parallel pass}
    \item \textbf{Exact identity:} $\min(p,q) + \max(0,p-q) = p(y)\implies$ \textbf{100\% lossless}!
    \item \textbf{Acceptance rate:} $\alpha = \sum_v \min(p(v),q(v))$; yields up to $\tau+1$ tokens/step
  \end{csbul}}

    \textbf{\textcolor{csE}{Mechanism}} & \textbf{\textcolor{csE}{Step Cost}} & \textbf{\textcolor{csE}{Bottleneck / Gain}}\\ \hline
    Prefill ($m$ tok) & $O(m^2 d + m d^2)$ & Compute-bound (TTFT)\\

    Continuous Batch & Iteration-level & Immediate EOS slot reuse\\
  \end{tabular}\par\vspace{2pt}
  \footnotesize \textbf{Chunked prefill:} splits long prompts to prevent ITL stalls.}

\csbranch{6}{F}{6\;\; TEST-TIME SCALING $-$ Best-of-$N$, PRM \& RLVR}{%
  \begin{csbul}
    \item \textbf{Parallel scaling:} Best-of-$N$ with verifier $V(\vecb{x},\vecb{y}^{(m)})$ or majority vote
    \item \textbf{Process Reward Model (PRM):} scores intermediate steps $V(\vecb{x},\vecb{a}_{\le i})$ for MCTS/beam search (beats terminal ORM)
    \item \textbf{RLVR (Verifiable Rewards):} binary $r\in\{0,\gamma\}$ (math/code tests) elicits native long CoT, backtracking \& self-verification
  \end{csbul}}

\csnote{8}{A}{TRAPS $-$ before you trust the output}{%
  \begin{csbul}

    \item top-$p$ must always keep $w_{(1)}$ when $p_{(1)} > p$ to avoid $\emptyset$
    \item unconstrained RLVR/PRM risks \textbf{overthinking} / hacking
  \end{csbul}}

\begin{cscard}{csA}{FORMULAS TO REMEMBER}
  \begin{csbul}
    \item Autoregressive Chain Rule \& Logits:\csfm{\log\Pr_\theta(\vecb{y}\mid\vecb{x}) = \sum_{i=1}^n \log\softmax(\mat{W}_o\T\vecb{h}_{m+i})_{y_i}}
    \item Top-$p$ (Nucleus) Support:\csfm{\mathcal{V}_p = \{w_{(1)},\dots,w_{(m^*)}\},\quad m^* = \min\Bigl\{m:\sum_{r=1}^m p_\tau(w_{(r)})\ge p\Bigr\}}
    \item KV Cache Footprint ($s = m + n$):\csfm{M_{\mathrm{KV}} = 2 \cdot b \cdot s \cdot n_{\mathrm{layers}} \cdot n_{\mathrm{kv}} \cdot d_h \cdot p_{\mathrm{bytes}}}
    \item Lossless Speculative Decomposition:\csfm{\Pr(Y=y) = \underbrace{q(y)\min\!\Bigl(1,\frac{p(y)}{q(y)}\Bigr)}_{\text{accept}} + \underbrace{\max(0,p(y)-q(y))}_{\text{reject \& resample}} = p(y)}
  \end{csbul}
\end{cscard}

\begin{cscard}{csB}{SPECULATIVE DECODING RECIPE}
  \begin{enumerate}[leftmargin=1.2em, itemsep=1pt, topsep=1pt, parsep=0pt]
    \item Draft $\tau$ tokens $\hat{y}_{1:\tau} \sim q(\cdot)$ autoregressively with small model.
    \item Run target model $p(\cdot)$ in \textbf{1 parallel forward pass} on prefix $+ \hat{y}_{1:\tau}$ to get $\tau{+}1$ distributions.
    \item Loop $t=1\dots\tau$: draw $r_t\sim\Uniform(0,1)$; if $r_t > p(\hat{y}_t)/q(\hat{y}_t)$, reject $\hat{y}_{\ge t}$, sample $\bar{y}_t \sim p_{\mathrm{res}}$, and break.
    \item If all $\tau$ accepted, sample bonus token $\bar{y}_{\tau+1}\sim p_{\tau+1}$.
  \end{enumerate}
\end{cscard}

\begin{cscard}{csG}{PITFALL CHECKLIST}
  \begin{csbul}
    \item \textbf{Filter order matters:} apply temperature $\tau$, top-$k$, and top-$p$ in a consistent pipeline and \emph{always} renormalize $\vecb{p}$ after zeroing tail tokens.
    \item \textbf{Always keep top-1 token $w_{(1)}$:} in nucleus sampling, include the boundary token $m^*$ that crosses threshold $p$ so $\mathcal{V}_p$ is never empty.
    \item \textbf{Speculative acceptance collapse:} if draft $q$ and target $p$ use different tokenizers or mismatched temperatures, $\alpha=\sum_v\min(p,q)$ plummets.
    \item \textbf{Head-of-line blocking:} a $32\text{k}$-token prefill stalls active decode batches unless split via \emph{chunked prefilling}.
    \item \textbf{PRM reward hacking \& overthinking:} uncalibrated verifiers get exploited at large $N$; RLVR without length penalties loops on easy prompts.
  \end{csbul}
\end{cscard}
