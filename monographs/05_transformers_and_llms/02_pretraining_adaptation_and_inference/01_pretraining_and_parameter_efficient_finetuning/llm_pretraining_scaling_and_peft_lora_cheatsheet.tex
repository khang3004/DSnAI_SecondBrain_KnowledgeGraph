\csbranch{1}{A}{1\;\; PRETRAINING $-$ CLM, MLM \& Perplexity}{%
  \csfm{\mathcal{L}_{\mathrm{CE}}(\theta;\vecb{x}) = -\frac{1}{m}\sum_{i=0}^{m-1}\log\Prob_\theta(x_{i+1}\mid x_{\le i}),\quad \mathrm{PPL} = \exp(\mathcal{L}_{\mathrm{CE}})}
  \begin{csbul}
    \item \textbf{MLE equivalence:} minimizing $\mathcal{L}_{\mathrm{CE}} \iff \argmax_\theta \sum\log\Prob_\theta(\vecb{x})$
    \item \textbf{Branching factor:} $\mathrm{PPL}=K \iff$ uniform uncertainty over $K$ tokens
    \item \textbf{Untrained check:} at init, $\mathcal{L}_{\mathrm{CE}}\approx \ln|\mathcal{V}| \implies \mathrm{PPL}\approx |\mathcal{V}|$
    \item \textbf{MLM (15\% corrupted):} 80\% \texttt{[MASK]}, 10\% random, 10\% unchanged
  \end{csbul}}

\csbranch{2}{B}{2\;\; SCALING LAWS $-$ Chinchilla vs.\ Kaplan}{%
  \begin{minipage}[t]{0.62\linewidth}

\csbranch{4}{D}{4\;\; LoRA $-$ Low-Rank Adaptation Math}{%
  \csfm{\vecb{h} = \mat{W}_0\vecb{x} + \frac{\alpha}{r}\mat{B}\mat{A}\vecb{x},\qquad \mat{A}\in\R^{r\times d_{\mathrm{in}}},\;\mat{B}\in\R^{d_{\mathrm{out}}\times r}}
  \begin{csbul}
    \item \textbf{Zero-perturbation init:} $\mat{A}\sim\mathcal{N}(0,\sigma^2),\;\mat{B}=\mat{0}\implies \Delta\mat{W}_0=\mat{0}$ at $t=0$
    \item \textbf{Gradients:} $\pd{\mathcal{L}}{\mat{B}}=\frac{\alpha}{r}(\pd{\mathcal{L}}{\vecb{h}})(\mat{A}\vecb{x})\T \neq \mat{0}$ at step 1; $\pd{\mathcal{L}}{\mat{A}}=\frac{\alpha}{r}\mat{B}\T(\pd{\mathcal{L}}{\vecb{h}})\vecb{x}\T$
    \item \textbf{Param savings:} $r(d_{\mathrm{in}}+d_{\mathrm{out}})$ vs.\ $d_{\mathrm{out}}d_{\mathrm{in}}$ ($\sim 0.5\%$ of 8B for $r=16$)
    \item \textbf{Zero inference overhead:} merge $\mat{W}_{\mathrm{eff}}=\mat{W}_0+\frac{\alpha}{r}\mat{B}\mat{A}$ offline
  \end{csbul}}

\csnote{8}{A}{TRAPS $-$ before you trust the output}{%
  \begin{csbul}

\begin{cscard}{csA}{FORMULAS TO REMEMBER}
  \begin{csbul}
    \item Causal LM Cross-Entropy \& Perplexity:\csfm{\mathcal{L}_{\mathrm{CE}} = -\frac{1}{m}\sum_{i=0}^{m-1}\log\Prob_\theta(x_{i+1}\mid x_{\le i}),\quad \mathrm{PPL} = e^{\mathcal{L}_{\mathrm{CE}}}}
    \item Chinchilla Scaling Law ($C \approx 6ND$):\csfm{L(N,D) = E + A N^{-\alpha} + B D^{-\beta},\quad \alpha A N^{-\alpha} = \beta B D^{-\beta}}
    \item Compute-Optimal Frontier:\csfm{N^*(C) \propto C^{\frac{\beta}{\alpha+\beta}}\approx C^{0.45},\quad D^*(C) \propto C^{\frac{\alpha}{\alpha+\beta}}\approx C^{0.55}}
    \item LoRA Forward \& Exact Gradients:\csfm{\vecb{h} = \mat{W}_0\vecb{x} + \frac{\alpha}{r}\mat{B}\mat{A}\vecb{x},\quad \pd{\mathcal{L}}{\mat{B}} = \frac{\alpha}{r}\pd{\mathcal{L}}{\vecb{h}}(\mat{A}\vecb{x})\T,\quad \pd{\mathcal{L}}{\mat{A}} = \frac{\alpha}{r}\mat{B}\T\pd{\mathcal{L}}{\vecb{h}}\vecb{x}\T}
  \end{csbul}
\end{cscard}

\begin{cscard}{csB}{QLoRA INSTRUCTION SFT RECIPE}
  \begin{enumerate}[leftmargin=1.2em, itemsep=1pt, topsep=1pt, parsep=0pt]
    \item Quantize base weights $\mat{W}_0\to\tilde{\mat{W}}_0$ in 4-bit NF4 + double-quantized FP8 scales.
    \item Attach BF16 $\mat{A}\sim\mathcal{N}(0,\sigma^2)$, $\mat{B}=\mat{0}$ on $\{Q,K,V,O,\mathrm{MLP}\}$ ($r=16,\alpha=32$).
    \item Apply chat template; set target $y_t=-100$ on system/user turns; pack with block-diagonal mask.
    \item Dequantize $\tilde{\mat{W}}_0\to\text{BF16}$ on-the-fly in forward/backward with gradient checkpointing.
    \item Step $(\mat{A},\mat{B})$ via Paged AdamW 8-bit ($\eta \approx 1\text{--}5\times 10^{-5}$).
  \end{enumerate}
\end{cscard}

\begin{cscard}{csG}{PITFALL CHECKLIST}
  \begin{csbul}
    \item \textbf{Asymmetric LoRA init is mandatory:} set $\mat{A}\sim\mathcal{N}(0,\sigma^2)$ and $\mat{B}=\mat{0}$; both zero freezes gradients, both random perturbs $t=0$.
    \item \textbf{Over-scaling $\alpha/r$:} $\alpha/r \gg 2$ overwrites pretrained features; keep $\alpha=2r$ (or rank-stabilized $\alpha/\sqrt{r}$) and SFT LR $10\text{--}100\times$ below pretraining.
    \item \textbf{Target attention + MLP:} applying LoRA to $\mat{W}_Q,\mat{W}_V$ only underperforms all-linear adaptation at the same total parameter count.
    \item \textbf{Packing cross-contamination:} always reset position IDs and apply block-diagonal causal masks between packed SFT examples.
    \item \textbf{Chat template mismatch:} serving with different role delimiters than SFT breaks instruction compliance and EOS stopping.
  \end{csbul}
\end{cscard}

\begin{cscard}{csD}{10-SECOND VISUAL RECAP}
  \sffamily\footnotesize
  Self-supervised CLM minimizes next-token cross-entropy ($\mathrm{PPL}=e^{\mathcal{L}_{\mathrm{CE}}}$) along the \textbf{Chinchilla frontier} ($C\approx 6ND$, $D^*\approx 20N^*$). Post-training masks user prompts with $-100$ and adapts frozen 4-bit \textbf{NF4} weights via trainable low-rank bypasses $\frac{\alpha}{r}\mat{B}\mat{A}\vecb{x}$, slashing 7B fine-tuning VRAM from $112\text{ GB}$ to $9.3\text{ GB}$.
\end{cscard}
