\csbranch{1}{A}{1\;\; $L$-SMOOTHNESS \& DESCENT LEMMA}{%
  \begin{minipage}[t]{0.60\linewidth}
    \csfm{\norm{\grad f(\vecb y)-\grad f(\vecb x)}_2\le L\norm{\vecb y-\vecb x}_2}
    \begin{csbul}
      \item \textbf{Majorant:} $f(\vecb y)\le f(\vecb x)+\inner{\grad f(\vecb x)}{\vecb y-\vecb x}+\tfrac{L}{2}\norm{\vecb y-\vecb x}_2^2$
      \item \textbf{1-step drop:} $f(\vecb x_{t+1})-f(\vecb x_t)\le -\eta(1-\tfrac{\eta L}{2})\norm{\grad f(\vecb x_t)}_2^2$
      \item \textbf{Stability:} $\eta\in(0,2/L)$; $\eta=1/L$ drops $\tfrac{1}{2L}\norm{\grad f}_2^2$
      \item \textbf{Non-convex FOSP:} $\min_{t<T}\norm{\grad f(\vecb x_t)}_2\le\sqrt{2L\Delta_0/T}$
    \end{csbul}
  \end{minipage}\hfill
  \begin{minipage}[t]{0.37\linewidth}\centering\vspace{4pt}
    \begin{tikzpicture}[scale=0.45, >=Stealth, every node/.style={font=\tiny}]

    \end{tikzpicture}
  \end{minipage}}

\csbranch{2}{B}{2\;\; CONVEX $L$-SMOOTH $-$ co-coercivity \& $O(1/T)$}{%
  \csfm{\inner{\grad f(\vecb y)-\grad f(\vecb x)}{\vecb y-\vecb x}\ge\frac{1}{L}\norm{\grad f(\vecb y)-\grad f(\vecb x)}_2^2}
  \begin{csbul}
    \item \textbf{Fej\'er monotonicity ($\eta\le 2/L$):} $\norm{\vecb x_{t+1}-\vecb x^*}_2\le\norm{\vecb x_t-\vecb x^*}_2\le R_0$ (traps trajectory inside $\mathbb B(\vecb x^*,R_0)$)
    \item \textbf{Sublinear gap ($\eta\le 1/L$):} $f(\vecb x_T)-f^*\le\frac{\norm{\vecb x_0-\vecb x^*}_2^2}{2\eta T}=O(1/T)$ via Lyapunov $E_t=\frac{1}{2\eta}\norm{\vecb x_t-\vecb x^*}_2^2+t(f(\vecb x_t)-f^*)$
    \item \textbf{Fast gradient norm:} $\min_{t<T}\norm{\grad f(\vecb x_t)}_2\le\frac{2L R_0}{T}\implies T=O(LR_0/\varepsilon)$
  \end{csbul}}

\csbranch{3}{C}{3\;\; $\mu$-STRONG CONVEXITY $-$ linear contraction}{%
  \begin{minipage}[t]{0.60\linewidth}
    \csfm{\mu\eye_d\preceq\hess f(\vecb x)\preceq L\eye_d,\qquad \kappa=\frac{L}{\mu}\ge 1}
    \begin{csbul}
      \item \textbf{Standard $\eta=1/L$:} $\norm{\vecb x_t-\vecb x^*}_2^2\le(1-\kappa\inv)^t R_0^2$, $T=O(\kappa\log\frac{1}{\varepsilon})$
      \item \textbf{Exact map $\eta\in(0,2/L)$:} $\rho(\eta)=\max\{\abs{1-\eta\mu},\abs{1-\eta L}\}$
      \item \textbf{Optimal $\eta^*=\frac{2}{\mu+L}$:} balances extreme modes to
        $\rho(\eta^*)=\frac{\kappa-1}{\kappa+1}\implies T=\frac{\kappa+1}{2}\log\frac{1}{\varepsilon}$
    \end{csbul}
  \end{minipage}\hfill
  \begin{minipage}[t]{0.37\linewidth}\centering\vspace{4pt}

\csbranch{4}{D}{4\;\; POLYAK--\L{}OJASIEWICZ (PL) \& KL LANDSCAPES}{%
  \csfm{\tfrac{1}{2}\norm{\grad f(\vecb x)}_2^2\ge\mu\bigl(f(\vecb x)-f^*\bigr)\quad(\text{no convexity needed!})}
  \begin{csbul}
    \item \textbf{Hierarchy:} $\mu\text{-SC}\implies\mu\text{-PL}\implies\text{QG: }f(\vecb x)-f^*\ge\tfrac{\mu}{2}\mathrm{dist}(\vecb x,\mathcal X^*)^2$
    \item \textbf{No spurious traps:} $\grad f(\vecb x)=\zeros\iff f(\vecb x)=f^*$ (curved $\mathcal X^*$ OK)
    \item \textbf{Global linear rate ($\eta=1/L$):} $f(\vecb x_T)-f^*\le(1-\mu/L)^T(f(\vecb x_0)-f^*)$
    \item \textbf{Examples:} $n<d$ least squares, overparam.\ NTK nets, $t^2+3\sin^2 t$
    \item \textbf{KL ($\theta\in(1/2,1)$):} $\norm{\grad f}_2\ge c(f-f^*)^\theta \implies O(T^{-1/(2\theta-1)})$
  \end{csbul}}

\csbranch{5}{E}{5\;\; SADDLE ESCAPE $-$ Stable Manifold \& Perturbed GD}{%
  \begin{minipage}[t]{0.60\linewidth}
    \begin{csbul}
      \item \textbf{Strict saddle:} $\grad f(\bar{\vecb x})=\zeros$, $\lambda_{\min}(\hess f(\bar{\vecb x}))<0$
      \item \textbf{Thm (Stable Manifold):} for $\eta<1/L$, $T(\vecb x)=\vecb x-\eta\grad f(\vecb x)$ is a diffeomorphism; stable set has \textbf{measure $0$} ($\Prob(\vecb x_t\to\bar{\vecb x})=0$)
      \item \textbf{Perturbed GD (PGD):} ball kick $\vecb\xi_t\sim\mathrm{Unif}(\mathbb B_0(r))$ when $\norm{\grad f}_2\le\varepsilon$ escapes thin pancake in $\tilde O(L\Delta_0/\varepsilon^2)$ to $(\varepsilon,\sqrt{\rho\varepsilon})$-SOSP
    \end{csbul}

\begin{cscard}{csA}{FORMULAS TO REMEMBER}
  \begin{csbul}
    \item $L$-smooth descent lemma \& gradient upper bound:
      \csfm{f(\vecb x_{t+1})\le f(\vecb x_t)-\eta\!\left(1-\frac{\eta L}{2}\right)\norm{\grad f(\vecb x_t)}_2^2,\quad \frac{\norm{\grad f(\vecb x)}_2^2}{2L}\le f(\vecb x)-f^*}
    \item Nesterov refined co-coercivity ($\mu$-SC + $L$-smooth):
      \csfm{\inner{\grad f(\vecb x)-\grad f(\vecb y)}{\vecb x-\vecb y}\ge\frac{\mu L}{\mu+L}\norm{\vecb x-\vecb y}_2^2+\frac{1}{\mu+L}\norm{\grad f(\vecb x)-\grad f(\vecb y)}_2^2}
    \item Optimal fixed step $\eta^*=\frac{2}{\mu+L}$ linear contraction ($\kappa=L/\mu$):
      \csfm{\norm{\vecb x_T-\vecb x^*}_2\le\left(\frac{\kappa-1}{\kappa+1}\right)^T\norm{\vecb x_0-\vecb x^*}_2}
    \item Polyak--\L{}ojasiewicz ($\mu$-PL) global contraction ($\eta\in(0,2/L)$):
      \csfm{\frac{1}{2}\norm{\grad f(\vecb x)}_2^2\ge\mu\bigl(f(\vecb x)-f^*\bigr)\implies f(\vecb x_T)-f^*\le\bigl(1-2\eta\mu(1-\tfrac{\eta L}{2})\bigr)^T\Delta_0}
  \end{csbul}
\end{cscard}

\begin{cscard}{csB}{RECIPE $-$ ARMIJO-BB \& PERTURBED GD}
  \footnotesize
  \textbf{Safeguarded Barzilai--Borwein + Armijo Backtracking:}
  \begin{enumerate}[leftmargin=1.2em, itemsep=0.5pt, topsep=1pt, parsep=0pt]
    \item If $t\ge 1$ and $\inner{\vecb s_{t-1}}{\vecb y_{t-1}}>0$, set trial $\eta=\mathrm{clip}\bigl(\frac{\norm{\vecb s_{t-1}}_2^2}{\inner{\vecb s_{t-1}}{\vecb y_{t-1}}},\eta_{\min},\eta_{\max}\bigr)$.
    \item While $f(\vecb x_t-\eta\vecb g_t)>f(\vecb x_t)-c_1\eta\norm{\vecb g_t}_2^2$, shrink $\eta\gets\beta\eta$ ($c_1=10^{-4}$).
    \item Update $\vecb x_{t+1}=\vecb x_t-\eta\vecb g_t$, record $\vecb s_t=\vecb x_{t+1}-\vecb x_t$, $\vecb y_t=\vecb g_{t+1}-\vecb g_t$.
  \end{enumerate}
  \vspace{1pt}
  \textbf{Perturbed GD (PGD) for $(\varepsilon,\sqrt{\rho\varepsilon})$-SOSP:}
  \begin{enumerate}[leftmargin=1.2em, itemsep=0.5pt, topsep=1pt, parsep=0pt]
    \item When $\norm{\grad f(\vecb x_t)}_2\le\varepsilon$, add kick $\vecb\xi_t\sim\mathrm{Unif}(\mathbb B_0(r))$, $r=\tilde\Theta(\varepsilon/L)$.
    \item Run $t_{\mathrm{thresh}}=\tilde\Theta(L/\sqrt{\rho\varepsilon})$ GD steps; if drop $<f_{\mathrm{thresh}}$, return $\varepsilon$-SOSP.
  \end{enumerate}
\end{cscard}

\begin{cscard}{csE}{KEY BOUNDS / SCALING CHEAT-LINE}
  \footnotesize
  \begin{csbul}
    \item \textbf{Eigenmode recursion:} on $\frac{1}{2}(\vecb x-\vecb x^*)\T\mat H(\vecb x-\vecb x^*)$, mode $i$ evolves as $\alpha_i^{(t)}=(1-\eta\lambda_i)^t\alpha_i^{(0)}$
    \item \textbf{Conditioning bottleneck:} $\eta=1/L$ kills $\lambda_{\max}=L$ in 1 step, but crawls on $\lambda_{\min}=\mu$ by $1-1/\kappa$
    \item \textbf{Localization:} Fej\'er / SC contraction keeps $\vecb x_t\in\mathbb B(\vecb x^*,R_0)$, so local $L,\mu$ on $\mathbb B(\vecb x^*,R_0)$ suffice!
  \end{csbul}
\end{cscard}

\begin{cscard}{csG}{PITFALL CHECKLIST}
  \begin{csbul}
    \item \textbf{Condition number squaring:} forming $\mat X\T\mat X$ squares $\kappa(\mat X\T\mat X)=\kappa(\mat X)^2$; standardize features to zero mean / unit variance first
    \item \textbf{Local vs.\ global smoothness:} polynomial/quartic losses have $\norm{\hess f(\vecb x)}_2\to\infty$ as $\norm{\vecb x}_2\to\infty$; fixed $\eta$ diverges without warm start or Armijo backtracking
    \item \textbf{Symmetric initialization trap:} initializing on a symmetry subspace (e.g.\ identical weights or $(z,0)$) lands directly on the measure-zero stable manifold of strict saddles
    \item \textbf{Worst-case $O(1/T)$ vs.\ KL:} flat convex bowls like $\theta^4$ decay as $O(t^{-2})$ ($\theta_{\mathrm{KL}}=3/4$), slower than linear $(1-\mu/L)^t$ because $\mu=0$ at the minimum
    \item \textbf{Degenerate saddles:} when $\lambda_{\min}(\hess f(\bar{\vecb x}))=0$, first-order PGD has no negative-curvature escape mode; use cubic-regularized second-order steps
  \end{csbul}
\end{cscard}

\begin{cscard}{csD}{10-SECOND VISUAL RECAP}
  \sffamily\footnotesize
  \textbf{Descent needs an upper parabola} ($L$-smoothness $\implies \eta<2/L$); \textbf{fast convergence needs a lower parabola} ($\mu$-SC or $\mu$-PL $\implies (1-\mu/L)^T$).
  At $\eta^*=\frac{2}{\mu+L}$, steep and flat eigenmodes contract equally at $\frac{\kappa-1}{\kappa+1}$.
  Even without convexity, \textbf{$\mu$-PL} guarantees every stationary point is a global minimum, while \textbf{strict saddles} are escaped almost surely (and in polylog$(d)$ time via ball kicks).
\end{cscard}
