\csbranch{1}{A}{1\;\; WHAT $-$ sets, epigraphs \& convex criteria}{%
  \begin{minipage}[t]{0.61\linewidth}
    \csfm{f(\theta\vecb x+(1-\theta)\vecb y)\le \theta f(\vecb x)+(1-\theta)f(\vecb y)}
    \begin{csbul}
      \item \textbf{Epigraph:} $f$ convex $\iff \operatorname{epi}f=\{(\vecb x,t):f(\vecb x)\le t\}$ is convex set
      \item \textbf{1st-order:} $f(\vecb y)\ge f(\vecb x)+\grad f(\vecb x)\T(\vecb y-\vecb x)$
      \item \textbf{2nd-order:} $\hess f(\vecb x)\succeq\zeros$ ($\succeq m\eye_n$ for $m$-strong)
      \item \textbf{Jensen:} $f(\E[\vecb X])\le\E[f(\vecb X)] \implies \KL{p}{q}\ge 0$
      \item \textbf{Opt:} local $=$ global; $-\grad f_0(\vecb x^*)\in\mathcal N_{\mathcal X}(\vecb x^*)$
    \end{csbul}
  \end{minipage}\hfill

\csbranch{2}{B}{2\;\; FENCHEL CONJUGATE $-$ slope-to-intercept dual}{%
  \csfm{f^*(\vecb y)=\sup_{\vecb x\in\operatorname{dom}f}\bigl(\inner{\vecb y}{\vecb x}-f(\vecb x)\bigr)\quad\text{(always convex!)}}
  \renewcommand{\arraystretch}{1.12}\scriptsize
  \begin{tabular}{@{}>{\raggedright\arraybackslash}p{2.7cm}>{\raggedright\arraybackslash}p{3.6cm}@{}}
    \textbf{\textcolor{csB}{Primal $f(\vecb x)$}} & \textbf{\textcolor{csB}{Conjugate $f^*(\vecb y)$ / Property}}\\ \hline
    Fenchel--Young gap & $f(\vecb x)+f^*(\vecb y)\ge\inner{\vecb y}{\vecb x}$ ($=$ iff $\vecb y\in\partial f(\vecb x)$)\\
    Biconjugate $f^{**}$ & $f^{**}=f$ iff proper, convex, l.s.c.\ (else $\operatorname{conv}\operatorname{epi}f$)\\
    $\frac{\lambda}{2}\vecb y\T\mat K\inv\vecb y$ ($\mat K\succ\zeros$) & $\frac{1}{2\lambda}\vecb\alpha\T\mat K\vecb\alpha$ \quad\textbar\quad $\norm{\vecb x} \leftrightarrow I_{\{\norm{\cdot}_*\le 1\}}(\vecb y)$\\
    $\sum_i x_i\ln x_i$ on $\Simplex^{n-1}$ & $\ln\sum_{i=1}^n e^{y_i}$ (log-sum-exp / Gibbs softmax)\\
  \end{tabular}}

\csbranch{3}{C}{3\;\; LAGRANGIAN DUALITY \& SLATER'S THEOREM}{%
  \begin{minipage}[t]{0.61\linewidth}
    \csfm{L(\vecb x,\vecb\lambda,\vecb\nu)=f_0(\vecb x)+\vecb\lambda\T\vecb f(\vecb x)+\vecb\nu\T\vecb h(\vecb x),\;\;\vecb\lambda\succeq\zeros}
    \begin{csbul}
      \item \textbf{Dual fn:} $g(\vecb\lambda,\vecb\nu)=\inf_{\vecb x\in\mathcal D}L(\vecb x,\vecb\lambda,\vecb\nu)$ (\textbf{always concave})
      \item \textbf{Weak duality:} $d^*=\sup_{\vecb\lambda\succeq\zeros,\vecb\nu}g(\vecb\lambda,\vecb\nu)\le p^*$ always
      \item \textbf{Slater CQ:} convex problem + $\exists\,\tilde{\vecb x}\in\operatorname{relint}\mathcal D$ with $f_i(\tilde{\vecb x})<0$ (non-affine), $\mat A\tilde{\vecb x}=\vecb b \implies d^*=p^*$
      \item \textbf{Saddle:} $L(\vecb x^*,\vecb\lambda,\vecb\nu)\le L(\vecb x^*,\vecb\lambda^*,\vecb\nu^*)\le L(\vecb x,\vecb\lambda^*,\vecb\nu^*)$
    \end{csbul}

\csbranch{6}{F}{6\;\; ML INSTANTIATIONS $-$ where duality shines}{%
  \begin{csbul}
    \item \textbf{Eq.-QP KKT saddle:} $\begin{bmatrix}\mat P & \mat A\T\\ \mat A & \zeros\end{bmatrix}\begin{bmatrix}\vecb x^*\\ \vecb\nu^*\end{bmatrix}=\begin{bmatrix}-\vecb q\\ \vecb b\end{bmatrix}$ ($\mat P\succeq\zeros$)
    \item \textbf{Ineq.-QP / SVM dual:} $\max_{\vecb\lambda\succeq\zeros}-\frac{1}{2}(\vecb c+\mat A\T\vecb\lambda)\T\mat Q\inv(\vecb c+\mat A\T\vecb\lambda)-\vecb\lambda\T\vecb b$
    \item \textbf{Fenchel--Rockafellar:} $\min_{\vecb x}f(\mat A\vecb x)+g(\vecb x)=\max_{\vecb u}[-f^*(\vecb u)-g^*(-\mat A\T\vecb u)]$
    \item \textbf{Geometric Program (GP):} $y_\ell=\ln x_\ell$ turns posynomials into convex log-sum-exp $\ln\sum_k\exp(\vecb a_{ik}\T\vecb y+\ln c_{ik})\le 0$
    \item \textbf{Rayleigh quotient:} $\max_{\norm{\vecb x}_2=1}\vecb x\T\mat A\T\mat A\vecb x \implies \lambda^*=\lambda_{\max}(\mat A\T\mat A)=\norm{\mat A}_2^2$
  \end{csbul}}

\csnote{8}{A}{TRAPS $-$ before you trust the output}{%
  \begin{csbul}
    \item \textbf{Sign flip:} $h_k\ge 0$ uses $-\sum\mu_k h_k$ ($\mu_k\ge 0$); $f_i\le 0$ uses $+\sum\lambda_i f_i$
    \item \textbf{Failed Slater:} $\min e^{-x}$ s.t.\ $x^2/y\le 0$ ($y>0$) has strict gap $p^*-d^*=1$
    \item \textbf{Quasiconvex $\neq$ convex:} convex sublevels $\not\Rightarrow$ convex $\operatorname{epi}f$ (e.g.\ $-e^{x+y}$)
  \end{csbul}}

\begin{cscard}{csA}{FORMULAS TO REMEMBER}
  \begin{csbul}
    \item Fenchel--Legendre conjugate \& Fenchel--Young:
      \csfm{f^*(\vecb y)=\sup_{\vecb x}\bigl(\inner{\vecb y}{\vecb x}-f(\vecb x)\bigr),\quad f(\vecb x)+f^*(\vecb y)\ge\inner{\vecb y}{\vecb x}}
    \item Lagrangian \& concave dual function ($\vecb\lambda\succeq\zeros$):
      \csfm{g(\vecb\lambda,\vecb\nu)=\inf_{\vecb x\in\mathcal D}\Bigl(f_0(\vecb x)+\sum_{i=1}^m\lambda_i f_i(\vecb x)+\sum_{j=1}^p\nu_j h_j(\vecb x)\Bigr)}
    \item KKT system (primal, dual, slackness, stationarity):
      \csfm{f_i(\vecb x^*)\le 0,\;\;\lambda_i^*\ge 0,\;\;\lambda_i^*f_i(\vecb x^*)=0}
      \csfm{\grad f_0(\vecb x^*)+\sum_{i=1}^m\lambda_i^*\grad f_i(\vecb x^*)+\sum_{j=1}^p\nu_j^*\grad h_j(\vecb x^*)=\zeros}
    \item Augmented Lagrangian (Rockafellar--Wets):
      \csfm{\mathcal L_\rho=f_0+\vecb\nu\T(\mat A\vecb x-\vecb b)+\frac{\rho}{2}\norm{\mat A\vecb x-\vecb b}_2^2+\frac{1}{2\rho}\sum_{i=1}^m\bigl((\lambda_i+\rho f_i)_+^2-\lambda_i^2\bigr)}
  \end{csbul}
\end{cscard}

\begin{cscard}{csB}{RECIPE $-$ PRIMAL--DUAL IPM vs.\ ALM}
  \textbf{Path-Following Interior-Point (IPM):}
  \begin{enumerate}[leftmargin=1.2em, itemsep=1pt, topsep=1pt, parsep=0pt]
    \item Start strictly feasible: $\vecb f(\vecb x^{(0)})\prec\zeros$, $\vecb\lambda^{(0)}\succ\zeros$.
    \item Set surrogate gap $\hat\eta=-\vecb f(\vecb x)\T\vecb\lambda$ and barrier $t=\mu m/\hat\eta$.
    \item Solve perturbed KKT Newton system $\mat H_{\mathrm{KKT}}(\Delta\vecb x,\Delta\vecb\lambda,\Delta\vecb\nu)\T=-\vecb r_t$ with $-\diag(\vecb\lambda)\vecb f(\vecb x)=\frac{1}{t}\ones_m$.
    \item Fraction-to-boundary step $\alpha\le 0.99\min_{\Delta\lambda_i<0}(-\lambda_i/\Delta\lambda_i)$ + backtrack.
  \end{enumerate}
  \vspace{2pt}
  \textbf{Method of Multipliers (ALM):}
  \begin{enumerate}[leftmargin=1.2em, itemsep=1pt, topsep=1pt, parsep=0pt]
    \item Inner solve: $\vecb x^{k+1}=\argmin_{\vecb x}\mathcal L_\rho(\vecb x,\vecb\lambda^k,\vecb\nu^k)$.
    \item Dual ascent: $\vecb\nu^{k+1}=\vecb\nu^k+\rho(\mat A\vecb x^{k+1}-\vecb b)$, $\lambda_i^{k+1}=(\lambda_i^k+\rho f_i(\vecb x^{k+1}))_+$.
  \end{enumerate}
\end{cscard}

\begin{cscard}{csE}{KEY BOUNDS \& DUALITY CHEAT-LINE}
  \begin{csbul}
    \item \textbf{Duality gap:} $\Delta_{\mathrm{gap}}=p^*-d^*\ge 0$; any dual-feasible $(\vecb\lambda,\vecb\nu)$ certifies $f_0(\vecb x)-p^*\le f_0(\vecb x)-g(\vecb\lambda,\vecb\nu)$
    \item \textbf{Log-barrier path:} $\vecb x^*(t)$ satisfies $\lambda_i^*(t)=-1/(tf_i(\vecb x^*(t)))$ and $f_0(\vecb x^*(t))-p^*\le m/t$
    \item \textbf{Strong convexity:} $f(\vecb x)-\frac{m}{2}\norm{\vecb x}_2^2$ convex $\iff \hess f(\vecb x)\succeq m\eye_n \iff f^*$ is $(1/m)$-smooth
  \end{csbul}
\end{cscard}

\begin{cscard}{csG}{PITFALL CHECKLIST}
  \begin{csbul}
    \item \textbf{Inequality orientation trap:} before writing $L$, move all inequalities to $f_i(\vecb x)\le 0$ so multipliers stay $\lambda_i\ge 0$ with $+\lambda_i f_i(\vecb x)$
    \item \textbf{Missing KKT multipliers:} if Slater fails (e.g.\ $\min x$ s.t.\ $x^2\le 0$), $\grad f_0(0)+2\lambda\cdot 0=1\neq 0$ has \emph{no} finite KKT multiplier $\lambda^*$
    \item \textbf{Dual kinks \& oscillation:} when $f_0$ is not strictly convex, $g(\vecb\lambda)$ has non-differentiable kinks; use ALM ($\rho>0$) instead of naive dual ascent
    \item \textbf{Quadratic penalty ill-conditioning:} sending $\rho\to\infty$ in $\frac{\rho}{2}\norm{\mat A\vecb x-\vecb b}_2^2$ without multiplier updates $\vecb\nu^{k+1}$ blows up $\kappa(\hess)\propto\rho$
    \item \textbf{Non-convex KKT sufficiency:} on non-convex problems, KKT points can be saddle points or local maxima ($\hess_{\vecb x}L\not\succeq\zeros$)
  \end{csbul}
\end{cscard}

\begin{cscard}{csD}{10-SECOND VISUAL RECAP}
  \sffamily\footnotesize
  \textbf{Convexity} means chords lie above the graph and tangents lie below ($\operatorname{epi}f$ is convex, so every local minimum is global).
  \textbf{Fenchel conjugation} $f^*(\vecb y)$ records the negative intercepts of supporting tangents ($f^{**}=f$).
  \textbf{Lagrange duality} relaxes hard barriers into linear prices $\vecb\lambda\succeq\zeros$: weak duality $d^*\le p^*$ always holds, \textbf{Slater's interior point} guarantees $d^*=p^*$, and \textbf{KKT} states that $-\grad f_0(\vecb x^*)$ is balanced by nonnegative forces from \emph{active} constraints ($\lambda_i^* f_i(\vecb x^*)=0$).
\end{cscard}
