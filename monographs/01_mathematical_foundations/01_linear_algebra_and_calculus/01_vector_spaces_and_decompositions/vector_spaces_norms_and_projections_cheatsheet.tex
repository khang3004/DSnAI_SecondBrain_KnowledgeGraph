\csbranch{2}{B}{2\;\; NORMS \& SPD INNER PRODUCTS}{%
  \begin{minipage}[t]{0.61\linewidth}
    \csfm{\inner{\vecb{x}}{\vecb{y}}_{\mat{M}} = \vecb{x}\T\mat{M}\vecb{y}\;(\mat{M}\succ\zeros)}
    \begin{csbul}
      \item $\ell_p$: $(\sum_i\abs{x_i}^p)^{1/p}$, $\ell_\infty$: $\max_i\abs{x_i}$
      \item \textbf{Chain:} $\norm{\vecb{x}}_\infty\le\norm{\vecb{x}}_2\le\norm{\vecb{x}}_1\le\sqrt{n}\norm{\vecb{x}}_2$
      \item \textbf{CS:} $\abs{\inner{\vecb{u}}{\vecb{v}}}\le\norm{\vecb{u}}\norm{\vecb{v}}$
      \item \textbf{Pythagoras:} $\vecb{u}\perp\vecb{v}\!\iff\!\norm{\vecb{u}+\vecb{v}}^2\!=\!\norm{\vecb{u}}^2+\norm{\vecb{v}}^2$
      \item \textbf{Matrix:} $\norm{\mat{A}}_2=\sigma_1$, $\norm{\mat{A}}_F=\sqrt{\Tr(\mat{A}\T\mat{A})}$
    \end{csbul}
  \end{minipage}\hfill

\csbranch{3}{C}{3\;\; ORTHOGONAL PROJECTOR ALGEBRA}{%
  \csfm{\mat{P}_W = \mat{B}(\mat{B}\T\mat{B})\inv\mat{B}\T = \mat{Q}\mat{Q}\T \quad (\mat{Q}\T\mat{Q}=\eye_r)}
  \begin{csbul}
    \item \textbf{Best $\ell_2$ approx:} $\hat{\vecb{y}} = \mat{P}_W\vecb{y} = \argmin_{\vecb{v}\in W}\norm{\vecb{y}-\vecb{v}}_2$
    \item \textbf{Characterization:} $\mat{P}$ is an orthogonal projector $\iff \mat{P}^2=\mat{P}$ and $\mat{P}\T=\mat{P}$
    \item \textbf{Complement:} $\mat{P}_{W^\perp} = \eye_m - \mat{P}_W$, with $\vecb{y} - \hat{\vecb{y}} \in W^\perp$
    \item \textbf{Spectrum:} eigenvalues $\in\set{0,1}$, $\rank(\mat{P}_W)=\Tr(\mat{P}_W)=r$, $\norm{\mat{P}_W}_2=1$
    \item \textbf{Affine shift:} for $L = \vecb{x}_0 + U$, $\pi_L(\vecb{x}) = \vecb{x}_0 + \mat{P}_U(\vecb{x}-\vecb{x}_0)$
  \end{csbul}}

\csbranch{4}{D}{4\;\; LEAST SQUARES, MIN-NORM \& $\mat{A}\pinv$}{%
  \begin{csbul}
    \item \textbf{Normal equations ($m\ge n$):} $\mat{A}\T\mat{A}\hat{\vecb{x}} = \mat{A}\T\vecb{b} \iff \vecb{b}-\mat{A}\hat{\vecb{x}}\perp\colsp(\mat{A})$
    \item \textbf{Full column rank ($r=n$):} $\hat{\vecb{x}}_{\mathrm{LS}} = (\mat{A}\T\mat{A})\inv\mat{A}\T\vecb{b}$
    \item \textbf{Underdetermined min-norm ($r=m<n$):} $\hat{\vecb{x}}_{\mathrm{LN}} = \mat{A}\T(\mat{A}\mat{A}\T)\inv\vecb{b}\in\mathrm{Row}(\mat{A})$
    \item \textbf{Moore--Penrose $\mat{A}\pinv$:} $\mat{A}\pinv = \mat{V}_r\mat{\Sigma}_r\inv\mat{U}_r\T = \lim_{\alpha\searrow 0}(\mat{A}\T\mat{A}+\alpha\eye)\inv\mat{A}\T$
    \item \textbf{Master property:} $\hat{\vecb{x}}^* = \mat{A}\pinv\vecb{b}$ minimizes $\norm{\vecb{x}}_2$ among all $\argmin_{\vecb{x}}\norm{\mat{A}\vecb{x}-\vecb{b}}_2$
    \item \textbf{Subspace projectors:} $\mat{A}\mat{A}\pinv = \mat{P}_{\colsp(\mat{A})}$, $\mat{A}\pinv\mat{A} = \mat{P}_{\mathrm{Row}(\mat{A})}$
  \end{csbul}}

\csnote{8}{A}{TRAPS $-$ before you trust the output}{%
  \begin{csbul}
    \item forming $\mat{A}\T\mat{A}$ \textbf{squares} $\kappa_2$: $\kappa_2(\mat{A}\T\mat{A}) = \kappa_2(\mat{A})^2$
    \item tall $\hat{\mat{Q}}$ has $\hat{\mat{Q}}\T\hat{\mat{Q}}=\eye_n$, \textbf{not} $\hat{\mat{Q}}\hat{\mat{Q}}\T=\eye_m$ ($=\mat{P}_W$)
    \item Classical GS loses $\perp$ in float; use MGS or Householder
    \item ``$\ell_0$ norm'' $\sum_i\mathbb{I}(x_i\neq 0)$ violates homogeneity $\norm{\alpha\vecb{x}}=\abs{\alpha}\norm{\vecb{x}}$
  \end{csbul}}

\begin{cscard}{csA}{FORMULAS TO REMEMBER}
  \begin{csbul}
    \item Orthogonal projector onto $W=\colsp(\mat{B})$:\csfm{\mat{P}_W = \mat{B}(\mat{B}\T\mat{B})\inv\mat{B}\T = \mat{Q}\mat{Q}\T}
    \item Normal equations \& QR least squares:\csfm{\mat{A}\T\mat{A}\hat{\vecb{x}} = \mat{A}\T\vecb{b} \;\Longleftrightarrow\; \hat{\mat{R}}\hat{\vecb{x}} = \hat{\mat{Q}}\T\vecb{b}}
    \item Underdetermined minimum-norm ($r=m<n$):\csfm{\hat{\vecb{x}}_{\mathrm{LN}} = \mat{A}\T(\mat{A}\mat{A}\T)\inv\vecb{b}}
    \item Moore--Penrose pseudoinverse:\csfm{\mat{A}\pinv = \mat{V}_r\mat{\Sigma}_r\inv\mat{U}_r\T = \sum_{k=1}^r \frac{1}{\sigma_k}\vecb{v}_k\vecb{u}_k\T}
    \item SPD Weighted projector ($\inner{\vecb{u}}{\vecb{v}}_{\mat{M}}=\vecb{u}\T\mat{M}\vecb{v}$):\csfm{\mat{P}_{\mat{M}} = \mat{A}(\mat{A}\T\mat{M}\mat{A})\inv\mat{A}\T\mat{M}}
  \end{csbul}
\end{cscard}

\begin{cscard}{csB}{RECIPE: HOUSEHOLDER QR vs.\ MGS}
  \textbf{Modified Gram--Schmidt (MGS):}
  \begin{enumerate}[leftmargin=1.2em, itemsep=1pt, topsep=1pt, parsep=0pt]
    \item Set $\vecb{v}_j \gets \vecb{a}_j$ for $j=1,\dots,n$.
    \item For $k=1,\dots,n$: $R_{kk}=\norm{\vecb{v}_k}_2$, $\vecb{q}_k=\vecb{v}_k/R_{kk}$.
    \item Immediately update remaining $j>k$: $R_{kj}=\vecb{q}_k\T\vecb{v}_j$, $\vecb{v}_j \gets \vecb{v}_j - R_{kj}\vecb{q}_k$.
  \end{enumerate}
  \vspace{2pt}
  \textbf{Householder Reflection QR Solve:}
  \begin{enumerate}[leftmargin=1.2em, itemsep=1pt, topsep=1pt, parsep=0pt]
    \item For $k=1,\dots,n$, take subcolumn $\vecb{z}=\mat{R}_{k:m,k}$, set $v_1 = z_1 + \sign(z_1)\norm{\vecb{z}}_2$ (avoids cancellation).
    \item Reflect $\mat{H}_k = \eye - \beta\vecb{v}\vecb{v}\T$ ($\beta=2/\norm{\vecb{v}}_2^2$) on $\mat{R}$ and $\vecb{b}$.
    \item Solve triangular $\hat{\mat{R}}\hat{\vecb{x}} = (\hat{\mat{Q}}\T\vecb{b})_{1:n}$ by back-substitution.
  \end{enumerate}
\end{cscard}

\begin{cscard}{csE}{KEY BOUNDS \& SCALING CHEAT-LINE}
  \begin{csbul}
    \item $\norm{\vecb{x}}_\infty \le \norm{\vecb{x}}_2 \le \norm{\vecb{x}}_1 \le \sqrt{n}\norm{\vecb{x}}_2 \le n\norm{\vecb{x}}_\infty$
    \item $\norm{\mat{A}}_2 = \sigma_{\max}(\mat{A}) \le \norm{\mat{A}}_F = (\sum_{k=1}^r\sigma_k^2)^{1/2} \le \sqrt{r}\norm{\mat{A}}_2$
    \item $\kappa_2(\mat{A}) = \sigma_{\max}/\sigma_{\min}$,\quad $\kappa_2(\mat{A}\T\mat{A}) = \kappa_2(\mat{A})^2$
    \item Pythagoras error: $\norm{\vecb{b}-\mat{A}\vecb{x}}_2^2 = \norm{\vecb{b}-\mat{A}\hat{\vecb{x}}}_2^2 + \norm{\mat{A}(\hat{\vecb{x}}-\vecb{x})}_2^2$
  \end{csbul}
\end{cscard}

\begin{cscard}{csG}{PITFALL CHECKLIST}
  \begin{csbul}
    \item \textbf{L\"auchli catastrophic squaring:} for $\varepsilon = 10^{-8}$, $\mathrm{fl}(1+\varepsilon^2)=1$ makes $\mat{A}\T\mat{A}$ singular in float64 while Householder QR/SVD succeed up to $\kappa_2\approx 10^{16}$
    \item \textbf{Never invert matrices explicitly:} replace $(\mat{A}\T\mat{A})\inv\mat{A}\T\vecb{b}$ or $\mat{R}\inv$ with triangular solves (\texttt{scipy.linalg.lstsq})
    \item \textbf{Oblique vs.\ orthogonal projector:} $\mat{P}^2=\mat{P}$ alone does \emph{not} minimize $\ell_2$ distance; symmetry $\mat{P}\T=\mat{P}$ is mandatory
    \item \textbf{Tiny singular values $\sigma_k\approx 0$:} $\mat{A}\pinv$ amplifies measurement noise by $1/\sigma_k$; truncate below $\tau = \max(m,n)\epsilon_{\mathrm{mach}}\sigma_1$ or add Tikhonov ridge $\alpha\eye$
    \item \textbf{Non-Euclidean inner product $\mat{M}\succ\zeros$:} standard $\mat{A}(\mat{A}\T\mat{A})\inv\mat{A}\T$ is not $\mat{M}$-orthogonal; must insert $\mat{M}$ into normal equations $\mat{A}\T\mat{M}\mat{A}\hat{\vecb{x}}=\mat{A}\T\mat{M}\vecb{b}$
  \end{csbul}
\end{cscard}

\begin{cscard}{csD}{10-SECOND VISUAL RECAP}
  \sffamily\footnotesize
  \textbf{Split space at right angles:} $\mat{A}$ maps $\mathrm{Row}(\mat{A})$ bijectively onto $\colsp(\mat{A})$ and crushes $\nullsp(\mat{A})$ to $\zeros$.
  When $\mat{A}\vecb{x}=\vecb{b}$ has no exact solution, drop a perpendicular $\hat{\vecb{b}}=\mat{P}_{\colsp(\mat{A})}\vecb{b}$ so the residual $\vecb{e}=\vecb{b}-\hat{\vecb{b}}$ lies in $\nullsp(\mat{A}\T)$, and solve via \textbf{QR} or \textbf{SVD}.
\end{cscard}
