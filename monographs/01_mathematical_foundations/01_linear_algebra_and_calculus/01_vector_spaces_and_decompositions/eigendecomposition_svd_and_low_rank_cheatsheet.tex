\csbranch{1}{A}{1\;\; WHAT $-$ the three-move machine}{%
  \begin{minipage}[t]{0.60\linewidth}
    \csfm{\mat A=\mat U\,\mat\Sigma\,\mat V\T}
    \begin{csbul}
      \item $\mat V\T$: rotate the \emph{input} basis
      \item $\mat\Sigma$: stretch axis $i$ by $\sigma_i\ge 0$, sorted $\sigma_1\ge\dots\ge\sigma_r>0$
      \item $\mat U$: rotate into the \emph{output} space
      \item exists for \textbf{every} $m\times n$ matrix
    \end{csbul}

\csbranch{2}{B}{2\;\; EVD vs SVD $-$ which one?}{%
  \renewcommand{\arraystretch}{1.15}\scriptsize
  \begin{tabular}{@{}>{\raggedright\arraybackslash}p{1.1cm}>{\raggedright\arraybackslash}p{2.6cm}>{\raggedright\arraybackslash}p{2.6cm}@{}}
    & \textbf{\textcolor{csB}{EVD}} & \textbf{\textcolor{csB}{SVD}}\\ \hline
    shape & square only & any $m\times n$\\
    form & $\mat P\mat\Lambda\mat P\inv$ (may be complex / defective) & $\mat U\mat\Sigma\mat V\T$, always real\\
    sym. & $\mat Q\mat\Lambda\mat Q\T$, $\mat Q$ orthogonal & $\sigma_i=|\lambda_i|$\\
    link & $\sigma_i=\sqrt{\lambda_i(\mat A\T\mat A)}$ & $\mat A\T\mat A=\mat V\mat\Sigma^2\mat V\T$\\
  \end{tabular}\par\vspace{2pt}
  \small\textbf{Rule:} symmetric $\Rightarrow$ EVD is safe; otherwise reach for SVD.}

\csbranch{3}{C}{3\;\; KEY THEOREM $-$ Eckart--Young--Mirsky}{%
  \begin{minipage}[t]{0.60\linewidth}
    \csfm{\mat A_k=\sum_{i=1}^{k}\sigma_i\vecb{u}_i\vecb{v}_i\T}
    \begin{csbul}
      \item best rank-$k$ approximation (spectral \& Frobenius)
      \item $\norm{\mat A-\mat A_k}_2=\sigma_{k+1}$
      \item $\norm{\mat A-\mat A_k}_F=\bigl(\sum_{i>k}\sigma_i^2\bigr)^{1/2}$
    \end{csbul}
  \end{minipage}\hfill
  \begin{minipage}[t]{0.37\linewidth}\centering\vspace{2pt}

\csbranch{4}{D}{4\;\; CONNECTIONS $-$ one object, many names}{%
  \begin{csbul}
    \item \textbf{PCA} = SVD of the \emph{centred} data matrix
    \item \textbf{Pseudoinverse:} $\mat A\pinv=\mat V_r\mat\Sigma_r\inv\mat U_r\T$
    \item \textbf{Rank} $r$ = number of $\sigma_i>0$ (not pivot count)
    \item \textbf{Conditioning:} $\kappa_2(\mat A)=\sigma_1/\sigma_r$
    \item \textbf{Rayleigh:} $\lambda_1=\max_{\vecb x}\frac{\vecb x\T\mat A\vecb x}{\vecb x\T\vecb x}$ (Courant--Fischer)
    \item $\mat U_r$ spans $\mathrm{col}(\mat A)$, $\mat V_r$ spans $\mathrm{row}(\mat A)$
  \end{csbul}}

\csbranch{5}{E}{5\;\; COST $-$ which solver?}{%
  \renewcommand{\arraystretch}{1.2}\scriptsize
  \begin{tabular}{@{}>{\raggedright\arraybackslash}p{1.7cm}>{\raggedright\arraybackslash}p{2.9cm}>{\raggedright\arraybackslash}p{2.2cm}@{}}
    \textbf{\textcolor{csE}{Solver}} & \textbf{\textcolor{csE}{Time}} & \textbf{\textcolor{csE}{Use}}\\ \hline
    Full / thin SVD & $O(\min(mn^2,m^2n))$ & small--medium dense\\
    Power + deflation & $O(Tk\,\mathrm{nnz})$ & sparse top-$k$\\
    Randomized SVD & $O((q{+}1)\,\mathrm{nnz}\,(k{+}p)+(k{+}p)^2n)$ & huge matrices, GPU PCA\\
  \end{tabular}\par\vspace{3pt}
  \small Storage of rank-$k$ factors: $k(m+n+1)$ vs.\ $mn$.
  \textbf{Randomized:} sketch $\mat Y=\mat A\mat\Omega$, QR, project, small SVD.}

\csnote{8}{A}{TRAPS $-$ before you trust the output}{%
  \begin{csbul}
    \item never form $\mat A\T\mat A$: it \textbf{squares} $\kappa$ ($\kappa^2$)
    \item $(\vecb u_i,\vecb v_i)\!\to\!(-\vecb u_i,-\vecb v_i)$: signs are arbitrary
    \item API returns $\mat V\T$, not $\mat V$
    \item tied $\sigma_i$: vectors unique only up to rotation
  \end{csbul}}

\begin{cscard}{csA}{FORMULAS TO REMEMBER}
  \begin{csbul}
    \item Symmetric EVD:\csfm{\mat A=\mat Q\mat\Lambda\mat Q\T,\quad \mat Q\T\mat Q=\mat I}
    \item Full SVD:\csfm{\mat A=\mat U\mat\Sigma\mat V\T,\quad \mat A\vecb v_i=\sigma_i\vecb u_i}
    \item Compact SVD:\csfm{\mat A=\mat U_r\mat\Sigma_r\mat V_r\T,\quad r=\rank(\mat A)}
    \item Pseudoinverse:\csfm{\mat A\pinv=\mat V_r\mat\Sigma_r\inv\mat U_r\T}
    \item Best rank-$k$:\csfm{\mat A_k=\mat U_k\mat\Sigma_k\mat V_k\T}
  \end{csbul}
\end{cscard}

\begin{cscard}{csB}{HOW TO COMPUTE IT BY HAND}
  \begin{enumerate}[leftmargin=1.2em, itemsep=1pt, topsep=1pt, parsep=0pt]
    \item eigen-decompose $\mat A\T\mat A=\mat V\mat\Sigma^2\mat V\T$ (teaching only!)
    \item $\sigma_i=\sqrt{\lambda_i}$, sort descending
    \item $\vecb u_i=\mat A\vecb v_i/\sigma_i$ for $\sigma_i>0$
    \item complete $\mat U$ to an orthonormal basis if full SVD is required
  \end{enumerate}
\end{cscard}
\columnbreak

\begin{cscard}{csC}{DECISION GUIDE}
  \renewcommand{\arraystretch}{1.3}\small
  \begin{tabular}{@{}>{\raggedright\arraybackslash}p{3.3cm}>{\raggedright\arraybackslash}p{4.7cm}@{}}
    \textbf{If you have\ldots} & \textbf{use\ldots}\\ \hline
    symmetric matrix & EVD ($\mat Q\mat\Lambda\mat Q\T$)\\
    rectangular / rank-deficient & SVD\\
    need a few top directions & randomized / power method\\
    noisy inverse problem & truncated SVD, not $\mat A\inv$\\
    rank decision & count $\sigma_i>\tau$, never pivots\\
  \end{tabular}
\end{cscard}

\begin{cscard}{csE}{ERROR CHEAT-LINE}
  \begin{csbul}
    \item $\norm{\mat A-\mat A_k}_2=\sigma_{k+1}$
    \item $\norm{\mat A-\mat A_k}_F^2=\sum_{i>k}\sigma_i^2$
    \item noise amplification of $\mat A\pinv$: $\le\delta/\sigma_r$
    \item pick $k$ where the scree plot bends (energy $\sum_{i\le k}\sigma_i^2/\sum_i\sigma_i^2$)
  \end{csbul}
\end{cscard}

\begin{cscard}{csG}{PITFALL CHECKLIST}
  \begin{csbul}
    \item $\kappa_2(\mat A\T\mat A)=\kappa_2(\mat A)^2$ $\to$ squares the damage; use a direct SVD routine
    \item pseudoinverse on ill-conditioned $\mat A$ explodes noise $\to$ truncate small $\sigma_i$
    \item defective / non-symmetric $\mat A$: EVD may be complex or non-diagonalizable
    \item signs and tied singular values are not unique
    \item library returns $\mat V\T$ (check your API)
  \end{csbul}
\end{cscard}

\begin{cscard}{csD}{10-SECOND RECAP}
  \sffamily\footnotesize
  \textbf{Any matrix} \,=\, \textbf{rotate} $\to$ \textbf{stretch} $\to$ \textbf{rotate}.
  Stretches are $\sigma_i$. Keep the top $k$ for the \emph{provably} best rank-$k$ copy.
  PCA, pseudoinverse, conditioning and compression are all the same factorization seen from different sides.
\end{cscard}
