\csbranch{1}{A}{1\;\; FR\'{E}CHET, JACOBIANS \& LAYOUTS}{%
  \csfm{\vecb{f}(\vecb{x}+\vecb{h}) = \vecb{f}(\vecb{x}) + \jac_{\vecb{f}}(\vecb{x})\vecb{h} + o(\norm{\vecb{h}}_2)}
  \begin{csbul}
    \item \textbf{Numerator Jacobian:} $\jac_{\vecb{f}}(\vecb{x}) = \pd{\vecb{f}}{\vecb{x}\T} \in \R^{m\times n}$ ($(\jac_{\vecb{f}})_{ij} = \pd{f_i}{x_j}$)
    \item \textbf{Shape-preserving gradient:} $\grad_{\vecb{x}} f = \jac_f(\vecb{x})\T \in \R^n$, $\grad_{\mat{X}} f \in \R^{m\times n}$
    \item \textbf{Chain rule:} $\jac_{\vecb{g}\circ\vecb{f}}(\vecb{x}) = \jac_{\vecb{g}}(\vecb{f}(\vecb{x}))\,\jac_{\vecb{f}}(\vecb{x}) \in \R^{m\times n}$
    \item \textbf{Scalar pullback:} $\grad_{\vecb{x}}(g\circ\vecb{f}) = \jac_{\vecb{f}}(\vecb{x})\T \grad_{\vecb{y}} g(\vecb{y}) \in \R^n$
    \item \textbf{Kronecker vec:} $\mathrm{vec}(\mat{A}\mat{X}\mat{B}) = (\mat{B}\T\otimes\mat{A})\mathrm{vec}(\mat{X})$
  \end{csbul}}

\csbranch{2}{B}{2\;\; FROBENIUS TRACE TRICK ($\mathrm{d}f$)}{%
  \csfm{\mathrm{d}f(\mat{X}) = \Tr\!\big((\grad_{\mat{X}} f)\T \mathrm{d}\mat{X}\big) = \inner{\grad_{\mat{X}} f}{\mathrm{d}\mat{X}}_F}
  \begin{csbul}
    \item \textbf{Recipe:} expand $\mathrm{d}f$ via product rule \& cyclic $\Tr(\mat{A}\mat{B})=\Tr(\mat{B}\mat{A})$ to isolate $\mathrm{d}\mat{X}$
    \item $\grad_{\mat{X}}\Tr(\mat{A}\mat{X}\mat{B}) = \mat{A}\T\mat{B}\T$,\quad $\grad_{\mat{X}}(\vecb{a}\T\mat{X}\vecb{b}) = \vecb{a}\vecb{b}\T$
    \item $\grad_{\mat{X}}\Tr(\mat{X}\T\mat{A}\mat{X}) = (\mat{A}+\mat{A}\T)\mat{X}$
    \item $\mathrm{d}(\mat{X}\inv) = -\mat{X}\inv\mathrm{d}\mat{X}\,\mat{X}\inv \implies \grad_{\mat{X}}\log\abs{\det\mat{X}} = \mat{X}^{-\top}$
    \item \textbf{Matmul VJP ($\mat{C}=\mat{A}\mat{B}$):} $\grad_{\mat{A}} L = \mat{G}\mat{B}\T$,\; $\grad_{\mat{B}} L = \mat{A}\T\mat{G}$
  \end{csbul}}

\csbranch{3}{C}{3\;\; FORWARD JVP vs.\ REVERSE VJP}{%
  \begin{minipage}[t]{0.63\linewidth}
    \csfm{\jac_T(\cdots(\jac_1\vecb{v})) \;\text{vs.}\; \jac_1\T(\cdots(\jac_T\T\vecb{u}))}
    \begin{csbul}
      \item \textbf{Forward JVP} ($\dot{\vecb{x}}=\vecb{v}$): dual numbers $x+\dot{x}\varepsilon$, $O(1)$ mem, 1 col of $\jac$ per pass
      \item \textbf{Reverse VJP} ($\bar{\vecb{y}}=\vecb{u}$): caches tape, 1 row of $\jac$ per sweep; $O(1)$ sweeps for scalar $L$!
      \item \textbf{Fan-out:} shared $v_q$ sums child adjoints $\bar{\vecb{x}}_q = \sum_{c\in\mathrm{Ch}(q)} (\partial_q\vecb{f}_c)\T[\bar{\vecb{x}}_c]$
    \end{csbul}
  \end{minipage}\hfill

\csbranch{4}{D}{4\;\; HESSIAN-VECTOR PRODUCTS \& IFT}{%
  \begin{csbul}
    \item \textbf{Pearlmutter HVP ($\mathcal{F}\circ\mathcal{R}$):} exact $\mat{H}\vecb{v}$ in $O(T)$ time \& memory:
    \csfm{\hess f(\vecb{x})\vecb{v} = \grad_{\vecb{x}}\!\Big(\grad_{\vecb{x}} f(\vecb{x})\T\mathrm{stopgrad}(\vecb{v})\Big)}
    \item \textbf{Implicit Function Theorem (IFT)} for $\vecb{g}(\vecb{x},\vecb{y}^*(\vecb{x}))=\zeros$:
    \csfm{\jac_{\vecb{y}^*} = -(\partial_{\vecb{y}}\vecb{g})\inv\partial_{\vecb{x}}\vecb{g},\quad (\partial_{\vecb{y}}\vecb{g})\T\vecb{\lambda}=\vecb{u} \implies \grad_{\vecb{x}} L = -(\partial_{\vecb{x}}\vecb{g})\T\vecb{\lambda}}
    \item Solves equilibrium / OptNet layers in $O(m+n)$ memory ($K$-independent)!
  \end{csbul}}

\csnote{8}{A}{TRAPS $-$ before you trust the output}{%
  \begin{csbul}
    \item \textbf{Missing \texttt{stop\_gradient}($\vecb{v}$) in HVP} leaks $\jac_{\vecb{v}}\T\grad f$ when $\vecb{v}=\vecb{v}(\vecb{x})$
    \item \textbf{In-place tensor mutation} corrupts cached forward activations for VJP
    \item \textbf{Float32 FD} at $\epsilon=10^{-6}$ fails from cancellation; use float64 central FD
    \item \textbf{ReLU kinks:} $\ReLU(x)-\ReLU(-x)=x$ gives AD $0$ at $x=0$ if $\ReLU'(0)=0$
  \end{csbul}}

\begin{cscard}{csA}{FORMULAS TO REMEMBER}
  \begin{csbul}
    \item Frobenius trace trick:\csfm{\mathrm{d}f = \Tr(\mat{G}\T\mathrm{d}\mat{X}) \;\Longrightarrow\; \grad_{\mat{X}} f(\mat{X}) = \mat{G}}
    \item Linear \& quadratic matrix forms:\csfm{\grad_{\mat{X}}\Tr(\mat{A}\mat{X}\mat{B}) = \mat{A}\T\mat{B}\T,\quad \grad_{\mat{X}}\Tr(\mat{X}\T\mat{A}\mat{X}) = (\mat{A}+\mat{A}\T)\mat{X}}
    \item Inverse \& log-determinant ($\mat{X}$ invertible):\csfm{\grad_{\mat{X}}\Tr(\mat{X}\inv\mat{A}) = -\mat{X}^{-\top}\mat{A}\T\mat{X}^{-\top},\quad \grad_{\mat{X}}\log\abs{\det\mat{X}} = \mat{X}^{-\top}}
    \item Matmul VJP ($\mat{C}=\mat{A}\mat{B}$, $\mat{G}=\grad_{\mat{C}} L$):\csfm{\grad_{\mat{A}} L = \mat{G}\mat{B}\T,\qquad \grad_{\mat{B}} L = \mat{A}\T\mat{G}}
    \item Implicit layer VJP ($\vecb{g}(\vecb{x},\vecb{y}^*)=\zeros$, $\vecb{u}=\grad_{\vecb{y}^*} L$):\csfm{\left(\pd{\vecb{g}}{\vecb{y}\T}\right)\T\vecb{\lambda} = \vecb{u},\qquad \grad_{\vecb{x}} L = -\left(\pd{\vecb{g}}{\vecb{x}\T}\right)\T\vecb{\lambda}}
  \end{csbul}
\end{cscard}

\begin{cscard}{csB}{RECIPE: TRACE TRICK \& REVERSE AD}
  \textbf{Matrix Gradient via Trace Trick:}
  \begin{enumerate}[leftmargin=1.2em, itemsep=1pt, topsep=1pt, parsep=0pt]
    \item Expand $\mathrm{d}f$ via $\mathrm{d}(\mat{U}\mat{V})=(\mathrm{d}\mat{U})\mat{V}+\mat{U}(\mathrm{d}\mat{V})$ and $\mathrm{d}(\mat{X}\inv)=-\mat{X}\inv(\mathrm{d}\mat{X})\mat{X}\inv$.
    \item Apply $\Tr(\mat{M})=\Tr(\mat{M}\T)$ and cyclic shifts $\Tr(\mat{A}\mat{B}\mat{C})=\Tr(\mat{C}\mat{A}\mat{B})$ until $\mathrm{d}f=\Tr(\mat{G}\T\mathrm{d}\mat{X})$, then read off $\grad_{\mat{X}} f = \mat{G}$.
  \end{enumerate}
  \vspace{1pt}
  \textbf{Topological Reverse-Mode Tape Sweep:}
  \begin{enumerate}[leftmargin=1.2em, itemsep=1pt, topsep=1pt, parsep=0pt]
    \item Forward $t=1\dots T$: evaluate $\vecb{x}_t = f_t(\vecb{x}_{\mathrm{Pa}(t)})$, push to tape $\mathcal{T}$.
    \item Seed $\bar{\vecb{x}}_T = \vecb{u}$ ($\bar{x}_T = 1$ for scalar loss $L$).
    \item Reverse $t=T\dots 1$: $\bar{\vecb{x}}_q \mathrel{+}= (\partial_q f_t)\T[\bar{\vecb{x}}_t]$ for $q\in\mathrm{Pa}(t)$; free $\vecb{x}_t$.
  \end{enumerate}
\end{cscard}

\begin{cscard}{csC}{DECISION GUIDE}

  \begin{tabular}{@{}>{\raggedright\arraybackslash}p{3.1cm}>{\raggedright\arraybackslash}p{4.6cm}@{}}
    \textbf{If you have\ldots} & \textbf{use\ldots}\\ \hline
    $n_{\mathrm{in}} \gg n_{\mathrm{out}}$ (scalar loss) & Reverse-mode AD (VJP)\\

  \end{tabular}

\begin{cscard}{csE}{KEY BOUNDS \& SCALING CHEAT-LINE}
  \begin{csbul}
    \item Full $\jac\in\R^{n_{\mathrm{out}}\times n_{\mathrm{in}}}$: JVP takes $n_{\mathrm{in}}$ passes; VJP takes $n_{\mathrm{out}}$ sweeps
    \item Central FD optimal step: $\epsilon^*_{\mathrm{cen}} \sim \varepsilon_{\mathrm{mach}}^{1/3} \approx 6\times 10^{-6}$ (float64)
    \item Forward FD optimal step: $\epsilon^*_{\mathrm{fwd}} \sim \varepsilon_{\mathrm{mach}}^{1/2} \approx 1.5\times 10^{-8}$ (float64)
    \item $\sqrt{L}$-checkpointing: $O(\sqrt{L})$ memory for $+1$ forward pass
  \end{csbul}
\end{cscard}

\begin{cscard}{csG}{PITFALL CHECKLIST}
  \begin{csbul}
    \item \textbf{Numerator vs.\ Denominator transpose bug:} mixing textbooks without checking outer dims $(n\times m)(m\times 1)=(n\times 1)$ silently transposes Jacobians
    \item \textbf{HVP tangent leak:} computing $\grad_{\vecb{x}}(\grad f(\vecb{x})\T\vecb{v})$ when $\vecb{v}=\vecb{v}(\vecb{x})$ without \texttt{stop\_gradient}($\vecb{v}$) adds spurious $\jac_{\vecb{v}}\T\grad f$
    \item \textbf{In-place buffer overwrite:} mutating $\vecb{x}_t$ in-place during the forward pass corrupts backward VJP adjoints $\partial f_c/\partial \vecb{x}_t$
    \item \textbf{Subdifferential chain-rule failure:} at non-smooth kinks ($\ReLU$, $\abs{x}$), Clarke subdifferentials violate strict chain rules; use $\softplus$/$\mathrm{GELU}$ for second-order or IFT steps
    \item \textbf{Vanishing/exploding product $\prod_{l=1}^L \jac_l$:} deep Jacobians with $\rho(\jac_l)\neq 1$ blow up or vanish exponentially; use residuals ($\eye+\jac_{\mathrm{res}}$) and norm clipping
  \end{csbul}
\end{cscard}

\begin{cscard}{csD}{10-SECOND VISUAL RECAP}
  \sffamily\footnotesize
  \textbf{Derivatives are linear maps:} JVP pushes tangents forward ($\jac\vecb{v}$, good for few inputs); VJP pulls adjoints backward ($\jac\T\vecb{u}$, good for scalar losses).
  For matrix formulas, take $\mathrm{d}f = \Tr(\mat{G}\T\mathrm{d}\mat{X})$ and read off $\grad_{\mat{X}} f = \mat{G}$.
\end{cscard}
