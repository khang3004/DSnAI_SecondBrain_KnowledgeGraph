\csbranch{2}{B}{2\;\; KL \& CROSS-ENTROPY $-$ M- vs.\ I-projection}{%
  \begin{minipage}[t]{0.59\linewidth}

\csbranch{3}{C}{3\;\; MUTUAL INFO, DPI \& FANO $-$ channel limits}{%
  \csfm{\MI(X;Y)=\KL{p(x,y)}{p(x)p(y)}=\Entropy(X)-\Entropy(X|Y)\ge 0}
  \begin{csbul}
    \item \textbf{Bivariate Gaussian}: $\MI(X;Y)=-\frac{1}{2}\ln(1-\rho^2)$
    \item \textbf{Data Processing Inequality (DPI)}: $X\to Y\to Z \implies \MI(X;Y)\ge\MI(X;Z)$ (equality iff $Z$ is a sufficient statistic: $\MI(X;Y\mid Z)=0$)
    \item \textbf{$f$-Divergence contraction}: Markov kernel $t(y|x)$ gives $D_f(p(x)\|q(x))\ge D_f(p(y)\|q(y))$
    \item \textbf{Fano's inequality}: $P_e=\Prob(\hat Y\ne Y) \ge \frac{\Entropy(Y|X)-\ln 2}{\ln\abs{\mathcal Y}} = \frac{\Entropy(Y)-\MI(X;Y)-\ln 2}{\ln\abs{\mathcal Y}}$
  \end{csbul}}

\csbranch{4}{D}{4\;\; $f$-DIVERGENCES \& PINSKER $-$ ratio geometry}{%
  \renewcommand{\arraystretch}{0.94}\scriptsize
  \begin{tabular}{@{}>{\raggedright\arraybackslash}p{1.9cm}>{\raggedright\arraybackslash}p{2.5cm}>{\raggedright\arraybackslash}p{2.1cm}@{}}
    \textbf{\textcolor{csD}{Divergence}} & \textbf{\textcolor{csD}{Canonical $\tilde f(u)$}} & \textbf{\textcolor{csD}{Range / Note}}\\ \hline
    Fwd KL $\KL{p}{q}$ & $u\ln u - u + 1$ & $[0,\infty]$, Bregman\\
    Rev KL $\KL{q}{p}$ & $-\ln u + u - 1$ & $[0,\infty]$, VI/ELBO\\
    Jensen--Shannon & $\frac12[u\ln u-(u{+}1)\ln\frac{u+1}{2}]$ & $[0,\ln 2]$, $\sqrt{\mathrm{JS}}$ metric\\
    Sq.\ Hellinger $D_H^2$ & $\frac12(\sqrt u-1)^2$ & $[0,1]$, symmetric\\
    Total Var.\ $D_{\mathrm{TV}}$ & $\frac12\abs{u-1}$ & $[0,1]$, $f$-div $\cap$ IPM\\
  \end{tabular}\par\vspace{0pt}
  \scriptsize\textbf{Pinsker:} $D_{\mathrm{TV}}\le\sqrt{\tfrac12\KL{p}{q}}$;\;
  \textbf{Local:} $D_f(p_\theta\|p_{\theta+\delta})=\frac{f''(1)}{2}\delta\T\mat F\delta+O(\norm{\delta}^3)$.}

\csbranch{5}{E}{5\;\; INFORMATION BOTTLENECK $-$ IB, VIB \& CEB}{%

      \item \textbf{CEB}: $\MI(X;T\mid Y)=\MI(X;T)-\MI(T;Y)$
    \end{csbul}

\csbranch{6}{F}{6\;\; ML CONNECTIONS $-$ one variational root}{%
  \begin{csbul}
    \item \textbf{Classifier Cross-Entropy} = MLE = minimizing forward $\KL{p_{\mathcal D}}{q_\theta}$
    \item \textbf{$\beta$-VAE \& Rate--Distortion}: $h(X)-D \le \MI(X;Z) \le R$; $\beta=1$ recovers exact ELBO
    \item \textbf{$f$-GANs \& Donsker--Varadhan}: $D_f(p\|q)=\sup_t \{\E_p[t(X)]-\E_q[f^\dagger(t(X))]\}$
    \item \textbf{InfoNCE / Contrastive}: $\MI(X;Y)\ge \MI_{\mathrm{NCE}}$, saturated by $\ln K$ (batch negatives $K$)
    \item \textbf{Natural Gradient}: $\KL{p_\theta}{p_{\theta+\delta}}\approx\frac12\delta\T\mat F(\theta)\delta$ defines Fisher--Rao metric
    \item \textbf{Exp.\ family}: $\KL{p_{\eta_1}}{p_{\eta_2}} = B_A(\eta_2\|\eta_1)$ (unique $f$-div $\cap$ Bregman divergence)
  \end{csbul}}

\begin{cscard}{csA}{FORMULAS TO REMEMBER}
  \begin{csbul}
    \item Entropy, Cross-Entropy \& KL Decomposition:
      \csfm{H_{\mathrm{ce}}(p,q)=-\E_p[\ln q(X)]=\Entropy(p)+\KL{p}{q},\quad \KL{p}{q}=\E_p[\ln(p/q)]\ge 0}
    \item Gaussian $h(p)=\frac12\ln\det(2\pi e\mat\Sigma_1)$ \& KL ($\Delta\vecb\mu=\vecb\mu_2-\vecb\mu_1$):
      \csfm{\KL{p}{q}=\frac12\Bigl[\Tr(\mat\Sigma_2\inv\mat\Sigma_1)+\Delta\vecb\mu\T\mat\Sigma_2\inv\Delta\vecb\mu-d+\ln(\det\mat\Sigma_2/\det\mat\Sigma_1)\Bigr]}
    \item Csisz\'ar $f$-Divergence \& Fenchel Dual ($f$ convex, $f(1)=0$, $t^\star=f'(p/q)$):
      \csfm{D_f(p\|q)=\E_q[f(p(X)/q(X))]=\sup_{t}\bigl\{\E_p[t(X)]-\E_q[f^\dagger(t(X))]\bigr\}}
    \item Pinsker's Inequality \& Local Fisher--Rao Expansion:
      \csfm{D_{\mathrm{TV}}(p,q)\le\sqrt{\tfrac12\KL{p}{q}},\qquad D_f(p_\theta\|p_{\theta+\delta})=\frac{f''(1)}{2}\delta\T\mat F(\theta)\delta+O(\norm{\delta}^3)}
    \item Variational Information Bottleneck (VIB) Loss:
      \csfm{\mathcal L_{\mathrm{VIB}}(\theta,\phi)=-\E_{p_{\mathcal D}(x,y)e_\theta(t|x)}[\ln d_\phi(y|t)]+\beta\,\E_{p_{\mathcal D}(x)}\bigl[\KL{e_\theta(t|x)}{m(t)}\bigr]}
  \end{csbul}
\end{cscard}

\begin{cscard}{csB}{BLAHUT--ARIMOTO IB ALGORITHM (DISCRETE)}
  \begin{enumerate}[leftmargin=1.2em, itemsep=1pt, topsep=1pt, parsep=0pt]
    \item Initialize $e^{(0)}(t|x)\in\Delta^{\abs{\mathcal T}-1}$; compute $p(x)$ and $p(y|x)$.
    \item Marginal: $p^{(k)}(t)\leftarrow\sum_x p(x)e^{(k)}(t|x)$.
    \item Decoder: $p^{(k)}(y|t)\leftarrow\frac{1}{p^{(k)}(t)}\sum_x p(x)e^{(k)}(t|x)p(y|x)$.
    \item Encoder: $e^{(k+1)}(t|x)\leftarrow\frac{p^{(k)}(t)}{Z(x,\beta)}e^{-\beta\inv\KL{p(y|x)}{p^{(k)}(y|t)}}$.
    \item Repeat until $\max_{x,t}\abs{e^{(k+1)}(t|x)-e^{(k)}(t|x)}<\varepsilon$.
  \end{enumerate}
\end{cscard}

\begin{cscard}{csE}{KEY IDENTITIES \& BOUNDS CHEAT-LINE}
  \begin{csbul}
    \item KL Chain Rule: $\KL{p(x,y)}{q(x,y)}=\KL{p(x)}{q(x)}+\E_{p(x)}[\KL{p(y|x)}{q(y|x)}]$
    \item Typical Set (AEP): $\abs{A_\epsilon^{(N)}}\approx e^{N\Entropy(p)}$, $p(\vecb x_{1:N})\approx e^{-N\Entropy(p)}$
    \item Donsker--Varadhan: $\KL{p}{q}=\sup_\phi\{\E_p[\phi(X)]-\ln\E_q[e^{\phi(X)}]\}$
  \end{csbul}

\begin{cscard}{csG}{PITFALL CHECKLIST}
  \begin{csbul}
    \item \textbf{Disjoint-support blowup}: When $p^*$ and $q_\theta$ lie on distinct low-dimensional manifolds, $\KL{p^*}{q_\theta}=+\infty$ and $\nabla_\theta\JS{p^*}{q_\theta}=\vecb 0$. Remedy: add Gaussian noise jitter, label smoothing, or switch to Wasserstein IPMs.
    \item \textbf{Differential entropy traps}: $h(X)$ can be negative (e.g.\ $\Normal(0,\sigma^2)$ for $\sigma^2<(2\pi e)^{-1}$) and changes under scaling $h(aX)=h(X)+\ln\abs{a}$. Never minimize raw $h(T)$ without a reference prior $m(T)$ ($\KL{e(t|x)}{m(t)}$).
    \item \textbf{Deterministic continuous IB}: If $T=g_\theta(X)$ is deterministic on continuous $X$, $\MI(X;T)=+\infty$ and IB breaks. Always inject encoder stochasticity $e_\theta(t|x)=\Normal(\vecb\mu_\theta(x),\diag(\vecb\sigma_\theta^2(x)))$.
    \item \textbf{Log-domain NaN \& InfoNCE saturation}: Compute $-\E_p[\ln\softmax(\vecb z)]$ via $\mathrm{LogSumExp}$, mask $0\ln 0\mapsto 0$, and remember $\MI_{\mathrm{NCE}}\le\ln K$ (requires large negative batch size $K$).
  \end{csbul}
\end{cscard}

\begin{cscard}{csD}{10-SECOND VISUAL RECAP}
  \sffamily\footnotesize
  \textbf{Entropy} $\Entropy(p)$ counts log-microstates and sets the source-coding floor.
  \textbf{KL divergence} $\KL{p}{q}$ measures excess coding cost: forward $\KL{p}{q}$ matches moments and covers modes (MLE), while reverse $\KL{q}{p}$ seeks a single mode (VI).
  \textbf{DPI \& IB} show that Markov layers only lose information ($\MI(X;Y)\ge\MI(T;Y)$), so optimal representations compress nuisance $\MI(X;T\mid Y)\to 0$.
\end{cscard}
