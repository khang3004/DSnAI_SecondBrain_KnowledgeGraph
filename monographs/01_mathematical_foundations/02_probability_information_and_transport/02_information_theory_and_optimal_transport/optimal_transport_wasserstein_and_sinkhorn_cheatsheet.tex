\csbranch{2}{B}{2\;\; DUALITY \& WASSERSTEIN-$p$ $-$ $c$-transforms}{%
  \csfm{\mathrm{OT}_c(\mu,\nu)=\sup_{f(\vecb x)+g(\vecb y)\le c}\E_\mu[f]+\E_\nu[g]=\sup_{f\in\mathcal F_c}\E_\mu[f]+\E_\nu[f^c]}
  \begin{csbul}
    \item \textbf{$c$-transform:} $f^c(\vecb y)=\inf_{\vecb x}[c(\vecb x,\vecb y)-f(\vecb x)]$, $f^{c\bar cc}=f^c$
    \item \textbf{Wasserstein-$p$:} $\Wass[p]{\mu}{\nu}=(\inf_{\pi\in\Pi}\int d(\vecb x,\vecb y)^p\,\mathrm{d}\pi)^{1/p}$ (metric)
    \item \textbf{KR ($p=1$):} $\Wass[1]{\mu}{\nu}=\sup_{\norm{f}_{\mathrm{Lip}}\le 1}(\E_\mu[f]-\E_\nu[f])$ ($f^d=-f$)
    \item \textbf{1D quantile:} $\Wass[p]{\mu}{\nu}=\norm{F_\mu^{-1}-F_\nu^{-1}}_{L^p}$, $T^\star=F_\nu^{-1}\circ F_\mu$
    \item \textbf{Bures $W_2^2$:} $\norm{\vecb m_1-\vecb m_2}_2^2+\mathcal B^2(\mat\Sigma_1,\mat\Sigma_2)$ ($\mathcal B^2=\Tr(\mat\Sigma_1+\mat\Sigma_2-2(\mat\Sigma_1^{1/2}\mat\Sigma_2\mat\Sigma_1^{1/2})^{1/2})$)
  \end{csbul}}

\csbranch{3}{C}{3\;\; BRENIER \& DYNAMIC OT $-$ convex potentials}{%
  \begin{minipage}[t]{0.60\linewidth}
    \csfm{T^\star(\vecb x)=\nabla u(\vecb x),\quad u(\vecb x)=\tfrac12\norm{\vecb x}_2^2-f^\star(\vecb x)\;\text{convex}}
    \begin{csbul}
      \item \textbf{Monge--Amp\`ere}: $p_{\mathrm{tgt}}(\nabla u(\vecb x))\det\nabla^2 u(\vecb x)=p_{\mathrm{src}}(\vecb x)$
      \item \textbf{McCann geodesic}: $T_t^\star(\vecb x)=(1-t)\vecb x+t\nabla u(\vecb x)$
      \item \textbf{Benamou--Brenier}: $\Wass[2]{p_0}{p_1}^2=\inf_{p_t,\vecb v_t}\int_0^1\E_{p_t}[\norm{\vecb v_t}_2^2]\,\mathrm{d}t$ s.t.\ $\partial_t p_t+\nabla\cdot(p_t\vecb v_t)=0$ ($\vecb v_t=\nabla\varphi_t$)
    \end{csbul}

\csbranch{4}{D}{4\;\; ENTROPIC OT \& SINKHORN $-$ matrix scaling}{%
  \begin{minipage}[t]{0.60\linewidth}
    \csfm{\mathrm{EOT}_\varepsilon(\mu,\nu)=\inf_{\pi\in\Pi(\mu,\nu)}\inner{c}{\pi}+\varepsilon\,\KL{\pi}{\mu\otimes\nu}}
    \begin{csbul}
      \item \textbf{Gibbs plan}: $\mat\pi_\varepsilon^\star=\diag(e^{\vecb f^\star/\varepsilon})\mat K\diag(e^{\vecb g^\star/\varepsilon})$, $K_{ij}=e^{-C_{ij}/\varepsilon}$
      \item \textbf{Limits}: $\varepsilon\to 0^+\Rightarrow\mathrm{OT}_c$; $\varepsilon\to\infty\Rightarrow\mu\otimes\nu$
      \item \textbf{Schr\"odinger Bridge}: $\mathrm{SB}_\varepsilon=\min_P\KL{P}{R^\varepsilon}\equiv\mathrm{EOT}_\varepsilon$
      \item \textbf{Sinkhorn Div.:} $S_\varepsilon(\mu,\nu)=\mathrm{EOT}_\varepsilon(\mu,\nu)-\tfrac12\mathrm{EOT}_{\mu\mu}-\tfrac12\mathrm{EOT}_{\nu\nu}$ ($S_\varepsilon(\mu,\mu)=0$)
    \end{csbul}

\csbranch{6}{F}{6\;\; GENERATIVE ML \& FLOWS $-$ where OT rules}{%
  \begin{csbul}
    \item \textbf{Disjoint manifolds}: $\Wass[1]{\delta_0}{\delta_\theta}=\abs{\theta}$ gives clean gradients where $\KL{\delta_0}{\delta_\theta}=+\infty$ and $\nabla_\theta\JS{\delta_0}{\delta_\theta}=\vecb 0$
    \item \textbf{WGAN-GP / Spectral Norm}: approximates $W_1$ dual via $1$-Lipschitz critic $D_\phi$
    \item \textbf{Neural Brenier maps}: parameterize convex $u_\theta$ via Input-Convex Neural Nets (ICNNs)
    \item \textbf{Flow Matching}: straight paths $\Psi_t=(1-t)\vecb x_0+t\vecb x_1$ upper-bound $W_2^2$; pair with \texttt{m-OT} couplings
    \item \textbf{Diffusion PF-ODE}: solves a one-sided Schr\"odinger half-bridge ($\ne$ exact $W_2$ map if Hessians don't commute)
  \end{csbul}}

\csnote{8}{A}{TRAPS $-$ before you trust the output}{%
  \begin{csbul}
    \item \textbf{Kernel underflow:} $K_{ij}=e^{-C_{ij}/\varepsilon}\to 0$ for small $\varepsilon$; update $(f,g)$ via $\mathrm{LSE}$
    \item \textbf{Self-bias $\mathrm{EOT}_\varepsilon(\mu,\mu)>0$:} shrinks barycenters; use debiased $S_\varepsilon(\mu,\nu)$
    \item \textbf{Minibatch \texttt{m-OT}:} random batches force cross-mode edges; use \texttt{m-POT}
  \end{csbul}}

\begin{cscard}{csA}{FORMULAS TO REMEMBER}
  \begin{csbul}
    \item Kantorovich Primal--Dual \& $c$-Transform ($f^c(\vecb y)=\inf_{\vecb x}[c(\vecb x,\vecb y)-f(\vecb x)]$):
      \csfm{\mathrm{OT}_c(\mu,\nu)=\min_{\pi\in\Pi(\mu,\nu)}\inner{c}{\pi}=\sup_{f(\vecb x)+g(\vecb y)\le c(\vecb x,\vecb y)}\E_\mu[f]+\E_\nu[g]}
    \item Wasserstein-$1$ (KR Dual) \& Bures--Wasserstein-$2$ (Gaussians):
      \csfm{\Wass[1]{\mu}{\nu}=\sup_{\norm{f}_{\mathrm{Lip}}\le 1}(\E_\mu[f]-\E_\nu[f]),\quad \Wass[2]{\Normal_1}{\Normal_2}^2=\norm{\vecb m_1-\vecb m_2}_2^2+\mathcal B^2(\mat\Sigma_1,\mat\Sigma_2)}
    \item Brenier Map \& Benamou--Brenier Action ($c=\frac12\norm{\vecb x-\vecb y}_2^2$, $\partial_t p_t+\nabla\cdot(p_t\vecb v_t)=0$):
      \csfm{T^\star(\vecb x)=\nabla u(\vecb x),\qquad \Wass[2]{p_0}{p_1}^2=\inf_{(p_t,\vecb v_t)}\int_0^1\E_{p_t}[\norm{\vecb v_t}_2^2]\,\mathrm{d}t}
    \item Entropic OT Dual \& Debiased Sinkhorn Divergence ($K_{ij}=e^{-C_{ij}/\varepsilon}$, $\pi_{\varepsilon,ij}^\star=e^{(f_i^\star+g_j^\star-C_{ij})/\varepsilon}$):
      \csfm{\mathrm{EOT}_\varepsilon(\vecb a,\vecb b)=\max_{\vecb f,\vecb g}\vecb f\T\vecb a+\vecb g\T\vecb b-\varepsilon(e^{\vecb f/\varepsilon})\T\mat K e^{\vecb g/\varepsilon}+\mathrm{const}}
      \csfm{S_\varepsilon(\mu,\nu)=\mathrm{EOT}_\varepsilon(\mu,\nu)-\tfrac12\mathrm{EOT}_\varepsilon(\mu,\mu)-\tfrac12\mathrm{EOT}_\varepsilon(\nu,\nu)}
  \end{csbul}
\end{cscard}

\begin{cscard}{csB}{LOG-STABILIZED SINKHORN + ROUNDING RECIPE}
  \begin{enumerate}[leftmargin=1.2em, itemsep=1pt, topsep=1pt, parsep=0pt]
    \item Init $\vecb f^{(0)}=\zeros_n$, $\vecb g^{(0)}=\zeros_m$; $\mathrm{LSE}_j(z_{ij})=m_i+\ln\sum_j e^{z_{ij}-m_i}$.
    \item Row/Col: $f_i\leftarrow\varepsilon\ln a_i-\varepsilon\,\mathrm{LSE}_j((g_j\!-\!C_{ij})/\varepsilon)$, $g_j\leftarrow\varepsilon\ln b_j-\varepsilon\,\mathrm{LSE}_i((f_i\!-\!C_{ij})/\varepsilon)$.
    \item Iterate until $\norm{\tilde{\mat\pi}\ones-\vecb a}_1+\norm{\tilde{\mat\pi}\T\ones-\vecb b}_1\le\delta$ with $\tilde\pi_{ij}=e^{(f_i+g_j-C_{ij})/\varepsilon}$.
    \item Round $\tilde{\mat\pi}\to\bar{\mat\pi}\in\Pi(\vecb a,\vecb b)$ via row/col clip + rank-$1$ fix $\vecb e_r\vecb e_c\T/\norm{\vecb e_r}_1$.
  \end{enumerate}
\end{cscard}

  \begin{csbul}

\begin{cscard}{csG}{PITFALL CHECKLIST}
  \begin{csbul}
    \item \textbf{Multiplicative Sinkhorn underflow}: $\mat K=e^{-\mat C/\varepsilon}$ underflows float32 once $C_{ij}/\varepsilon\gtrsim 88$. Normalize $\mat C$ by $\mathrm{median}(\mat C)$, run log-domain $\mathrm{LSE}$ updates, and use $\varepsilon$-scaling warm starts.
    \item \textbf{Un-debiased $\mathrm{EOT}_\varepsilon$ shrinks modes}: Because $\mathrm{EOT}_\varepsilon(\mu,\mu)>0$, minimizing $\mu\mapsto\mathrm{EOT}_\varepsilon(\mu,\nu)$ collapses $\mu$ toward the mean of $\nu$. Always subtract self-terms via $S_\varepsilon(\mu,\nu)$.
    \item \textbf{Unbalanced mass \& minibatch distortion}: Standard OT fails if $\sum a_i\ne\sum b_j$ or outliers exist, and random minibatches (\texttt{m-OT}) force cross-cluster edges; use Partial/Unbalanced OT (\texttt{m-POT}).
    \item \textbf{WGAN clipping \& Flow non-optimality}: Weight clipping causes critic collapse (use spectral norm + GP), and standard diffusion PF-ODEs / Reflow do not guarantee the exact $W_2$ Brenier map $\nabla u$.
  \end{csbul}
\end{cscard}

\begin{cscard}{csD}{10-SECOND VISUAL RECAP}
  \sffamily\footnotesize
  \textbf{Kantorovich} relaxes deterministic Monge maps $T_\sharp\mu=\nu$ into convex couplings $\pi\in\Pi(\mu,\nu)$, whose dual potentials give $1$-Lipschitz $W_1$ critics and convex Brenier maps $T^\star=\nabla u$ for $W_2^2$.
  \textbf{Entropic regularization} $\varepsilon\,\KL{\pi}{\mu\otimes\nu}$ equates static OT to Schr\"odinger Bridges and enables $O(n^2)$ log-domain Sinkhorn scaling; debiasing ($S_\varepsilon$) removes self-entropy bias across all $\varepsilon$.
\end{cscard}
