\csbranch{1}{A}{1\;\; PROBABILITY SPACES \& JACOBIANS}{%
  \csfm{p_{\vecb{y}}(\vecb{y}) = p_{\vecb{x}}(\vecb{f}\inv(\vecb{y}))\abs{\det\jac_{\vecb{f}\inv}(\vecb{y})}}
  \begin{csbul}
    \item \textbf{Kolmogorov $(\Omega,\mathcal{F},P)$:} $P(A)\ge 0$, $P(\Omega)=1$, $P(\bigsqcup_i A_i)=\sum_i P(A_i)$
    \item \textbf{Sum \& Product:} $p(\vecb{x}_a)=\int p(\vecb{x}_a,\vecb{x}_b)\,\mathrm{d}\nu(\vecb{x}_b)$,\; $p(\vecb{x}_a,\vecb{x}_b)=p(\vecb{x}_a\mid\vecb{x}_b)p(\vecb{x}_b)$
    \item \textbf{1D PIT:} $U = F_X(X) \sim \Uniform(0,1)$ and $F_X\inv(U) \sim p_X$
    \item \textbf{MGF/CGF:} $M_{\vecb{X}}(\vecb{s})=\E[e^{\inner{\vecb{s}}{\vecb{X}}}]$; $K_{\vecb{X}}=\log M_{\vecb{X}}$ sums over $\indep$ RVs
  \end{csbul}}

\csbranch{3}{C}{3\;\; SCHUR CONDITIONING \& DUALITY}{%
  \csfm{\vecb{\mu}_{a\mid b} = \vecb{\mu}_a + \mat{\Sigma}_{ab}\mat{\Sigma}_{bb}\inv(\vecb{x}_b-\vecb{\mu}_b) = \vecb{\mu}_a - \mat{\Lambda}_{aa}\inv\mat{\Lambda}_{ab}(\vecb{x}_b-\vecb{\mu}_b)}
  \csfm{\mat{\Sigma}_{a\mid b} = \mat{\Lambda}_{aa}\inv = \mat{\Sigma}_{aa} - \mat{\Sigma}_{ab}\mat{\Sigma}_{bb}\inv\mat{\Sigma}_{ba}}
  \begin{csbul}
    \item \textbf{Marginalization} is $\mathcal{O}(1)$ in $\mat{\Sigma}$ ($p(\vecb{x}_a)=\Normal(\vecb{\mu}_a,\mat{\Sigma}_{aa})$)
    \item \textbf{Conditioning} is $\mathcal{O}(1)$ in precision $\mat{\Lambda}=\mat{\Sigma}\inv$ ($\mat{\Lambda}_{a\mid b}=\mat{\Lambda}_{aa}$)
    \item \textbf{Markov graph:} $\Lambda_{ij}=0 \iff X_i \indep X_j \mid \vecb{X}_{-ij}$
    \item \textbf{Linear-Gaussian Bayes:} $\mat{\Sigma}_{z\mid y}\inv = \mat{\Sigma}_z\inv + \mat{W}\T\mat{\Sigma}_y\inv\mat{W}$
  \end{csbul}}

\csbranch{4}{D}{4\;\; CANONICAL EXPONENTIAL FAMILIES}{%
  \begin{minipage}[t]{0.60\linewidth}
    \csfm{p(\vecb{x}\mid\vecb{\eta}) = h(\vecb{x})\exp\!\big(\inner{\vecb{\eta}}{\vecb{T}(\vecb{x})} - A(\vecb{\eta})\big)}
    \begin{csbul}
      \item \textbf{1st deriv (Mean):} $\grad_{\vecb{\eta}} A(\vecb{\eta}) = \E[\vecb{T}(\vecb{x})] = \vecb{m}$
      \item \textbf{2nd deriv (Fisher):} $\hess_{\vecb{\eta}} A(\vecb{\eta}) = \Cov[\vecb{T}(\vecb{x})] = \mat{I}_F(\vecb{\eta})$
      \item \textbf{Legendre dual:} $\vecb{\eta} = \grad A^*(\vecb{m})$ ($A^*$ neg.\ entropy)
      \item \textbf{MVN natural:} $\vecb{\eta}_1=\mat{\Lambda}\vecb{\mu}$, $\mat{\eta}_2=-\tfrac{1}{2}\mat{\Lambda}$
    \end{csbul}
  \end{minipage}\hfill

\begin{cscard}{csA}{FORMULAS TO REMEMBER}
  \begin{csbul}
    \item Change of variables ($\vecb{y}=\vecb{f}(\vecb{x})$):\csfm{p_{\vecb{y}}(\vecb{y}) = p_{\vecb{x}}\!\left(\vecb{f}\inv(\vecb{y})\right)\abs{\det\jac_{\vecb{f}\inv}(\vecb{y})}}
    \item MVN conditional $\vecb{x}_a\mid\vecb{x}_b \sim \Normal(\vecb{\mu}_{a\mid b},\mat{\Sigma}_{a\mid b})$:\csfm{\vecb{\mu}_{a\mid b} = \vecb{\mu}_a + \mat{\Sigma}_{ab}\mat{\Sigma}_{bb}\inv(\vecb{x}_b-\vecb{\mu}_b),\quad \mat{\Sigma}_{a\mid b} = \mat{\Sigma}_{aa} - \mat{\Sigma}_{ab}\mat{\Sigma}_{bb}\inv\mat{\Sigma}_{ba}}
    \item Linear-Gaussian Bayes ($\vecb{y}=\mat{W}\vecb{z}+\vecb{b}+\vecb{\varepsilon}$):\csfm{\mat{\Sigma}_{z\mid y}\inv = \mat{\Sigma}_z\inv + \mat{W}\T\mat{\Sigma}_y\inv\mat{W},\quad \vecb{\mu}_{z\mid y} = \mat{\Sigma}_{z\mid y}\big(\mat{W}\T\mat{\Sigma}_y\inv(\vecb{y}-\vecb{b})+\mat{\Sigma}_z\inv\vecb{\mu}_z\big)}
    \item Log-partition cumulants:\csfm{\grad_{\vecb{\eta}} A(\vecb{\eta}) = \E[\vecb{T}(\vecb{x})] = \vecb{m},\qquad \hess_{\vecb{\eta}} A(\vecb{\eta}) = \Cov[\vecb{T}(\vecb{x})] = \mat{I}_F(\vecb{\eta})}
    \item Conjugate prior \& log marginal evidence:\csfm{\vecb{\tau}_N = \vecb{\tau}_0 + \sum_{n=1}^N\vecb{T}(\vecb{x}_n),\quad \log p(\mathcal{D}) = \sum_{n=1}^N\log h(\vecb{x}_n) + \psi(\vecb{\tau}_N,\nu_N) - \psi(\vecb{\tau}_0,\nu_0)}
  \end{csbul}
\end{cscard}

\begin{cscard}{csB}{RECIPE: CHOLESKY SCHUR \& CONJUGATE UPDATE}
  \textbf{Cholesky-Stabilized MVN Conditioning:}
  \begin{enumerate}[leftmargin=1.2em, itemsep=1pt, topsep=1pt, parsep=0pt]
    \item Factor $\mat{L}_{bb} = \mathrm{Cholesky}(\mat{\Sigma}_{bb} + \epsilon\eye_{D_b})$ in $\mathcal{O}(D_b^3)$.
    \item Forward-solve $\mat{L}_{bb}\vecb{\alpha} = \vecb{x}_b - \vecb{\mu}_b$ and $\mat{L}_{bb}\mat{K}\T = \mat{\Sigma}_{ba}$.
    \item Return $\vecb{\mu}_{a\mid b} = \vecb{\mu}_a + \mat{K}\vecb{\alpha}$ and $\mat{\Sigma}_{a\mid b} = \mat{\Sigma}_{aa} - \mat{K}\mat{K}\T$ (never form $\mat{\Sigma}_{bb}\inv$!).
  \end{enumerate}
  \vspace{1pt}
  \textbf{Exponential-Family Conjugate Update:}
  \begin{enumerate}[leftmargin=1.2em, itemsep=1pt, topsep=1pt, parsep=0pt]
    \item Accumulate sufficient statistics $\vecb{S}_{\mathcal{D}} = \sum_{n=1}^N \vecb{T}(\vecb{x}_n)$.
    \item Update $\vecb{\tau}_N = \vecb{\tau}_0 + \vecb{S}_{\mathcal{D}}$, $\nu_N = \nu_0 + N$; posterior mean $\E[\vecb{m}\mid\mathcal{D}] = \vecb{\tau}_N/\nu_N$.
  \end{enumerate}
\end{cscard}

\begin{cscard}{csE}{KEY BOUNDS \& SCALING CHEAT-LINE}
  \begin{csbul}
    \item Correlation bound: $\abs{\Corr[X,Y]} = \abs{\Cov[X,Y]}/\sqrt{\Var[X]\Var[Y]} \le 1$
    \item Conditioning reduces variance: $\mat{\Sigma}_{a\mid b} = \mat{\Sigma}_{aa} - \mat{\Sigma}_{ab}\mat{\Sigma}_{bb}\inv\mat{\Sigma}_{ba} \preceq \mat{\Sigma}_{aa}$
    \item High-$D$ typical radius: $\norm{\vecb{x}-\vecb{\mu}}_2 = \sigma\sqrt{D} + \mathcal{O}_P(\sigma)$ as $D\to\infty$
    \item Log-MVN via Cholesky: $\log\abs{\mat{\Sigma}} = 2\sum_{i=1}^D \log L_{ii}$ (avoids $\det$ overflow)
  \end{csbul}
\end{cscard}

\begin{cscard}{csG}{PITFALL CHECKLIST}
  \begin{csbul}
    \item \textbf{Uncorrelated $\neq$ independent:} $X\sim\Normal(0,1)$ and $Y=\pm X$ have $\Cov[X,Y]=0$ yet $X^2=Y^2$; joint normality is required
    \item \textbf{Sample covariance collapse ($N\le D$):} $\hat{\mat{\Sigma}}_{\mathrm{MLE}}$ has $D-(N-1)$ zero eigenvalues; never invert without ridge/shrinkage
    \item \textbf{Evaluating $\det(\mat{\Sigma})$ directly:} underflows or overflows in float64 for $D\gtrsim 100$; always sum $2\sum_i\log L_{ii}$ in log space
    \item \textbf{Zero-count MLE divergence:} when empirical count $\bar{t}_k=0$, natural parameter $\hat{\eta}_k\to -\infty$ (black-swan trap); regularize with prior $\vecb{\tau}_0>\zeros$
    \item \textbf{Overcomplete exponential family:} linearly dependent $\vecb{T}(\vecb{x})$ makes $\hess A(\vecb{\eta})$ singular and $\vecb{\eta}$ non-unique
  \end{csbul}
\end{cscard}

\begin{cscard}{csD}{10-SECOND VISUAL RECAP}
  \sffamily\footnotesize
  \textbf{Two dual coordinates:} for Gaussians, $\mat{\Sigma}$ makes marginals free while $\mat{\Lambda}=\mat{\Sigma}\inv$ makes conditioning and Bayes rule additive.
  For any exponential family, $A(\vecb{\eta})$ is a convex cumulant machine: $\grad A=\E[\vecb{T}]$, $\hess A=\Cov[\vecb{T}]$, and Bayes updates just count $\vecb{T}(\vecb{x})$.
\end{cscard}
