\csbranch{1}{A}{1\;\; POLICY GRADIENT THEOREM $-$ no $\grad\Trans$ needed}{%
  \csfm{\grad_{\bm{\theta}} J(\bm{\theta}) = \E_{s\sim\mu,\,a\sim\pi_{\bm{\theta}}}\!\Big[\Adv^{\pi_{\bm{\theta}}}(s,a)\,\grad_{\bm{\theta}}\log\pi_{\bm{\theta}}(a\mid s)\Big]}
  \begin{minipage}[t]{0.61\linewidth}
    \begin{csbul}
      \item \textbf{Unrolling} $\grad\Vval^\pi(s)$ absorbs state shifts into $\mu(s)$
      \item \textbf{REINFORCE:} causal return-to-go $G_t=\sum_{k\ge 0}\gamma^k r_{t+k}$
      \item \textbf{Baseline invariance:} $\E_{a\sim\pi}[b(s)\grad\log\pi(a\mid s)]=\zeros$
      \item Optimal $b^*(s_t)\approx \Vval_{\bm{\phi}}(s_t) \Rightarrow \Adv^\pi=\Qval^\pi-\Vval^\pi$
    \end{csbul}
  \end{minipage}\hfill
  \begin{minipage}[t]{0.36\linewidth}\centering\vspace{2pt}
    \begin{tikzpicture}[scale=0.42, every node/.style={font=\tiny}]
      \draw[->, thick, gray] (-2,0)--(2,0) node[right]{$a$};

    \end{tikzpicture}
  \end{minipage}}

\csbranch{2}{B}{2\;\; ACTOR--CRITIC \& $\mathrm{GAE}(\gamma,\lambda)$ $-$ bias vs.\ variance}{%
  \csfm{\hat{A}_t^{\mathrm{GAE}(\gamma,\lambda)} = \sum_{l=0}^{\infty}(\gamma\lambda)^l \delta_{t+l}^V,\quad \delta_t^V = r_t + \gamma\Vval_{\bm{\phi}}(s_{t+1})-\Vval_{\bm{\phi}}(s_t)}
  \begin{csbul}
    \item \textbf{Backward recursion:} $\hat{A}_t = \delta_t^V + \gamma\lambda\hat{A}_{t+1}$ with $\hat{A}_T=0$
    \item $\lambda=0 \Rightarrow \hat{A}_t=\delta_t^V$ (1-step TD: low variance, critic bias)
    \item $\lambda=1 \Rightarrow \hat{A}_t=G_t-\Vval_{\bm{\phi}}(s_t)$ (Monte Carlo: unbiased, high variance)
    \item Value target $y_t = \hat{A}_t + \Vval_{\bm{\phi}}(s_t)$ trains critic via MSE $\frac{1}{2}(\Vval_{\bm{\phi}}(s_t)-y_t)^2$
  \end{csbul}}

\csbranch{3}{C}{3\;\; NPG, TRPO \& PPO $-$ monotonic trust regions}{%
  \csfm{L^{\mathrm{CLIP}}(\bm{\theta})=\E_t\!\Big[\min\!\big(r_t(\bm{\theta})\hat{A}_t,\,\mathrm{clip}(r_t(\bm{\theta}),1{-}\varepsilon,1{+}\varepsilon)\hat{A}_t\big)\Big]}
  \begin{minipage}[t]{0.62\linewidth}
    \begin{csbul}
      \item \textbf{NPG / TRPO:} $\Delta\bm{\theta}\propto F(\bm{\theta})\inv\grad J$ s.t.\ $\E_s[\KL{\pi_k}{\pi_{\bm{\theta}}}]\le\delta$
      \item \textbf{IS ratio:} $r_t(\bm{\theta})=\pi_{\bm{\theta}}(a_t\mid s_t)/\pi_{\bm{\theta}_{\mathrm{old}}}(a_t\mid s_t)$
      \item \textbf{Pessimistic bound:} zeroes gradient when $r_t$ exceeds $1{\pm}\varepsilon$ in the direction favored by $\hat{A}_t$
    \end{csbul}
  \end{minipage}\hfill
  \begin{minipage}[t]{0.35\linewidth}\centering\vspace{2pt}
    \begin{tikzpicture}[scale=0.41, every node/.style={font=\tiny}]

\csbranch{4}{D}{4\;\; CONTROL AS INFERENCE $-$ MaxEnt RL \& SAC}{%
  \csfm{p(o_t=1\mid s_t,a_t)=\exp(\alpha\inv r_t)\implies \max_{\pi}\E_\tau\!\left[\sum_t\big(r_t+\alpha\Entropy(\pi(\cdot\mid s_t))\big)\right]}
  \begin{csbul}
    \item Variational ELBO $-\KL{q_{\bm{\theta}}(\tau)}{p(\tau\mid o_{1:T}=1)}$ equals entropy-regularized return
    \item \textbf{Soft Actor-Critic (SAC):} off-policy twin critics $\min_{i=1,2}\Qval_{\bar{\mathbf{w}}_i}(s',a') - \alpha\log\pi_{\bm{\theta}}(a'\mid s')$
    \item \textbf{Reparameterization trick:} $a_t=f_{\bm{\theta}}(s_t,\bm{\xi}_t)$, $\bm{\xi}_t\sim\Normal(\zeros,\eye)$ propagates $\grad_a\Qval$ directly
  \end{csbul}}

\csbranch{5}{E}{5\;\; CRITIC-FREE LLM RL $-$ RLOO, GRPO \& GSPO}{%
  \begin{minipage}[t]{0.60\linewidth}
    \csfm{A_{i,t}^{\mathrm{GRPO}} = \frac{r_i - \mathrm{mean}(r_{1:G})}{\mathrm{std}(r_{1:G}) + 10^{-8}}}
    \begin{csbul}
      \item Sample $G$ completions $\{y_i\}_{i=1}^G \sim \pi_{\bm{\theta}_{\mathrm{old}}}(\cdot\mid x)$; no critic $\Vval_{\bm{\phi}}$!
      \item \textbf{Dr.\ GRPO} ($\tilde{A}_i=r_i-\mu_G$) $\equiv \frac{G-1}{G}A_i^{\mathrm{RLOO}}$
      \item \textbf{GSPO / CISPO:} sequence-level IS or clipped IS weights fix token-level variance in MoE LLMs
    \end{csbul}
  \end{minipage}\hfill
  \begin{minipage}[t]{0.37\linewidth}\centering\vspace{2pt}

\csnote{8}{A}{TRAPS $-$ before you trust the output}{%

\begin{cscard}{csA}{FORMULAS TO REMEMBER}
  \begin{csbul}
    \item Policy Gradient Theorem:\csfm{\grad_{\bm{\theta}} J(\bm{\theta}) = \E_{s\sim\mu,\,a\sim\pi_{\bm{\theta}}}\!\big[\Qval^{\pi_{\bm{\theta}}}(s,a)\,\grad_{\bm{\theta}}\log\pi_{\bm{\theta}}(a\mid s)\big]}
    \item Variance-Optimal Baseline:\csfm{b^*(s_t) = \frac{\E_{a_t}[G_t\norm{\grad_{\bm{\theta}}\log\pi_{\bm{\theta}}(a_t\mid s_t)}_2^2]}{\E_{a_t}[\norm{\grad_{\bm{\theta}}\log\pi_{\bm{\theta}}(a_t\mid s_t)}_2^2]}}
    \item Generalized Advantage ($\mathrm{GAE}(\gamma,\lambda)$):\csfm{\hat{A}_t = \sum_{l=0}^{\infty}(\gamma\lambda)^l\delta_{t+l}^V,\quad \delta_t^V = r_t+\gamma\Vval_{\bm{\phi}}(s_{t+1})-\Vval_{\bm{\phi}}(s_t)}
    \item Natural Policy Gradient \& Fisher Matrix:\csfm{F(\bm{\theta})=\E\!\big[\grad\log\pi_{\bm{\theta}}\,\grad\log\pi_{\bm{\theta}}\T\big],\quad \Delta\bm{\theta}\propto F(\bm{\theta})\inv\grad J}
    \item PPO Clipped Surrogate:\csfm{L^{\mathrm{CLIP}}(\bm{\theta})=\E_t\!\Big[\min\!\big(r_t(\bm{\theta})\hat{A}_t,\;\mathrm{clip}(r_t(\bm{\theta}),1{-}\varepsilon,1{+}\varepsilon)\hat{A}_t\big)\Big]}
    \item SAC Actor Objective (Reparameterized $a_t=f_{\bm{\theta}}(s_t,\bm{\xi}_t)$):\csfm{J_\pi(\bm{\theta})=\E_{s_t\sim\mathcal{D},\,\bm{\xi}_t\sim\Normal}\!\Big[\alpha\log\pi_{\bm{\theta}}(a_t\mid s_t)-\min_{i=1,2}\Qval_{\mathbf{w}_i}(s_t,a_t)\Big]}
  \end{csbul}
\end{cscard}

\begin{cscard}{csB}{PPO + GAE vs.\ GRPO RECIPE}
  \begin{enumerate}[leftmargin=1.2em, itemsep=1pt, topsep=1pt, parsep=0pt]
    \item \textbf{Rollout:} collect trajectories under $\pi_{\bm{\theta}_{\mathrm{old}}}$ (or $G$ completions per prompt $x$ in GRPO).
    \item \textbf{Advantages:}
      \emph{PPO:} backward sweep $\hat{A}_t=\delta_t^V+\gamma\lambda\hat{A}_{t+1}$, target $y_t=\hat{A}_t+\Vval_{\bm{\phi}}(s_t)$, normalize $\hat{A}_t$.
      \emph{GRPO:} group score $A_{i,t}=(r_i-\mu_G)/(\sigma_G+10^{-8})$.
    \item \textbf{Inner epochs ($m=1\dots M$):} compute $r_t(\bm{\theta})=\pi_{\bm{\theta}}/\pi_{\bm{\theta}_{\mathrm{old}}}$; ascend clipped surrogate $L^{\mathrm{CLIP}}$ ($-\beta\KL{\pi_{\bm{\theta}}}{\pi_{\mathrm{ref}}}$ in GRPO) and update $\bm{\phi}$ on $(\Vval_{\bm{\phi}}-y_t)^2$.
  \end{enumerate}
\end{cscard}

\begin{cscard}{csE}{KEY BOUNDS \& SCALING CHEAT-LINE}
  \begin{csbul}
    \item \textbf{TRPO Improvement Lower Bound:}
      $J(\pi)-J(\pi_k) \ge L(\pi,\pi_k) - \frac{2\gamma C_{\pi,\pi_k}}{(1-\gamma)^2}\E_{s}\!\big[D_{\mathrm{TV}}(\pi\|\pi_k)\big]$
    \item \textbf{Pinsker + Local KL:} $D_{\mathrm{TV}}(p,q)\le\sqrt{\frac{1}{2}\KL{q}{p}}$, $\E_s[\KL{\pi_{\bm{\theta}}}{\pi_{\bm{\theta}+\Delta\bm{\theta}}}]\approx\frac{1}{2}\Delta\bm{\theta}\T F(\bm{\theta})\Delta\bm{\theta}$
    \item \textbf{RLOO $\iff$ Dr.\ GRPO:} $r_i - \frac{1}{G}\sum_{j=1}^G r_j = \frac{G-1}{G}\big(r_i - \frac{1}{G-1}\sum_{j\ne i} r_j\big)$
  \end{csbul}
\end{cscard}

\begin{cscard}{csG}{PITFALL CHECKLIST}
  \begin{csbul}
    \item \textbf{Missing causality:} multiplying $\grad\log\pi_{\bm{\theta}}(a_t\mid s_t)$ by past rewards $r_{t'<t}$ has zero mean but adds $\mathcal{O}(T)$ variance; always use reward-to-go $G_t$.
    \item \textbf{Unnormalized advantages:} raw $\hat{A}_t$ scales shift across training; standardize $(\hat{A}_t-\mu_{\hat{A}})/(\sigma_{\hat{A}}+10^{-8})$ per minibatch in PPO.
    \item \textbf{GRPO $\sigma_G$ division bias:} dividing by group $\sigma_G$ upweights easy/impossible prompts where $\sigma_G\approx 0$, and normalizing by $|y_i|$ biases length; consider Dr.\ GRPO.
    \item \textbf{Token dropping in PPO/GRPO:} once $r_{i,t}\notin[1-\varepsilon,1+\varepsilon]$, gradient vanishes; use CISPO $\mathrm{sg}(\mathrm{clip}(r_t))\hat{A}_t\log\pi_{\bm{\theta}}$ or GSPO sequence ratios.
    \item \textbf{Critic bootstrapping without target nets:} off-policy Q-learning + neural nets diverges without Polyak targets $\bar{\mathbf{w}}$ and clipped double-Q $\min_{i=1,2}\Qval_{\bar{\mathbf{w}}_i}$.
  \end{csbul}
\end{cscard}

\begin{cscard}{csD}{10-SECOND VISUAL RECAP}
  \sffamily\footnotesize
  \textbf{Policy Gradient} turns RL into weighted maximum likelihood: push up $\log\pi_{\bm{\theta}}(a_t\mid s_t)$ in proportion to how much better $a_t$ was than expected ($\hat{A}_t$).
  \textbf{GAE($\gamma,\lambda$)} blends 1-step TD and Monte Carlo returns to control variance, while \textbf{PPO} and \textbf{GRPO} clip the importance ratio $r_t(\bm{\theta})$ so multi-epoch updates never step outside the local trust region.
\end{cscard}
