\csbranch{1}{A}{1\;\; WHAT $-$ Markov decision process}{%
  \begin{minipage}[t]{0.60\linewidth}
    \csfm{G_t = \sum_{k=0}^\infty \gamma^k R_{t+k+1} = R_{t+1} + \gamma G_{t+1}}
    \begin{csbul}
      \item Tuple $\langle\States, \Actions, p, R, \gamma, p_0\rangle$, $\gamma \in [0,1)$
      \item Markov: $p(s',r\mid s,a)$ independent of $H_{t-1}$
      \item $V^\pi(s) = \E_\pi[G_t\mid S_t=s]$, $Q^\pi(s,a)$
      \item Advantage $A^\pi(s,a) = Q^\pi(s,a) - V^\pi(s)$
    \end{csbul}

\csbranch{2}{B}{2\;\; BELLMAN $-$ expectation vs.\ optimality}{%
  \renewcommand{\arraystretch}{1.15}\scriptsize
  \begin{tabular}{@{}>{\raggedright\arraybackslash}p{1.35cm}>{\raggedright\arraybackslash}p{2.75cm}>{\raggedright\arraybackslash}p{2.75cm}@{}}
    & \textbf{\textcolor{csB}{Expectation $\Bellman^\pi$}} & \textbf{\textcolor{csB}{Optimality $\Bellman^*$}}\\ \hline
    Backup & $\sum_a \pi(a\mid s)\,Q^\pi(s,a)$ & $\max_a Q^*(s,a)$\\
    System & Linear: $(\eye{-}\gamma\mat{P}_\pi)\vecb{v}_\pi=\vecb{r}_\pi$ & Nonlinear fixed point\\
    Solution & $(\eye-\gamma\mat{P}_\pi)\inv\vecb{r}_\pi$ & VI / PI / LP\\
    Norm $\norm{\cdot}_\infty$ & $\gamma$-contraction to $V^\pi$ & $\gamma$-contraction to $V^*$\\
  \end{tabular}\par\vspace{2pt}
  \small\textbf{One-step lookahead:} knowing $Q^*(s,a)$ makes $\pi^*(s)=\argmax_a Q^*(s,a)$ model-free.}

\csbranch{3}{C}{3\;\; KEY THEOREMS $-$ Banach \& improvement}{%
  \begin{minipage}[t]{0.61\linewidth}
    \csfm{\norm{\Bellman^* U - \Bellman^* V}_\infty \le \gamma\norm{U - V}_\infty}
    \begin{csbul}
      \item \textbf{Banach:} unique $V^*$, $\norm{V_k-V^*}_\infty \le \gamma^k\norm{V_0-V^*}_\infty$
      \item \textbf{Policy Improvement:} if $Q^\pi(s,\pi'(s)) \ge V^\pi(s)\;\forall s$, then $V^{\pi'} \ge V^\pi$ componentwise
      \item Strict improvement until $\Bellman^* V^\pi = V^\pi = V^*$
    \end{csbul}
  \end{minipage}\hfill

\csbranch{4}{D}{4\;\; CONNECTIONS $-$ GPI, LP dual \& POMDPs}{%
  \begin{csbul}
    \item \textbf{GPI:} interleaves evaluation ($V\to V^\pi$) and improvement ($\pi\to\mathrm{greedy}(V)$)
    \item \textbf{Modified PI:} $m$ evaluation sweeps interpolate VI ($m=1$) and PI ($m=\infty$)
    \item \textbf{Primal LP:} $\min_{\vecb{v}} \vecb{p}_0\T\vecb{v}$ s.t.\ $\vecb{v} \ge \Bellman^*\vecb{v}$ ($|\States|$ vars, $|\States||\Actions|$ constraints)
    \item \textbf{Dual LP:} $\max_{d\ge 0}\sum_{s,a} d(s,a)R(s,a)$ over discounted occupancy measures
    \item \textbf{RTDP / Async DP:} backup along sampled trajectories without full sweeps
    \item \textbf{POMDP:} reduces to belief-state MDP over $b_t(w_t)=p(w_t\mid o_{1:t},a_{1:t-1})$
  \end{csbul}}

\csbranch{5}{E}{5\;\; COST $-$ exact planning algorithms}{%
  \renewcommand{\arraystretch}{1.15}\scriptsize

    \textbf{\textcolor{csE}{Algorithm}} & \textbf{\textcolor{csE}{Per-Iter Cost}} & \textbf{\textcolor{csE}{Convergence}}\\ \hline
    Policy Eval. & $O(|\States|^2)$ (iter) / $O(|\States|^3)$ & Linear $O(\gamma^k)$ / exact\\

  \end{tabular}\par\vspace{2pt}
  \small\textbf{Effective horizon:} $(1-\gamma)\inv$; as $\gamma\to 1$, VI slows down as $1/(1-\gamma)$.}

\begin{cscard}{csA}{FORMULAS TO REMEMBER}
  \begin{csbul}
    \item Bellman expectation ($V^\pi$ \& matrix form):\csfm{V^\pi(s) = \sum_{a}\pi(a\mid s)\sum_{s',r}p(s',r\mid s,a)\bigl[r+\gamma V^\pi(s')\bigr]}
    \item Exact Neumann inversion ($\gamma<1$):\csfm{\vecb{v}_\pi = (\eye - \gamma\mat{P}_\pi)\inv\vecb{r}_\pi = \sum_{t=0}^\infty \gamma^t\mat{P}_\pi^t\vecb{r}_\pi}
    \item Bellman optimality ($V^*$ \& $Q^*$):\csfm{V^*(s) = \max_{a\in\Actions(s)}\sum_{s',r}p(s',r\mid s,a)\bigl[r + \gamma V^*(s')\bigr]}
    \csfm{Q^*(s,a) = \sum_{s',r}p(s',r\mid s,a)\Bigl[r + \gamma\max_{a'}Q^*(s',a')\Bigr]}
    \item Advantage function:\csfm{A^\pi(s,a) = Q^\pi(s,a) - V^\pi(s),\quad \E_{a\sim\pi}[A^\pi(s,a)]=0}
  \end{csbul}
\end{cscard}

\begin{cscard}{csB}{POLICY ITERATION vs.\ VALUE ITERATION}
  \begin{enumerate}[leftmargin=1.2em, itemsep=1pt, topsep=1pt, parsep=0pt]
    \item \textbf{PI Step 1 (Eval):} sweep $V(s)\gets (\Bellman^\pi V)(s)$ until $\Delta<\theta$ (or solve $(\eye-\gamma\mat{P}_\pi)\vecb{v}_\pi=\vecb{r}_\pi$).
    \item \textbf{PI Step 2 (Improve):} $a^*\gets\argmax_a Q_V(s,a)$; if $Q_V(s,a^*)>Q_V(s,\pi(s))$ strictly, set $\pi(s)\gets a^*$.
    \item \textbf{VI Shortcut:} fuse into 1 sweep $V_{k+1}\gets\Bellman^* V_k$ until $\norm{V_k-V_{k-1}}_\infty < \frac{\epsilon(1-\gamma)}{2\gamma}$, then extract greedy $\pi$.
  \end{enumerate}
\end{cscard}

\begin{cscard}{csE}{CONTRACTION \& STOPPING BOUNDS}
  \begin{csbul}
    \item Geometric rate: $\norm{V_k - V^*}_\infty \le \gamma^k\norm{V_0 - V^*}_\infty$, horizon $(1-\gamma)\inv$
    \item Residual-to-true error: $\norm{V_k - V^*}_\infty \le \frac{\gamma}{1-\gamma}\norm{V_k - V_{k-1}}_\infty$
    \item Greedy policy loss: $\norm{V^{\pi_k} - V^*}_\infty \le \frac{2\gamma}{1-\gamma}\norm{V_k - V_{k-1}}_\infty$
  \end{csbul}
\end{cscard}

\begin{cscard}{csG}{PITFALL CHECKLIST}
  \begin{csbul}
    \item \textbf{Tie-breaking oscillation in PI:} if $Q_V(s,a_1)=Q_V(s,a_2)$ and $\argmax$ flips between $a_1,a_2$, $\text{policy\_stable}$ never triggers $\to$ require strict inequality $Q_V(s,a^*) > Q_V(s,\pi(s))$
    \item \textbf{Unscaled stopping threshold $\theta$:} at $\gamma=0.999$, $\Delta_k = 10^{-3}$ still allows $\norm{V_k-V^*}_\infty \approx 1.0$ $\to$ always scale $\theta$ by $(1-\gamma)/(2\gamma)$
    \item \textbf{Undiscounted continuing loops ($\gamma=1$):} $\mat{P}_\pi\ones=\ones \implies \eye-\mat{P}_\pi$ is non-invertible and $\Bellman^*$ is non-contractive
    \item \textbf{Reward-on-entry vs.\ transition:} if reward depends on $s'$, compute $R(s,a) = \sum_{s',r} r\,p(s',r\mid s,a)$ before forming $\vecb{r}_\pi$
    \item \textbf{In-place (Gauss--Seidel) vs.\ synchronous sweeps:} in-place $V(s)$ updates converge faster than two-array sweeps while preserving $\gamma$-contraction
  \end{csbul}
\end{cscard}

\begin{cscard}{csD}{10-SECOND VISUAL RECAP}
  \sffamily\footnotesize
  Every MDP is governed by two contractions: \textbf{$\Bellman^\pi$} (linear expectation) and \textbf{$\Bellman^*$} (nonlinear max-backup).
  \textbf{GPI} bounces between the evaluation line $V=V^\pi$ and the greedy curve $\pi=\mathrm{greedy}(V)$ until hitting the unique Banach fixed point $(V^*,\pi^*)$.
\end{cscard}
