\csbranch{1}{A}{1\;\; WHAT $-$ MC vs.\ TD(0) prediction}{%
  \begin{minipage}[t]{0.61\linewidth}
    \csfm{\delta_t = R_{t+1} + \gamma V(S_{t+1}) - V(S_t)}
    \begin{csbul}
      \item \textbf{MC:} targets full return $G_t$; zero bias, high variance, episodic only
      \item \textbf{TD(0):} $V(S_t)\gets V(S_t)+\alpha\delta_t$; online, biased, low variance
      \item \textbf{Batch:} MC $\to$ empirical MSE; TD(0) $\to$ certainty-equivalence MLE MDP
      \item \textbf{Off-policy IS:} $\rho_{t:T-1} = \prod_{k=t}^{T-1}\frac{\pi(A_k\mid S_k)}{b(A_k\mid S_k)}$
    \end{csbul}
  \end{minipage}\hfill

\csbranch{2}{B}{2\;\; CONTROL $-$ SARSA, Q-Learning \& Double Q}{%
  \renewcommand{\arraystretch}{1.15}\scriptsize
  \begin{tabular}{@{}>{\raggedright\arraybackslash}p{1.65cm}>{\raggedright\arraybackslash}p{3.65cm}>{\raggedright\arraybackslash}p{1.35cm}@{}}
    \textbf{\textcolor{csB}{Algorithm}} & \textbf{\textcolor{csB}{Bootstrap Target $Y_t$}} & \textbf{\textcolor{csB}{Type}}\\ \hline
    SARSA & $R_{t+1} + \gamma Q(S_{t+1}, A_{t+1})$ & On-policy\\
    Expected Sarsa & $R_{t+1} + \gamma \sum_{a'}\pi(a'\mid S_{t+1})Q(S_{t+1},a')$ & On / Off\\
    Q-Learning & $R_{t+1} + \gamma \max_{a'} Q(S_{t+1}, a')$ & Off-policy\\
    Double Q & $R_{t+1} + \gamma Q_2(S_{t+1},\argmax_{a'}Q_1)$ & Unbiased\\
  \end{tabular}\par\vspace{2pt}
  \small\textbf{Maximization bias:} Jensen $\E[\max_a X_a] \ge \max_a \E[X_a]$ $\Rightarrow$ split selection ($Q_1$) and evaluation ($Q_2$).}

\csbranch{3}{C}{3\;\; N-STEP \& TD($\lambda$) $-$ eligibility traces}{%
  \begin{minipage}[t]{0.62\linewidth}
    \csfm{G_t^\lambda = (1-\lambda)\sum_{n=1}^\infty \lambda^{n-1} G_{t:t+n}}
    \begin{csbul}
      \item $n$-step: $G_{t:t+n} = \sum_{k=1}^n \gamma^{k-1}R_{t+k} + \gamma^n V(S_{t+n})$
      \item \textbf{Trace:} $\vecb{z}_t = \gamma\lambda\vecb{z}_{t-1} + \nabla_{\vecb{w}}\hat{v}(S_t,\vecb{w}_t)$
      \item \textbf{Update:} $\vecb{w}_{t+1} = \vecb{w}_t + \alpha\delta_t\vecb{z}_t$
      \item $\lambda=0 \implies \mathrm{TD}(0)$; $\lambda=1 \implies \mathrm{MC}$
    \end{csbul}
  \end{minipage}\hfill
  \begin{minipage}[t]{0.35\linewidth}\centering\vspace{2pt}
    \begin{tikzpicture}[scale=0.38, every node/.style={font=\tiny}]
      \draw[->, gray] (0,0) -- (4.0,0) node[right] {$t$};
      \draw[->, gray] (0,0) -- (0,2.6) node[above] {$\vecb{z}_t$};
      \draw[csC, very thick] (0,0)--(0.5,0)--(0.5,1.8) plot[domain=0.5:2.2, samples=15] (\x, {1.8*exp(-1.1*(\x-0.5))}) -- (2.2,2.3) plot[domain=2.2:3.8, samples=15] (\x, {2.3*exp(-1.1*(\x-2.2))});

  \end{minipage}}

\csbranch{4}{D}{4\;\; DEADLY TRIAD \& Gradient-TD}{%
  \begin{minipage}[t]{0.62\linewidth}
    \begin{csbul}
      \item \textbf{Semi-grad TD:} $\vecb{w}_{t+1} = \vecb{w}_t + \alpha\delta_t\nabla\hat{v}(S_t,\vecb{w}_t)$
      \item Linear on-policy converges to $\mathrm{PBE}(\vecb{w}) = \norm{\Pi\overline{\delta}_{\vecb{w}}}_\mu^2 = 0$
      \item \textbf{Baird's 7-state MDP:} off-policy ($\mu\ne d^\pi$) + bootstrapping + FA $\implies \Pi\Bellman_\pi$ non-contractive, $\norm{\vecb{w}_t}\to\infty$
      \item \textbf{TDC / GTD2:} two-timescale $(\vecb{w}_t,\vecb{v}_t)$ true SGD on $\mathrm{PBE}$ in $O(d)$
    \end{csbul}
  \end{minipage}\hfill
  \begin{minipage}[t]{0.35\linewidth}\centering\vspace{2pt}

\csbranch{6}{F}{6\;\; DEEP Q-NETWORKS $-$ Rainbow stabilizers}{%
  \begin{csbul}
    \item \textbf{1.\ Experience Replay ($\mathcal{D}$ / PER):} priority $p_i=(\abs{\delta_i}+\epsilon)^\alpha$, IS weight $w_i=(NP(i))^{-\beta}$
    \item \textbf{2.\ Target Net ($\vecb{\theta}^-$):} frozen $C$ steps or Polyak $\vecb{\theta}^-\gets\tau\vecb{\theta}+(1-\tau)\vecb{\theta}^-$
    \item \textbf{3.\ Double DQN:} $y_i = r_i + \gamma Q_{\vecb{\theta}^-}\bigl(s'_i, \argmax_{a'} Q_{\vecb{\theta}}(s'_i,a')\bigr)$
    \item \textbf{4.\ Dueling ($V+A$):} $Q_{\vecb{\theta}}(s,a) = V_{\vecb{\theta}}(s) + A_{\vecb{\theta}}(s,a) - \frac{1}{|\Actions|}\sum_{a'}A_{\vecb{\theta}}(s,a')$
    \item \textbf{5.\ Rainbow / BBF:} $n=3$ step returns + C51 distributional + soft resets (high UTD)
    \item \textbf{6.\ HER:} relabel sparse-reward goals $g'\gets s_T$ inside replay buffer $\mathcal{D}$
  \end{csbul}}

\csnote{7}{G}{HOOK $-$ say it in one breath}{%
  \centering\normalsize\sffamily\bfseries\textcolor{csG}{TD Bootstraps One Step; Traces Bridge to MC; Replay + $\vecb{\theta}^-$ Tame the Triad} \\[2pt]
  \footnotesize\mdseries \textbf{Never use the same noisy network to pick $\argmax$ and evaluate it.}}

\csnote{8}{A}{TRAPS $-$ before you trust the output}{%
  \begin{csbul}
    \item Ordinary IS $\rho_{t:T-1}$ has $\infty$ variance $\to$ use Weighted IS
    \item Residual-gradient SGD has double-sampling bias $\to$ use PBE (TDC)
    \item Q-learning overestimates noisy actions $\to$ use Double Q / DDQN
    \item High UTD in DQN causes plasticity loss $\to$ layer norm + resets
  \end{csbul}}

\begin{cscard}{csA}{FORMULAS TO REMEMBER}
  \begin{csbul}
    \item TD(0) / SARSA / Q-Learning:\csfm{Q(S_t,A_t)\gets Q(S_t,A_t)+\alpha\bigl[Y_t - Q(S_t,A_t)\bigr]}
    \item Backward-view $\mathrm{TD}(\lambda)$:\csfm{\vecb{z}_t = \gamma\lambda\vecb{z}_{t-1} + \nabla_{\vecb{w}}\hat{v}(S_t,\vecb{w}_t),\quad \vecb{w}_{t+1} = \vecb{w}_t + \alpha\delta_t\vecb{z}_t}
    \item TDC (Gradient-TD off-policy):\csfm{\vecb{w}_{t+1} = \vecb{w}_t + \alpha\rho_t\bigl(\delta_t\vecb{x}_t - \gamma\vecb{x}_{t+1}(\vecb{x}_t\T\vecb{v}_t)\bigr)}
    \item Double DQN target:\csfm{y_i^{\mathrm{DDQN}} = r_i + \gamma(1-\mathrm{done}_i)\,Q_{\vecb{\theta}^-}\!\bigl(s'_i,\argmax_{a'}Q_{\vecb{\theta}}(s'_i,a')\bigr)}
    \item Dueling Q-head (identifiable):\csfm{Q_{\vecb{\theta}}(s,a) = V_{\vecb{\theta}}(s) + \Bigl(A_{\vecb{\theta}}(s,a) - \frac{1}{|\Actions|}\sum_{a'}A_{\vecb{\theta}}(s,a')\Bigr)}
  \end{csbul}
\end{cscard}

\begin{cscard}{csB}{DOUBLE DQN + PER TRAINING LOOP}
  \begin{enumerate}[leftmargin=1.2em, itemsep=1pt, topsep=1pt, parsep=0pt]
    \item Act $\epsilon$-greedily in $Q_{\vecb{\theta}}(s_t,\cdot)$; push $(s_t,a_t,r_{t+1},s_{t+1},\mathrm{done}_t)$ to $\mathcal{D}$ with $\max_j p_j$.
    \item Sample minibatch $i\sim P(i)=p_i^\alpha/\sum_j p_j^\alpha$; set $w_i=(NP(i))^{-\beta}/\max_j w_j$.
    \item Compute $y_i^{\mathrm{DDQN}}$ using $\vecb{\theta}$ to select $a^*$ and $\vecb{\theta}^-$ to evaluate; $\delta_i = y_i - Q_{\vecb{\theta}}(s_i,a_i)$.
    \item Update $p_i\gets (\abs{\delta_i}+\epsilon)^\alpha$, step $\vecb{\theta}$ on $\frac{1}{2B}\sum_i w_i\delta_i^2$, sync $\vecb{\theta}^-\gets\vecb{\theta}$ every $C$ steps.
  \end{enumerate}
\end{cscard}

\begin{cscard}{csE}{BIAS--VARIANCE \& ERROR CHEAT-LINE}
  \begin{csbul}
    \item TD fixed point (linear on-policy): $\norm{\hat{v}(\cdot,\vecb{w}_{\mathrm{TD}}) - v_\pi}_\mu \le \frac{1}{\sqrt{1-\gamma^2}}\min_{\vecb{w}}\norm{\hat{v}(\cdot,\vecb{w})-v_\pi}_\mu$
    \item Forward--backward equivalence: $\sum_{t=0}^{T-1}\delta_t\vecb{z}_t = \sum_{t=0}^{T-1}(G_t^\lambda - \vecb{w}\T\vecb{x}_t)\vecb{x}_t$ (offline)
    \item PER annealing: anneal IS exponent $\beta: \beta_0 \to 1.0$ by end of training
  \end{csbul}
\end{cscard}

\begin{cscard}{csG}{PITFALL CHECKLIST}
  \begin{csbul}
    \item \textbf{Deadly Triad divergence:} never combine off-policy data ($\mu\ne d^\pi$), bootstrapping, and semi-gradient updates without target nets + replay or TDC
    \item \textbf{Forgetting $(1-\mathrm{done}_i)$ mask:} bootstrapping from terminal states $s'_i$ injects phantom future value and corrupts Q-estimates
    \item \textbf{Unidentifiable Dueling heads:} $Q=V+A$ is unidentifiable up to a constant shift unless you subtract $\frac{1}{|\Actions|}\sum_{a'}A(s,a')$
    \item \textbf{Double-sampling bias in Bellman Error:} $\nabla\norm{\overline{\delta}_{\vecb{w}}}_\mu^2$ requires two independent samples of $S_{t+1}$ from $S_t$; single-sample SGD converges to the wrong point
    \item \textbf{Uncorrected PER bias:} non-uniform sampling skews the expectation unless weighted by $w_i=(NP(i))^{-\beta}$ with $\beta\to 1$
  \end{csbul}
\end{cscard}

\begin{cscard}{csD}{10-SECOND VISUAL RECAP}
  \sffamily\footnotesize
  \textbf{MC} waits for episode end (unbiased, noisy); \textbf{TD(0)} bootstraps in 1 step (fast, low variance); \textbf{$\mathrm{TD}(\lambda)$} blends all horizons via eligibility traces $\vecb{z}_t$.
  When scaling off-policy Q-learning to deep nets, break the \textbf{Deadly Triad} with \textbf{Replay ($\mathcal{D}$)}, \textbf{Target Nets ($\vecb{\theta}^-$)}, \textbf{Double Q}, and \textbf{Dueling ($V+A$)} heads.
\end{cscard}
